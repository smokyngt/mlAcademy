import numpy as np

def z_test(xa, na, xb, nb):
    p = (xa + xb) / (na + nb)
    se = np.sqrt(p * (1 - p) * (1 / na + 1 / nb))
    return float((xb / nb - xa / na) / se)

def false_positive_rate(n_sims, n_per_arm, looks, p, rng):
    A = rng.random((n_sims, n_per_arm)) < p
    B = rng.random((n_sims, n_per_arm)) < p
    ca = np.cumsum(A, axis=1)
    cb = np.cumsum(B, axis=1)
    rejected = np.zeros(n_sims, dtype=bool)
    for n in np.linspace(n_per_arm // looks, n_per_arm, looks).astype(int):
        xa, xb = ca[:, n - 1], cb[:, n - 1]
        pooled = (xa + xb) / (2 * n)
        se = np.sqrt(np.maximum(pooled * (1 - pooled) * 2 / n, 1e-12))
        z = (xb / n - xa / n) / se
        rejected |= np.abs(z) > 1.96
    return float(rejected.mean())

def thompson_pulls(p_a, p_b, horizon, rng):
    probs = np.array([p_a, p_b])
    succ = np.ones(2)
    fail = np.ones(2)
    pulls = np.zeros(2)
    for _ in range(horizon):
        arm = int(np.argmax(rng.beta(succ, fail)))
        reward = rng.random() < probs[arm]
        succ[arm] += reward
        fail[arm] += 1 - reward
        pulls[arm] += 1
    return pulls
