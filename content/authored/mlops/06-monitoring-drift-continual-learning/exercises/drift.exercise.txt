import numpy as np

def ks_statistic(a, b):
    a = np.sort(np.asarray(a, dtype=float))
    b = np.sort(np.asarray(b, dtype=float))
    grid = np.concatenate([a, b])
    fa = np.searchsorted(a, grid, side="right") / len(a)
    fb = np.searchsorted(b, grid, side="right") / len(b)
    return float(np.max(np.abs(fa - fb)))

def ks_pvalue(d, n, m):
    ne = n * m / (n + m)
    lam = (np.sqrt(ne) + 0.12 + 0.11 / np.sqrt(ne)) * d
    if lam < 1e-3:
        return 1.0
    k = np.arange(1, 101)
    p = 2 * np.sum((-1.0) ** (k - 1) * np.exp(-2 * k * k * lam * lam))
    return float(min(1.0, max(0.0, p)))

def psi(reference, new, bins=10, eps=1e-4):
    reference = np.asarray(reference, dtype=float)
    new = np.asarray(new, dtype=float)
    edges = np.quantile(reference, np.linspace(0, 1, bins + 1)[1:-1])
    pr = np.bincount(np.searchsorted(edges, reference, side="right"), minlength=bins) / len(reference)
    pn = np.bincount(np.searchsorted(edges, new, side="right"), minlength=bins) / len(new)
    pr = np.clip(pr, eps, None)
    pn = np.clip(pn, eps, None)
    return float(np.sum((pn - pr) * np.log(pn / pr)))

def drift_report(reference, new, alpha=0.01, psi_alert=0.2):
    d = ks_statistic(reference, new)
    p = ks_pvalue(d, len(reference), len(new))
    s = psi(reference, new)
    return {"ks": d, "p": p, "psi": s, "drifted": bool(p < alpha or s > psi_alert)}
