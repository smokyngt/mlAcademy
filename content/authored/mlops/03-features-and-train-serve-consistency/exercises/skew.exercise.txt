- `fit_scaler(X)` -> `(mean, std)` per column of the training matrix, ignoring NaN (`std` with `ddof=0`).

- `unseen_category_rate(train_cats, serve_cats)` -> share of serving values that never appeared in training.

import numpy as np

def fit_scaler(X):
    X = np.asarray(X, dtype=float)
    return np.nanmean(X, axis=0), np.nanstd(X, axis=0)

def apply_scaler(X, mean, std):
    return (np.asarray(X, dtype=float) - mean) / std

def time_split(timestamps, test_frac):
    order = np.argsort(timestamps, kind="stable")
    n_train = int(round(len(order) * (1 - test_frac)))
    return order[:n_train], order[n_train:]

def skew_check(train, serve, names, mean_z=0.5, miss_tol=0.02, oor_tol=0.05):
    train = np.asarray(train, dtype=float)
    serve = np.asarray(serve, dtype=float)
    report = {}
    for j, name in enumerate(names):
        a, b = train[:, j], serve[:, j]
        issues = []
        if abs(np.isnan(b).mean() - np.isnan(a).mean()) > miss_tol:
            issues.append("missing")
        a_ok, b_ok = a[~np.isnan(a)], b[~np.isnan(b)]
        if len(b_ok):
            sd = a_ok.std()
            if sd > 0 and abs(b_ok.mean() - a_ok.mean()) / sd > mean_z:
                issues.append("shift")
            outside = np.mean((b_ok < a_ok.min()) | (b_ok > a_ok.max()))
            if outside > oor_tol:
                issues.append("range")
        if issues:
            report[name] = sorted(issues)
    return report

def unseen_category_rate(train_cats, serve_cats):
    seen = set(train_cats)
    return float(np.mean([c not in seen for c in serve_cats]))
