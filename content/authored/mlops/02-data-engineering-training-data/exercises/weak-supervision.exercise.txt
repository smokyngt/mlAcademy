A labelling function (LF) votes `1`, `0`, or abstains (`-1`) on each example. Given a vote matrix `L` of shape `(n, m)`:

- `lf_stats(L)` -> `(coverage, conflict)`: `coverage` is an array of length `m` (share of rows where each LF does not abstain); `conflict` is the share of rows that contain both a 0 and a 1 vote.

import numpy as np

def lf_stats(L):
    L = np.asarray(L)
    cov = (L != -1).mean(axis=0)
    conflict = float(np.mean(((L == 0).any(axis=1)) & ((L == 1).any(axis=1))))
    return cov, conflict

def majority_vote(L):
    L = np.asarray(L)
    ones = (L == 1).sum(axis=1)
    zeros = (L == 0).sum(axis=1)
    out = np.full(len(L), -1)
    out[ones > zeros] = 1
    out[zeros > ones] = 0
    return out

def lf_accuracy(L, gold):
    L = np.asarray(L)
    gold = np.asarray(gold)
    acc = np.zeros(L.shape[1])
    for j in range(L.shape[1]):
        m = L[:, j] != -1
        acc[j] = np.mean(L[m, j] == gold[m]) if m.any() else 0.0
    return acc

def weighted_vote(L, acc):
    L = np.asarray(L)
    acc = np.clip(np.asarray(acc, dtype=float), 1e-3, 1 - 1e-3)
    w = np.log(acc / (1 - acc))
    sign = np.where(L == 1, 1.0, np.where(L == 0, -1.0, 0.0))
    return (sign @ w > 0).astype(int)
