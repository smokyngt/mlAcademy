Three small tools for reasoning about an ML system's requirements.

- `latency_stats(lat)` -> `(mean, p50, p90, p99)` for a list of latencies. Use the **nearest-rank** percentile: sort the values and return the one at 1-based rank `ceil(q/100 * n)`.
- `throughput_qps(batch_size, batch_ms)` -> queries per second if you serve `batch_size` queries per batch and one batch takes `batch_ms` milliseconds.
- `pareto_front(quality, engagement)` -> sorted list of indices of items that are not **dominated**: item `j` dominates `i` if it is at least as good on both scores and strictly better on one.
- `blended_ranking(quality, engagement, alpha, beta)` -> item indices sorted by `alpha*quality + beta*engagement`, best first (ties by lower index first).

import numpy as np

def latency_stats(lat):
    s = np.sort(np.asarray(lat, dtype=float))
    def pct(q):
        r = int(np.ceil(q / 100 * len(s)))
        return float(s[max(r, 1) - 1])
    return float(s.mean()), pct(50), pct(90), pct(99)

def throughput_qps(batch_size, batch_ms):
    return batch_size * 1000.0 / batch_ms

def pareto_front(quality, engagement):
    q = np.asarray(quality, dtype=float)
    e = np.asarray(engagement, dtype=float)
    out = []
    for i in range(len(q)):
        dominated = False
        for j in range(len(q)):
            if j != i and q[j] >= q[i] and e[j] >= e[i] and (q[j] > q[i] or e[j] > e[i]):
                dominated = True
                break
        if not dominated:
            out.append(i)
    return out

def blended_ranking(quality, engagement, alpha, beta):
    score = alpha * np.asarray(quality, dtype=float) + beta * np.asarray(engagement, dtype=float)
    return [int(i) for i in np.argsort(-score, kind="stable")]
