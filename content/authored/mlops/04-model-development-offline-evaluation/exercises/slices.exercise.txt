# Simpson's paradox with counts (correct, total) per group
A_grp = {"A": (81, 87), "B": (192, 263)}   # model A
B_grp = {"A": (234, 270), "B": (55, 80)}   # model B
def build(counts):
    y_true, y_pred, sl = [], [], []
    for g, (c, n) in counts.items():
        y_true += [1] * n
        y_pred += [1] * c + [0] * (n - c)
        sl += [g] * n
    return np.array(y_true), np.array(y_pred), np.array(sl)

import numpy as np
from math import comb

def slice_metrics(y_true, y_pred, slice_ids):
    y_true, y_pred, slice_ids = map(np.asarray, (y_true, y_pred, slice_ids))
    out = {}
    for s in np.unique(slice_ids):
        m = slice_ids == s
        key = s.item() if hasattr(s, "item") else s
        out[key] = (int(m.sum()), float(np.mean(y_true[m] == y_pred[m])))
    return out

def worst_slice(metrics, min_n):
    cand = [(acc, k) for k, (n, acc) in metrics.items() if n >= min_n]
    return min(cand, key=lambda t: (t[0], str(t[1])))[1]

def random_baseline(p, q):
    precision, recall = p, q
    f1 = 2 * precision * recall / (precision + recall)
    acc = p * q + (1 - p) * (1 - q)
    return f1, acc

def majority_accuracy(p, n):
    return sum(comb(n, k) * p ** k * (1 - p) ** (n - k) for k in range(n // 2 + 1, n + 1))
