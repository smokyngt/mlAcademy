#@@ prompt
Implement a small roofline calculator. Hardware is a dict `hw = {"peak": FLOP/s, "bw": bytes/s}`.

- `ridge(hw)`: the arithmetic intensity (FLOP/byte) where the roof turns flat.
- `attainable(hw, intensity)`: `min(peak, bw * intensity)`.
- `matmul_cost(M, N, K, elem_bytes)` -> `(flops, bytes)` assuming each matrix crosses memory once: `flops = 2*M*N*K`, `bytes = elem_bytes * (M*K + K*N + M*N)`.
- `softmax_cost(R, C, elem_bytes)` -> `(5*R*C, 2*R*C*elem_bytes)` (read once, write once, 5 FLOPs per element by our convention).
- `layernorm_cost(R, C, elem_bytes)` -> `(8*R*C, 2*R*C*elem_bytes)` (convention: 8 FLOPs per element, parameters ignored).
- `analyse(hw, ops)`: `ops` is a list of `(name, flops, bytes)`. Return a list of dicts with keys `name`, `intensity`, `bound` (`"memory"` if intensity is below the ridge, else `"compute"`), `time` (`max(flops/peak, bytes/bw)`), and `utilisation` (`flops / time / peak`).
- `amdahl(p, s)`: overall speed-up when a fraction `p` of the time is sped up by `s`. Also `amdahl_limit(p)` (as `s` goes to infinity) and `pipeline_speedup(times, speedups)`: stage `i` takes `times[i]` and is sped up by `speedups[i]`; return old total over new total.

import numpy as np

def ridge(hw):
    return hw["peak"] / hw["bw"]

def attainable(hw, intensity):
    return min(hw["peak"], hw["bw"] * intensity)

def matmul_cost(M, N, K, elem_bytes):
    return 2 * M * N * K, elem_bytes * (M * K + K * N + M * N)

def softmax_cost(R, C, elem_bytes):
    return 5 * R * C, 2 * R * C * elem_bytes

def layernorm_cost(R, C, elem_bytes):
    return 8 * R * C, 2 * R * C * elem_bytes

def analyse(hw, ops):
    out = []
    for name, flops, nbytes in ops:
        inten = flops / nbytes
        t = max(flops / hw["peak"], nbytes / hw["bw"])
        out.append({"name": name, "intensity": inten,
                    "bound": "memory" if inten < ridge(hw) else "compute",
                    "time": t, "utilisation": flops / t / hw["peak"]})
    return out

def amdahl(p, s):
    return 1.0 / ((1.0 - p) + p / s)

def amdahl_limit(p):
    return 1.0 / (1.0 - p)

def pipeline_speedup(times, speedups):
    return sum(times) / sum(t / s for t, s in zip(times, speedups))
