#@@ prompt
A single-head attention "language model" with a deterministic next-token rule: the next embedding is `tanh(out_last @ Wo)`, where `out_last` is the attention output of the last position.

Implement:

- `kv_cache_bytes(layers, kv_heads, d_head, seq_len, bytes_per=2, batch=1)`
- `flops_naive_step(t, d_model, d_head)` and `flops_cached_step(t, d_model, d_head)` with the counting convention of the lesson (a matmul of $(a,b)\times(b,c)$ costs $2abc$)
- `generate_naive` (recompute Q, K, V of every row at every step, causal mask) and `generate_cached` (compute k and v once per token and append them)

The two generators must agree to numerical precision. Then compare the total FLOPs.

import numpy as np

def softmax(x):
    z = x - x.max(axis=-1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def make_weights(d_model, d_head, seed=0):
    rng = np.random.default_rng(seed)
    s = 1 / np.sqrt(d_model)
    return {"Wq": rng.normal(0, s, (d_model, d_head)), "Wk": rng.normal(0, s, (d_model, d_head)),
            "Wv": rng.normal(0, s, (d_model, d_head)), "Wo": rng.normal(0, 1 / np.sqrt(d_head), (d_head, d_model))}

def kv_cache_bytes(layers, kv_heads, d_head, seq_len, bytes_per=2, batch=1):
    return 2 * layers * kv_heads * d_head * seq_len * bytes_per * batch

def flops_naive_step(t, d_model, d_head):
    return 6 * t * d_model * d_head + 2 * t * (t + 1) * d_head

def flops_cached_step(t, d_model, d_head):
    return 6 * d_model * d_head + 4 * t * d_head

def generate_naive(X0, steps, W):
    X = X0.copy()
    d_head = W["Wq"].shape[1]
    for _ in range(steps):
        Q, K, V = X @ W["Wq"], X @ W["Wk"], X @ W["Wv"]
        S = Q @ K.T / np.sqrt(d_head)
        S = np.where(np.tril(np.ones_like(S)) == 1, S, -np.inf)
        out = softmax(S) @ V
        X = np.vstack([X, np.tanh(out[-1] @ W["Wo"])])
    return X

def generate_cached(X0, steps, W):
    d_head = W["Wq"].shape[1]
    K, V = X0 @ W["Wk"], X0 @ W["Wv"]
    q = X0[-1] @ W["Wq"]
    rows = [r for r in X0]
    out = softmax(q @ K.T / np.sqrt(d_head)) @ V
    for _ in range(steps):
        nxt = np.tanh(out @ W["Wo"])
        rows.append(nxt)
        K = np.vstack([K, nxt @ W["Wk"]])
        V = np.vstack([V, nxt @ W["Wv"]])
        out = softmax((nxt @ W["Wq"]) @ K.T / np.sqrt(d_head)) @ V
    return np.array(rows)
