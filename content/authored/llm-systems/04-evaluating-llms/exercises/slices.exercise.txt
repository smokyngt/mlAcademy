import numpy as np
# model A: group A 81/87, group B 192/263 ; model B: group A 234/270, group B 55/80
cA = np.concatenate([np.ones(81), np.zeros(6), np.ones(192), np.zeros(71)])

cB = np.concatenate([np.ones(234), np.zeros(36), np.ones(55), np.zeros(25)])

import numpy as np

def overall_accuracy(correct):
    return float(np.mean(correct))

def slice_accuracy(correct, groups):
    correct = np.asarray(correct, float)
    groups = np.asarray(groups)
    return {str(g): float(correct[groups == g].mean()) for g in np.unique(groups)}

def simpson_reversal(cA, cB, gA, gB):
    sA, sB = slice_accuracy(cA, gA), slice_accuracy(cB, gB)
    a_wins_all = all(sA[g] > sB[g] for g in sA)
    b_wins_all = all(sB[g] > sA[g] for g in sA)
    return bool((a_wins_all and overall_accuracy(cB) > overall_accuracy(cA)) or
                (b_wins_all and overall_accuracy(cA) > overall_accuracy(cB)))
