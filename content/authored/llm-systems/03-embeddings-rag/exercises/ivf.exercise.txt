import numpy as np

def exact_knn(X, q, k):
    return np.argsort(((X - q) ** 2).sum(1))[:k]

def kmeans(X, k, iters, seed):
    rng = np.random.default_rng(seed)
    C = X[rng.choice(len(X), k, replace=False)].copy()
    for _ in range(iters):
        a = np.argmin(((X[:, None] - C[None]) ** 2).sum(-1), axis=1)
        for j in range(k):
            if (a == j).any():
                C[j] = X[a == j].mean(0)
    a = np.argmin(((X[:, None] - C[None]) ** 2).sum(-1), axis=1)
    return C, a

def ivf_search(q, X, centroids, assign, k, nprobe):
    probe = np.argsort(((centroids - q) ** 2).sum(1))[:nprobe]
    cand = np.where(np.isin(assign, probe))[0]
    d = ((X[cand] - q) ** 2).sum(1)
    return cand[np.argsort(d)[:k]], len(cand)

def recall_at_k(exact, approx):
    return len(set(map(int, exact)) & set(map(int, approx))) / len(exact)
