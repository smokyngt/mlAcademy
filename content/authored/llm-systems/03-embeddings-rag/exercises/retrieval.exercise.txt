import numpy as np
import re
from collections import Counter

def tokenize(text):
    return re.findall(r"[a-z0-9]+", text.lower())

def build_vocab(docs):
    return {w: i for i, w in enumerate(sorted({w for d in docs for w in tokenize(d)}))}

def tfidf_matrix(docs, vocab):
    N = len(docs)
    tf = np.zeros((N, len(vocab)))
    for i, d in enumerate(docs):
        for w, c in Counter(tokenize(d)).items():
            if w in vocab:
                tf[i, vocab[w]] = c
    df = (tf > 0).sum(axis=0)
    idf = np.log((1 + N) / (1 + df)) + 1
    X = tf * idf
    n = np.linalg.norm(X, axis=1, keepdims=True)
    return X / np.where(n == 0, 1, n), idf

def query_vector(q, vocab, idf):
    v = np.zeros(len(vocab))
    for w, c in Counter(tokenize(q)).items():
        if w in vocab:
            v[vocab[w]] = c * idf[vocab[w]]
    n = np.linalg.norm(v)
    return v / n if n > 0 else v

def cosine_search(qv, X, k):
    sims = X @ qv / (np.linalg.norm(X, axis=1) * (np.linalg.norm(qv) or 1) + 1e-12)
    order = np.argsort(-sims, kind="stable")[:k]
    return order, sims[order]

def bm25_scores(query, docs, k1=1.2, b=0.75):
    toks = [tokenize(d) for d in docs]
    N = len(docs)
    avgdl = np.mean([len(t) for t in toks])
    df = Counter(w for t in toks for w in set(t))
    scores = np.zeros(N)
    for i, t in enumerate(toks):
        c = Counter(t)
        for w in set(tokenize(query)):
            if w not in c:
                continue
            idf = np.log((N - df[w] + 0.5) / (df[w] + 0.5) + 1)
            tf = c[w]
            scores[i] += idf * tf * (k1 + 1) / (tf + k1 * (1 - b + b * len(t) / avgdl))
    return scores

def rrf(rankings, k=60):
    sc = {}
    for r in rankings:
        for pos, d in enumerate(r, 1):
            sc[d] = sc.get(d, 0.0) + 1.0 / (k + pos)
    return sorted(sc, key=lambda d: (-sc[d], d)), sc

def chunk_words(text, size, overlap):
    w = text.split()
    step = size - overlap
    out = []
    for s in range(0, len(w), step):
        out.append(" ".join(w[s:s + size]))
        if s + size >= len(w):
            break
    return out
