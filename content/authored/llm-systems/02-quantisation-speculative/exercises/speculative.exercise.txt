import numpy as np

def sample(probs, u):
    return int(min(np.searchsorted(np.cumsum(probs), u, side="right"), len(probs) - 1))

def residual(q, p):
    r = np.maximum(q - p, 0.0)
    s = r.sum()
    return q.copy() if s == 0 else r / s

def acceptance_rate(q, p):
    return float(np.minimum(q, p).sum())

def expected_tokens(alpha, n):
    return float(n + 1) if alpha >= 1 else (1 - alpha ** (n + 1)) / (1 - alpha)

def speedup(alpha, n, c):
    return expected_tokens(alpha, n) / (n * c + 1)

def spec_decode(Q, P, start, length, n_draft, rng):
    seq = [start]
    rounds = 0
    while len(seq) - 1 < length:
        rounds += 1
        drafts, prev = [], seq[-1]
        for _ in range(n_draft):
            d = sample(P[prev], rng.random())
            drafts.append(d)
            prev = d
        prev = seq[-1]
        accepted_all = True
        for d in drafts:
            q, p = Q[prev], P[prev]
            if rng.random() < min(1.0, q[d] / p[d]):
                seq.append(d)
                prev = d
            else:
                seq.append(sample(residual(q, p), rng.random()))
                accepted_all = False
                break
        if accepted_all:
            seq.append(sample(Q[prev], rng.random()))
    return seq[1:length + 1], rounds
