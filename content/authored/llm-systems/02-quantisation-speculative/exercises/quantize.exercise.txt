Implement the quantisers used in LLM deployment.

- `quantize_sym(x, bits=8, axis=None)`: symmetric, integers in $[-q_{max}, q_{max}]$ with $q_{max}=2^{b-1}-1$, scale $s=\max|x|/q_{max}$. With `axis=1` use one scale per row (per-channel).
- `quantize_asym(x, bits=8)`: unsigned integers $0..2^b-1$ with a scale and a **zero point**, so that real zero is represented exactly.
- the matching dequantisers and `mse`.

Then observe: an outlier row ruins the per-tensor scale for all other rows, per-channel scales fix it, and one-sided data (like ReLU outputs) wastes half the integer range under the symmetric scheme.

import numpy as np

def quantize_sym(x, bits=8, axis=None):
    qmax = 2 ** (bits - 1) - 1
    amax = np.max(np.abs(x), axis=axis, keepdims=axis is not None)
    scale = np.where(amax == 0, 1.0, amax / qmax)
    q = np.clip(np.round(x / scale), -qmax, qmax).astype(np.int32)
    return q, scale

def dequantize_sym(q, scale):
    return q * scale

def quantize_asym(x, bits=8):
    lo, hi = min(float(x.min()), 0.0), max(float(x.max()), 0.0)
    levels = 2 ** bits - 1
    scale = (hi - lo) / levels if hi > lo else 1.0
    zp = int(np.round(-lo / scale))
    q = np.clip(np.round(x / scale) + zp, 0, levels).astype(np.int32)
    return q, scale, zp

def dequantize_asym(q, scale, zp):
    return (q - zp) * scale

def mse(a, b):
    return float(np.mean((a - b) ** 2))
