You will build the harness for a small controlled experiment on the two-moons classifier (2-16-1 tanh MLP, minibatch SGD, 300 steps): does **learning-rate warmup** or the **initialisation scale** matter?

Implement:

- `lr_at(step, lr, warmup)`: linear warmup schedule.
- `train_mlp(...)`: one training run, fully determined by `seed`, returning test accuracy.
- `run_seeds(cfg, seeds)`: accuracies of one configuration over several seeds.
- `summarize(accs)`: mean and sample standard deviation (`ddof=1`).
- `paired_stats(a, b)`: for two configurations run with the **same** seeds: mean difference, std of the differences, paired t statistic and the number of seeds where `a` wins.
- `welch_t(a, b)`: the unpaired t statistic, to see how much power pairing buys.

The same seed means the same initial weights and the same minibatch order for both configurations (common random numbers), which is what makes the paired comparison sharp.

import numpy as np

def make_moons(n, noise, seed):
    rng = np.random.default_rng(seed)
    n1 = n // 2
    n2 = n - n1
    t1 = rng.uniform(0, np.pi, n1)
    t2 = rng.uniform(0, np.pi, n2)
    a = np.c_[np.cos(t1), np.sin(t1)]
    b = np.c_[1 - np.cos(t2), 0.5 - np.sin(t2)]
    X = np.vstack([a, b]) + rng.normal(0, noise, (n, 2))
    y = np.r_[np.zeros(n1), np.ones(n2)]
    p = rng.permutation(n)
    return X[p], y[p]

def sigmoid(z):
    return 1 / (1 + np.exp(-np.clip(z, -30, 30)))

Xtr, ytr = make_moons(200, 0.2, 0)
Xte, yte = make_moons(1000, 0.2, 1)

def lr_at(step, lr, warmup):
    return lr * min(1.0, (step + 1) / warmup) if warmup > 0 else lr

def train_mlp(seed, init_scale=1.0, warmup=0, lr=0.5, steps=300, hidden=16, batch=32):
    rng = np.random.default_rng(seed)
    d = Xtr.shape[1]
    W1 = rng.normal(0, init_scale / np.sqrt(d), (d, hidden))
    b1 = np.zeros(hidden)
    W2 = rng.normal(0, init_scale / np.sqrt(hidden), (hidden, 1))
    b2 = np.zeros(1)
    n = len(Xtr)
    for t in range(steps):
        idx = rng.integers(0, n, batch)
        x, y = Xtr[idx], ytr[idx][:, None]
        h = np.tanh(x @ W1 + b1)
        p = sigmoid(h @ W2 + b2)
        g = (p - y) / batch
        gW2 = h.T @ g
        gb2 = g.sum(0)
        gh = (g @ W2.T) * (1 - h ** 2)
        gW1 = x.T @ gh
        gb1 = gh.sum(0)
        eta = lr_at(t, lr, warmup)
        W1 -= eta * gW1
        b1 -= eta * gb1
        W2 -= eta * gW2
        b2 -= eta * gb2
    p = sigmoid(np.tanh(Xte @ W1 + b1) @ W2 + b2)[:, 0]
    return float(((p > 0.5) == yte).mean())

def run_seeds(cfg, seeds):
    return np.array([train_mlp(s, **cfg) for s in seeds])

def summarize(accs):
    accs = np.asarray(accs, float)
    return float(accs.mean()), float(accs.std(ddof=1))

def paired_stats(a, b):
    d = np.asarray(a, float) - np.asarray(b, float)
    n = len(d)
    sd = float(d.std(ddof=1))
    t = float(d.mean() / (sd / np.sqrt(n))) if sd > 0 else float("inf")
    return {"mean_diff": float(d.mean()), "sd_diff": sd, "t": t, "wins": int((d > 0).sum())}

def welch_t(a, b):
    a, b = np.asarray(a, float), np.asarray(b, float)
    return float((a.mean() - b.mean()) / np.sqrt(a.var(ddof=1) / len(a) + b.var(ddof=1) / len(b)))
