import numpy as np

def softmax(x):
    e = np.exp(x - x.max(axis=-1, keepdims=True))
    return e / e.sum(axis=-1, keepdims=True)

def attention(Q, K, V, mask=None):
    d_k = Q.shape[-1]
    S = Q @ np.swapaxes(K, -1, -2) / np.sqrt(d_k)
    if mask is not None:
        S = np.where(mask, -np.inf, S)
    A = softmax(S)
    return A @ V, A

def score_variance(d, n=20000, seed=0):
    rng = np.random.default_rng(seed)
    q = rng.normal(size=(n, d)); k = rng.normal(size=(n, d))
    s = (q * k).sum(axis=1)
    return float(s.var()), float((s / np.sqrt(d)).var())

X = np.array([[1., 0], [0, 1], [1, 1]])
Wq = np.array([[1., 1], [0, 1]]); Wk = np.array([[1., 0], [1, 1]]); Wv = np.array([[1., 0], [0, 2]])
out, A = attention(X @ Wq, X @ Wk, X @ Wv)
print(np.round(A, 4)); print(np.round(out, 4))
