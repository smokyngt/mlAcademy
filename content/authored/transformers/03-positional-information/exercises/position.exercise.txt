import numpy as np

def sinusoidal(T, d, base=10000.0):
    p = np.arange(T)[:, None]
    i = np.arange(d // 2)[None, :]
    ang = p / base ** (2 * i / d)
    pe = np.zeros((T, d))
    pe[:, 0::2] = np.sin(ang)
    pe[:, 1::2] = np.cos(ang)
    return pe

def rope(x, pos, base=10000.0):
    d = x.shape[-1]
    th = base ** (-2 * np.arange(d // 2) / d)
    ang = np.asarray(pos)[:, None] * th[None, :]
    c, s = np.cos(ang), np.sin(ang)
    out = np.empty_like(x, dtype=float)
    a, b = x[..., 0::2], x[..., 1::2]
    out[..., 0::2] = a * c - b * s
    out[..., 1::2] = a * s + b * c
    return out

def rel_shift_gap(d_h, shift, seed=0):
    rng = np.random.default_rng(seed)
    q = rng.normal(size=d_h); k = rng.normal(size=d_h)
    T = 10 + shift
    Q = rope(np.tile(q, (T, 1)), np.arange(T))
    K = rope(np.tile(k, (T, 1)), np.arange(T))
    worst = 0.0
    for m in range(10):
        for n in range(10):
            worst = max(worst, abs(Q[m] @ K[n] - Q[m + shift] @ K[n + shift]))
    return float(worst)

print(np.round(sinusoidal(4, 4), 4))
print(rel_shift_gap(8, 100))
