import numpy as np

def softmax(x):
    e = np.exp(x - x.max(axis=-1, keepdims=True))
    return e / e.sum(axis=-1, keepdims=True)

def split_heads(x, h):
    B, T, d = x.shape
    return x.reshape(B, T, h, d // h).transpose(0, 2, 1, 3)

def merge_heads(x):
    B, h, T, dh = x.shape
    return x.transpose(0, 2, 1, 3).reshape(B, T, h * dh)

def mha(X, Wq, Wk, Wv, Wo, h, causal=True, pad=None):
    B, T, d = X.shape
    Q, K, V = (split_heads(X @ W, h) for W in (Wq, Wk, Wv))
    S = Q @ K.transpose(0, 1, 3, 2) / np.sqrt(d // h)
    forbid = np.zeros((B, 1, T, T), dtype=bool)
    if causal:
        forbid = forbid | np.triu(np.ones((T, T), dtype=bool), k=1)
    if pad is not None:
        forbid = forbid | pad[:, None, None, :]
    A = softmax(np.where(forbid, -np.inf, S))
    return merge_heads(A @ V) @ Wo, A

def n_params(d, bias=False):
    return 4 * d * d + (4 * d if bias else 0)

rng = np.random.default_rng(0)
X = rng.normal(size=(2, 5, 8)); Ws = [rng.normal(size=(8, 8)) * 0.3 for _ in range(4)]
out, A = mha(X, *Ws, h=2)
print(out.shape, A.shape, n_params(768))
