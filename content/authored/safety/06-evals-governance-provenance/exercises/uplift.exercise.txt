import math

def risk_ratio(k1, n1, k0, n0):
    return (k1 / n1) / (k0 / n0)

def risk_ratio_ci(k1, n1, k0, n0, z=1.96):
    rr = risk_ratio(k1, n1, k0, n0)
    se = math.sqrt(1 / k1 - 1 / n1 + 1 / k0 - 1 / n0)
    return math.exp(math.log(rr) - z * se), math.exp(math.log(rr) + z * se)

def wilson_upper(k, n, z=1.96):
    p = k / n
    d = 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return c + h

def cannot_rule_out(k, n, threshold):
    return wilson_upper(k, n) >= threshold

def above_chance_z(k, n, chance):
    return (k - n * chance) / math.sqrt(n * chance * (1 - chance))

def unlearning_report(target_before, target_after, general_before, general_after, chance):
    return {
        "target_drop": target_before - target_after,
        "near_chance": target_after <= chance + 0.05,
        "general_loss": general_before - general_after,
    }
