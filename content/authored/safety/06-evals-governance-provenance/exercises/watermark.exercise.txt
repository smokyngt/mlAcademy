Token-level watermarking biases a model's sampling towards a pseudo-random **green list** that depends on the previous token and a secret key. A detector who knows the key counts green tokens and runs a z-test. The helper `is_green(prev, tok, key, gamma)` (given) is the keyed hash that defines the lists; `base_probs(V)` is a toy language model.

Implement:

- `green_fraction(tokens, key, gamma)`: fraction of the `n-1` consecutive pairs that are green.
- `watermark_z(tokens, key, gamma)`: the one-sided z-score `(g - gamma*T) / sqrt(T*gamma*(1-gamma))`.
- `p_value(z)`: one-sided normal tail probability.
- `generate(T, delta, rng, key, gamma)`: sample `T` tokens; at each step multiply the base probability of green tokens by `exp(delta)`, renormalise and sample. (`delta = 0` is ordinary unwatermarked sampling.)
- `detect(tokens, key, gamma, z_threshold)`: `True` if `z >= z_threshold`.

import math
import numpy as np

V = 500

def base_probs(V=V):
    p = np.arange(1, V + 1) ** -0.5
    return p / p.sum()

def is_green(prev, tok, key=42, gamma=0.5):
    x = (np.asarray(prev, dtype=np.uint64) * np.uint64(1000003) + np.asarray(tok, dtype=np.uint64) * np.uint64(10007) + np.uint64(key))
    x = (x * np.uint64(2654435761)) % np.uint64(2 ** 32)
    x = x ^ (x >> np.uint64(15))
    x = (x * np.uint64(0x45d9f3b)) % np.uint64(2 ** 32)
    x = x ^ (x >> np.uint64(13))
    return (x.astype(np.float64) / 2 ** 32) < gamma

def green_fraction(tokens, key=42, gamma=0.5):
    tokens = np.asarray(tokens)
    return float(is_green(tokens[:-1], tokens[1:], key, gamma).mean())

def watermark_z(tokens, key=42, gamma=0.5):
    tokens = np.asarray(tokens)
    gr = is_green(tokens[:-1], tokens[1:], key, gamma)
    T = len(gr)
    return float((gr.sum() - gamma * T) / math.sqrt(T * gamma * (1 - gamma)))

def p_value(z):
    return 0.5 * math.erfc(z / math.sqrt(2.0))

def generate(T, delta, rng, key=42, gamma=0.5):
    base = base_probs()
    allt = np.arange(V)
    toks = [int(rng.choice(V, p=base))]
    for _ in range(T - 1):
        gr = is_green(np.full(V, toks[-1]), allt, key, gamma)
        w = base * np.exp(delta * gr)
        w = w / w.sum()
        toks.append(int(rng.choice(V, p=w)))
    return np.array(toks)

def detect(tokens, key=42, gamma=0.5, z_threshold=4.0):
    return watermark_z(tokens, key, gamma) >= z_threshold
