import numpy as np

def make_items(T, seed):
    rng = np.random.default_rng(seed)
    return rng.normal(size=T), rng.normal(size=T)

def human_accuracy(d, k, sigma, rng):
    votes = np.sign(d[:, None] + sigma * rng.normal(size=(len(d), k))).sum(axis=1)
    return float(np.mean(np.sign(votes) == np.sign(d)))

def ai_accuracy(d, l, c):
    return float(np.mean(np.sign(d + c * l) == np.sign(d)))
