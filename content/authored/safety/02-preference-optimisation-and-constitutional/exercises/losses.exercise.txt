Implement the per-example preference losses on sequence log-probabilities.

- `sigmoid(z)`.
- `dpo_loss(lp_w, lp_l, ref_w, ref_l, beta)`: `-log sigmoid(beta*((lp_w-ref_w)-(lp_l-ref_l)))`.
- `dpo_grad(lp_w, lp_l, ref_w, ref_l, beta)` -> `(dL/dlp_w, dL/dlp_l)`.
- `ipo_loss(lp_w, lp_l, ref_w, ref_l, beta)`: squared distance of the log-ratio difference from `1/(2*beta)`.
- `simpo_loss(lp_w, lp_l, len_w, len_l, beta, gamma)`: `-log sigmoid(beta*lp_w/len_w - beta*lp_l/len_l - gamma)`; no reference model.
- `kto_loss(lp, ref, desirable, beta, z_ref)`: with `r = lp - ref`, return `1 - sigmoid(beta*r - z_ref)` if `desirable` else `1 - sigmoid(z_ref - beta*r)` (a simplified KTO value, without the loss-aversion weights).
- `orpo_or_loss(p_w, p_l)`: `-log sigmoid(log(odds(p_w)/odds(p_l)))` where `odds(p) = p/(1-p)`.
- `implicit_reward(pi, pi_ref, beta)`: `beta*log(pi/pi_ref)` elementwise.

import numpy as np

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-z))

def dpo_loss(lp_w, lp_l, ref_w, ref_l, beta):
    m = beta * ((lp_w - ref_w) - (lp_l - ref_l))
    return float(-np.log(sigmoid(m)))

def dpo_grad(lp_w, lp_l, ref_w, ref_l, beta):
    m = beta * ((lp_w - ref_w) - (lp_l - ref_l))
    s = sigmoid(-m)
    return (-beta * s, beta * s)

def ipo_loss(lp_w, lp_l, ref_w, ref_l, beta):
    d = (lp_w - ref_w) - (lp_l - ref_l)
    return float((d - 1.0 / (2.0 * beta)) ** 2)

def simpo_loss(lp_w, lp_l, len_w, len_l, beta, gamma):
    m = beta * lp_w / len_w - beta * lp_l / len_l - gamma
    return float(-np.log(sigmoid(m)))

def kto_loss(lp, ref, desirable, beta, z_ref):
    r = lp - ref
    if desirable:
        return float(1.0 - sigmoid(beta * r - z_ref))
    return float(1.0 - sigmoid(z_ref - beta * r))

def orpo_or_loss(p_w, p_l):
    lo = lambda p: np.log(p / (1.0 - p))
    return float(-np.log(sigmoid(lo(p_w) - lo(p_l))))

def implicit_reward(pi, pi_ref, beta):
    return beta * np.log(np.asarray(pi) / np.asarray(pi_ref))
