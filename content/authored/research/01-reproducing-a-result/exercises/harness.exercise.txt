You are reproducing a published number. Build the pieces of a minimal harness (the Student critical value `t_crit` is given).

- `run_seeds(fn, seeds)` -> 1-D float array `[fn(s) for s in seeds]`.
- `summarize(x)` -> dict with `n`, `mean`, `sd` (sample standard deviation, `ddof=1`), `lo`, `hi` where `[lo, hi]` is the 95 percent Student-t confidence interval of the mean: `mean +- t_crit(n-1) * sd / sqrt(n)`.
- `paired_summary(a, b)` -> `summarize(a - b)` for two arrays produced with the same seeds.
- `compare_to_reported(x, reported, tol)` -> one of
  - `"reproduced"`: the reported number lies inside the 95 percent interval of your runs;
  - `"within_tolerance"`: it lies outside the interval but `abs(mean - reported) <= tol`;
  - `"not_reproduced"`: otherwise.

import numpy as np

T975 = [12.706, 4.303, 3.182, 2.776, 2.571, 2.447, 2.365, 2.306, 2.262, 2.228,
        2.201, 2.179, 2.160, 2.145, 2.131, 2.120, 2.110, 2.101, 2.093, 2.086,
        2.080, 2.074, 2.069, 2.064, 2.060, 2.056, 2.052, 2.048, 2.045, 2.042]

def t_crit(df):
    if df <= 30:
        return T975[df - 1]
    z = 1.959963984540054
    return z + (z**3 + z) / (4 * df) + (5 * z**5 + 16 * z**3 + 3 * z) / (96 * df**2)

def run_seeds(fn, seeds):
    return np.array([fn(s) for s in seeds], dtype=float)

def summarize(x):
    x = np.asarray(x, dtype=float)
    n = len(x)
    mean = float(x.mean())
    sd = float(x.std(ddof=1))
    half = t_crit(n - 1) * sd / np.sqrt(n)
    return {"n": n, "mean": mean, "sd": sd, "lo": mean - half, "hi": mean + half}

def paired_summary(a, b):
    return summarize(np.asarray(a, dtype=float) - np.asarray(b, dtype=float))

def compare_to_reported(x, reported, tol):
    s = summarize(x)
    if s["lo"] <= reported <= s["hi"]:
        return "reproduced"
    if abs(s["mean"] - reported) <= tol:
        return "within_tolerance"
    return "not_reproduced"
