#@@ prompt
Two checkers every reproduction needs.

**1. `determinism_check(fn, seed, repeats=3)`** calls `fn(seed)` `repeats` times and returns a dict `{"deterministic": bool, "max_abs_diff": float}`. `fn` returns a number or an array. `deterministic` is true only when every repeat equals the first output **exactly** (difference 0.0). `max_abs_diff` is the largest absolute difference between any repeat and the first (0.0 if identical).

**2. `report_completeness(report, required)`** takes a reproduction report (a dict) and the list of required field names. A field counts as *missing* when its key is absent, its value is `None`, or it is an empty string, list or dict (an honest report writes `"none"` rather than an empty list of deviations). Return `{"missing": [...], "score": float}` where `missing` keeps the order of `required` and `score = 1 - len(missing) / len(required)`.

import numpy as np

def determinism_check(fn, seed, repeats=3):
    first = np.asarray(fn(seed), dtype=float)
    worst = 0.0
    for _ in range(repeats - 1):
        out = np.asarray(fn(seed), dtype=float)
        worst = max(worst, float(np.max(np.abs(out - first))))
    return {"deterministic": worst == 0.0, "max_abs_diff": worst}

def report_completeness(report, required):
    missing = []
    for k in required:
        v = report.get(k)
        if v is None or (isinstance(v, (str, list, dict)) and len(v) == 0):
            missing.append(k)
    return {"missing": missing, "score": 1 - len(missing) / len(required)}
