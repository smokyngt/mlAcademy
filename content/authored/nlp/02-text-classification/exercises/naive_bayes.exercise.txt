import numpy as np

def train_nb(docs, labels, alpha=1.0):
    vocab = sorted({w for d in docs for w in d})
    ix = {w: i for i, w in enumerate(vocab)}
    classes = sorted(set(labels))
    log_prior = np.zeros(len(classes))
    log_theta = np.zeros((len(classes), len(vocab)))
    for c, cls in enumerate(classes):
        cls_docs = [d for d, l in zip(docs, labels) if l == cls]
        log_prior[c] = np.log(len(cls_docs) / len(docs))
        counts = np.zeros(len(vocab))
        for d in cls_docs:
            for w in d:
                counts[ix[w]] += 1
        log_theta[c] = np.log((counts + alpha) / (counts.sum() + alpha * len(vocab)))
    return classes, vocab, log_prior, log_theta

def nb_scores(doc, vocab, log_prior, log_theta):
    ix = {w: i for i, w in enumerate(vocab)}
    s = np.array(log_prior, dtype=float)
    for w in doc:
        if w in ix:
            s = s + log_theta[:, ix[w]]
    return s

def nb_predict(doc, classes, vocab, log_prior, log_theta):
    return classes[int(np.argmax(nb_scores(doc, vocab, log_prior, log_theta)))]

def nb_proba(scores):
    e = np.exp(scores - np.max(scores))
    return e / e.sum()

def prf(y_true, y_pred, positive):
    tp = sum(1 for t, p in zip(y_true, y_pred) if t == positive and p == positive)
    fp = sum(1 for t, p in zip(y_true, y_pred) if t != positive and p == positive)
    fn = sum(1 for t, p in zip(y_true, y_pred) if t == positive and p != positive)
    P = tp / (tp + fp) if tp + fp else 0.0
    R = tp / (tp + fn) if tp + fn else 0.0
    F = 2 * P * R / (P + R) if P + R else 0.0
    return P, R, F
