import numpy as np

def viterbi(obs, log_pi, log_A, log_B):
    n, K = len(obs), len(log_pi)
    delta = np.full((n, K), -np.inf)
    back = np.zeros((n, K), dtype=int)
    delta[0] = log_pi + log_B[:, obs[0]]
    for i in range(1, n):
        cand = delta[i - 1][:, None] + log_A
        back[i] = cand.argmax(axis=0)
        delta[i] = cand.max(axis=0) + log_B[:, obs[i]]
    path = [int(delta[-1].argmax())]
    for i in range(n - 1, 0, -1):
        path.append(int(back[i, path[-1]]))
    return path[::-1], float(delta[-1].max())

def estimate_hmm(sents, K, V, alpha):
    pi = np.zeros(K) + alpha
    A = np.zeros((K, K)) + alpha
    B = np.zeros((K, V)) + alpha
    for s in sents:
        pi[s[0][1]] += 1
        for i, (w, t) in enumerate(s):
            B[t, w] += 1
            if i > 0:
                A[s[i - 1][1], t] += 1
    return pi / pi.sum(), A / A.sum(axis=1, keepdims=True), B / B.sum(axis=1, keepdims=True)

def bio_spans(tags):
    spans = []
    cur = None
    for i, tag in enumerate(tags):
        if tag == "O":
            if cur:
                spans.append((cur[0], cur[1], i))
                cur = None
            continue
        prefix, typ = tag.split("-", 1)
        if prefix == "B" or cur is None or cur[0] != typ:
            if cur:
                spans.append((cur[0], cur[1], i))
            cur = (typ, i)
    if cur:
        spans.append((cur[0], cur[1], len(tags)))
    return spans
