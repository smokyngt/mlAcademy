import re
import numpy as np
from collections import Counter

def best_span(start, end, max_len):
    T = len(start)
    best = (0, 0, -np.inf)
    for i in range(T):
        for j in range(i, min(i + max_len, T)):
            sc = start[i] + end[j]
            if sc > best[2]:
                best = (i, j, float(sc))
    return best

def normalize(text):
    text = re.sub(r"[^\w\s]", " ", text.lower())
    return [t for t in text.split() if t not in ("a", "an", "the")]

def exact_match(pred, gold):
    return 1.0 if normalize(pred) == normalize(gold) else 0.0

def token_f1(pred, gold):
    p, g = normalize(pred), normalize(gold)
    if not p and not g:
        return 1.0
    o = sum((Counter(p) & Counter(g)).values())
    if o == 0:
        return 0.0
    P, R = o / len(p), o / len(g)
    return 2 * P * R / (P + R)
