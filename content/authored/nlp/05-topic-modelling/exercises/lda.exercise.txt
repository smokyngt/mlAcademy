Implement a minimal LDA and its read-outs.

- `conditional(ndk_d, nkw_w, nk, alpha, beta, V)` -> vector `(K,)` summing to 1: the probability of each topic for the current token, given counts that EXCLUDE that token (`ndk_d`: tokens of this document per topic, `nkw_w`: how often this word is assigned to each topic, `nk`: tokens per topic).
- `lda_gibbs(docs, K, V, alpha, beta, iters, seed)` -> `(ndk, nkw)` with shapes `(D, K)` and `(K, V)` after `iters` sweeps. Use `rng = np.random.default_rng(seed)`; initialise topics at random.
- `estimate(ndk, nkw, alpha, beta)` -> `(theta, phi)` with shapes `(D, K)` and `(K, V)`.
- `top_words(phi, n)` -> list of K lists with the ids of the `n` most probable words of each topic.
- `topic_diversity(tops)` -> distinct words over all topics divided by the total number of listed words.

import numpy as np

def conditional(ndk_d, nkw_w, nk, alpha, beta, V):
    p = (ndk_d + alpha) * (nkw_w + beta) / (nk + V * beta)
    return p / p.sum()

def lda_gibbs(docs, K, V, alpha, beta, iters, seed):
    rng = np.random.default_rng(seed)
    ndk = np.zeros((len(docs), K))
    nkw = np.zeros((K, V))
    nk = np.zeros(K)
    z = [rng.integers(K, size=len(d)) for d in docs]
    for d, doc in enumerate(docs):
        for i, w in enumerate(doc):
            k = z[d][i]
            ndk[d, k] += 1
            nkw[k, w] += 1
            nk[k] += 1
    for _ in range(iters):
        for d, doc in enumerate(docs):
            for i, w in enumerate(doc):
                k = z[d][i]
                ndk[d, k] -= 1
                nkw[k, w] -= 1
                nk[k] -= 1
                k = rng.choice(K, p=conditional(ndk[d], nkw[:, w], nk, alpha, beta, V))
                z[d][i] = k
                ndk[d, k] += 1
                nkw[k, w] += 1
                nk[k] += 1
    return ndk, nkw

def estimate(ndk, nkw, alpha, beta):
    K = ndk.shape[1]
    V = nkw.shape[1]
    theta = (ndk + alpha) / (ndk.sum(axis=1, keepdims=True) + K * alpha)
    phi = (nkw + beta) / (nkw.sum(axis=1, keepdims=True) + V * beta)
    return theta, phi

def top_words(phi, n):
    return [list(np.argsort(-row)[:n]) for row in phi]

def topic_diversity(tops):
    flat = [int(w) for t in tops for w in t]
    return len(set(flat)) / len(flat)
