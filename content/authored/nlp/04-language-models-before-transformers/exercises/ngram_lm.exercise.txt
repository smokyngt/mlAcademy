Implement a bigram language model on whitespace-tokenised sentences.

- `build_counts(sentences)` -> `(uni, bi, ctx)` as `collections.Counter`s. Pad each sentence with `<s>` and `</s>`. `uni` counts predicted tokens (everything after `<s>`, including `</s>`), `ctx` counts tokens used as context (everything except `</s>`), `bi` counts adjacent pairs.
- `p_mle(a, b, uni, bi, ctx)` = `c(a,b)/c(a)` (0 if the context is unseen).
- `p_addk(a, b, uni, bi, ctx, k)` using `|V| = len(uni)`.
- `p_interp(a, b, uni, bi, ctx, lam)` = `lam * p_mle + (1 - lam) * unigram probability`.
- `p_stupid(a, b, uni, bi, ctx, alpha=0.4)`: the MLE bigram estimate if the bigram was seen, else `alpha` times the unigram probability.
- `perplexity(sentences, prob)` where `prob(a, b)` is a function of one bigram; return `inf` if any probability is zero.

import numpy as np
from collections import Counter

def build_counts(sentences):
    uni, bi, ctx = Counter(), Counter(), Counter()
    for s in sentences:
        t = ["<s>"] + s.split() + ["</s>"]
        uni.update(t[1:])
        ctx.update(t[:-1])
        bi.update(zip(t[:-1], t[1:]))
    return uni, bi, ctx

def p_mle(a, b, uni, bi, ctx):
    return bi[(a, b)] / ctx[a] if ctx[a] else 0.0

def p_addk(a, b, uni, bi, ctx, k):
    return (bi[(a, b)] + k) / (ctx[a] + k * len(uni))

def p_interp(a, b, uni, bi, ctx, lam):
    return lam * p_mle(a, b, uni, bi, ctx) + (1 - lam) * uni[b] / sum(uni.values())

def p_stupid(a, b, uni, bi, ctx, alpha=0.4):
    if bi[(a, b)] > 0:
        return bi[(a, b)] / ctx[a]
    return alpha * uni[b] / sum(uni.values())

def perplexity(sentences, prob):
    ll, n = 0.0, 0
    for s in sentences:
        t = ["<s>"] + s.split() + ["</s>"]
        for a, b in zip(t[:-1], t[1:]):
            p = prob(a, b)
            if p <= 0:
                return float("inf")
            ll += np.log(p)
            n += 1
    return float(np.exp(-ll / n))
