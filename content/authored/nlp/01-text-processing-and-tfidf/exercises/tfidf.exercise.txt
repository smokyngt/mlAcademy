Implement the classical text pipeline with NumPy:

- `tokenize(text)`: lowercase, then return the list of runs of letters/digits, allowing one inner apostrophe (`don't` stays one token). Punctuation is dropped.
- `build_vocab(docs)`: `docs` is a list of token lists; return the sorted list of distinct tokens.
- `count_matrix(docs, vocab)`: array of shape `(N, V)` of counts; tokens not in `vocab` are ignored.
- `tfidf(X)`: smoothed IDF `ln((1+N)/(1+df)) + 1`, multiply the counts, then L2-normalise every row (a row of zeros stays zero).
- `cosine_matrix(W)`: pairwise cosine similarity of the rows of an already normalised matrix.
- `ngrams(tokens, n)`: list of n-gram tuples.

import re
import numpy as np

def tokenize(text):
    return re.findall(r"[a-z0-9]+(?:'[a-z]+)?", text.lower())

def build_vocab(docs):
    return sorted({w for d in docs for w in d})

def count_matrix(docs, vocab):
    index = {w: i for i, w in enumerate(vocab)}
    X = np.zeros((len(docs), len(vocab)))
    for d, tokens in enumerate(docs):
        for w in tokens:
            if w in index:
                X[d, index[w]] += 1
    return X

def tfidf(X):
    N = X.shape[0]
    df = (X > 0).sum(axis=0)
    idf = np.log((1 + N) / (1 + df)) + 1
    T = X * idf
    norm = np.linalg.norm(T, axis=1, keepdims=True)
    return T / np.where(norm == 0, 1, norm)

def cosine_matrix(W):
    return W @ W.T

def ngrams(tokens, n):
    return [tuple(tokens[i:i + n]) for i in range(len(tokens) - n + 1)]
