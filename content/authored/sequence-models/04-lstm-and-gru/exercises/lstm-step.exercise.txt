import numpy as np

def sigmoid(s):
    return 0.5 * (1.0 + np.tanh(0.5 * s))

def lstm_step(x, h_prev, c_prev, W, U, b):
    H = h_prev.shape[0]
    a = W @ x + U @ h_prev + b
    i = sigmoid(a[:H]); f = sigmoid(a[H:2 * H]); g = np.tanh(a[2 * H:3 * H]); o = sigmoid(a[3 * H:])
    c = f * c_prev + i * g
    tc = np.tanh(c)
    h = o * tc
    return h, c, (x, h_prev, c_prev, W, U, i, f, g, o, tc)

def lstm_step_backward(dh, dc, cache):
    x, h_prev, c_prev, W, U, i, f, g, o, tc = cache
    do = dh * tc
    dc_tot = dc + dh * o * (1.0 - tc ** 2)
    df = dc_tot * c_prev
    di = dc_tot * g
    dg = dc_tot * i
    dc_prev = dc_tot * f
    da = np.concatenate([di * i * (1 - i), df * f * (1 - f), dg * (1 - g ** 2), do * o * (1 - o)])
    dW = np.outer(da, x)
    dU = np.outer(da, h_prev)
    db = da
    dx = W.T @ da
    dh_prev = U.T @ da
    return dx, dh_prev, dc_prev, dW, dU, db

rng = np.random.default_rng(0)
n, H = 3, 4
W = rng.normal(0, 0.5, (4 * H, n)); U = rng.normal(0, 0.5, (4 * H, H)); b = np.zeros(4 * H)
h, c, cache = lstm_step(rng.normal(size=n), np.zeros(H), np.zeros(H), W, U, b)
print(h.shape, c.shape)
