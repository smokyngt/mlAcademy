import numpy as np

def sigmoid(s):
    return 0.5 * (1.0 + np.tanh(0.5 * s))

def gru_step(x, h_prev, W, U, b):
    H = h_prev.shape[0]
    r = sigmoid(W[:H] @ x + U[:H] @ h_prev + b[:H])
    z = sigmoid(W[H:2 * H] @ x + U[H:2 * H] @ h_prev + b[H:2 * H])
    n_ = np.tanh(W[2 * H:] @ x + r * (U[2 * H:] @ h_prev) + b[2 * H:])
    return (1.0 - z) * n_ + z * h_prev

def n_params(kind, n, H):
    gates = {"rnn": 1, "lstm": 4, "gru": 3}[kind]
    return gates * (H * n + H * H + H)

print(n_params("lstm", 128, 256), n_params("gru", 128, 256))
