import numpy as np

def out_size(H, K, P, S):
    return (H - K + 2 * P) // S + 1

def n_params(C_in, C_out, K):
    return C_out * C_in * K * K + C_out

def receptive_field(layers):
    r, j = 1, 1
    for K, S in layers:
        r += (K - 1) * j
        j *= S
    return r

def maxpool_forward(x, k):
    C, H, W = x.shape
    blocks = x.reshape(C, H // k, k, W // k, k).transpose(0, 1, 3, 2, 4).reshape(C, H // k, W // k, k * k)
    arg = blocks.argmax(axis=-1)
    y = blocks.max(axis=-1)
    return y, (arg, x.shape, k)

def maxpool_backward(dout, cache):
    arg, shape, k = cache
    C, H, W = shape
    blocks = np.zeros((C, H // k, W // k, k * k))
    np.put_along_axis(blocks, arg[..., None], dout[..., None], axis=-1)
    return blocks.reshape(C, H // k, W // k, k, k).transpose(0, 1, 3, 2, 4).reshape(C, H, W)

print(out_size(32, 3, 1, 2), n_params(3, 64, 3), receptive_field([(3, 1)] * 3))
