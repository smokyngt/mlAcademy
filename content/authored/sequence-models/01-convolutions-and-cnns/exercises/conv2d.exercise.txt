def naive(x, W, b, s, p):
    C, H, Wd = x.shape
    Co, _, K, _ = W.shape
    xp = np.pad(x, ((0, 0), (p, p), (p, p)))
    Ho = (H + 2 * p - K) // s + 1; Wo = (Wd + 2 * p - K) // s + 1
    o = np.zeros((Co, Ho, Wo))
    for c in range(Co):
        for i in range(Ho):
            for j in range(Wo):
                o[c, i, j] = np.sum(xp[:, i * s:i * s + K, j * s:j * s + K] * W[c]) + b[c]
    return o

import numpy as np

def im2col(x, K, stride, pad):
    C, H, W = x.shape
    xp = np.pad(x, ((0, 0), (pad, pad), (pad, pad)))
    Ho = (H + 2 * pad - K) // stride + 1
    Wo = (W + 2 * pad - K) // stride + 1
    cols = np.zeros((C * K * K, Ho * Wo))
    for i in range(Ho):
        for j in range(Wo):
            cols[:, i * Wo + j] = xp[:, i * stride:i * stride + K, j * stride:j * stride + K].reshape(-1)
    return cols, Ho, Wo

def col2im(cols, x_shape, K, stride, pad):
    C, H, W = x_shape
    Ho = (H + 2 * pad - K) // stride + 1
    Wo = (W + 2 * pad - K) // stride + 1
    xp = np.zeros((C, H + 2 * pad, W + 2 * pad))
    for i in range(Ho):
        for j in range(Wo):
            xp[:, i * stride:i * stride + K, j * stride:j * stride + K] += cols[:, i * Wo + j].reshape(C, K, K)
    return xp[:, pad:pad + H, pad:pad + W]

def conv2d_forward(x, W, b, stride=1, pad=0):
    Co, Ci, K, _ = W.shape
    cols, Ho, Wo = im2col(x, K, stride, pad)
    out = W.reshape(Co, -1) @ cols + b[:, None]
    return out.reshape(Co, Ho, Wo), (cols, x.shape, W, stride, pad)

def conv2d_backward(dout, cache):
    cols, x_shape, W, stride, pad = cache
    Co, Ci, K, _ = W.shape
    dY = dout.reshape(Co, -1)
    dW = (dY @ cols.T).reshape(W.shape)
    db = dY.sum(axis=1)
    dcols = W.reshape(Co, -1).T @ dY
    dx = col2im(dcols, x_shape, K, stride, pad)
    return dx, dW, db

rng = np.random.default_rng(0)
x = rng.normal(size=(2, 6, 6)); W = rng.normal(size=(3, 2, 3, 3)); b = rng.normal(size=3)
out, cache = conv2d_forward(x, W, b, stride=2, pad=1)
print("out shape:", out.shape)
