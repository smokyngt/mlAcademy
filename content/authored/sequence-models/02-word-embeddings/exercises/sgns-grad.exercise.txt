import numpy as np

def sigmoid(s):
    return 0.5 * (1.0 + np.tanh(0.5 * s))

def sgns_loss_grads(V, U, c, o, negs):
    negs = np.asarray(negs)
    v = V[c]
    s_pos = U[o] @ v
    s_neg = U[negs] @ v
    loss = float(np.logaddexp(0.0, -s_pos) + np.logaddexp(0.0, s_neg).sum())
    g_pos = sigmoid(s_pos) - 1.0
    g_neg = sigmoid(s_neg)
    dV = np.zeros_like(V); dU = np.zeros_like(U)
    dV[c] = g_pos * U[o] + g_neg @ U[negs]
    dU[o] += g_pos * v
    np.add.at(dU, negs, np.outer(g_neg, v))
    return loss, dV, dU

V = np.zeros((4, 2)); U = np.zeros((4, 2))
V[0] = [0.5, 1.0]
U[1] = [1.0, 0.5]; U[2] = [-0.5, 0.5]; U[3] = [1.0, -1.0]
loss, dV, dU = sgns_loss_grads(V, U, 0, 1, np.array([2, 3]))
