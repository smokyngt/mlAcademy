    return 0.5 * (1.0 + np.tanh(0.5 * s))

def sgns_step(V, U, c, o, negs, lr):
    """One SGD step on a single (centre c, context o, negatives) example. Returns the loss."""
    v = V[c].copy()
    s_pos = U[o] @ v
    s_neg = U[negs] @ v
    loss = float(np.logaddexp(0.0, -s_pos) + np.logaddexp(0.0, s_neg).sum())

    return loss

import numpy as np

def sigmoid(s):
    return 0.5 * (1.0 + np.tanh(0.5 * s))

def sgns_step(V, U, c, o, negs, lr):
    v = V[c].copy()
    s_pos = U[o] @ v
    s_neg = U[negs] @ v
    loss = float(np.logaddexp(0.0, -s_pos) + np.logaddexp(0.0, s_neg).sum())
    g_pos = sigmoid(s_pos) - 1.0
    g_neg = sigmoid(s_neg)
    V[c] -= lr * (g_pos * U[o] + g_neg @ U[negs])
    U[o] -= lr * g_pos * v
    np.add.at(U, negs, -lr * np.outer(g_neg, v))
    return loss

CORPUS = [
    "the cat drinks milk", "the dog drinks water", "the cat chases the mouse", "the dog chases the cat",
    "the dog chases the mouse", "a cat eats fish", "a dog eats meat", "the cat eats fish", "the dog eats meat",
    "the king rules the land", "the queen rules the land", "a king wears a crown", "a queen wears a crown",
]

def build_vocab(sentences):
    words = sorted({w for s in sentences for w in s.split()})
    return {w: i for i, w in enumerate(words)}

def make_pairs(sentences, vocab, window):
    pairs = []
    for s in sentences:
        ids = [vocab[w] for w in s.split()]
        for i, c in enumerate(ids):
            for j in range(max(0, i - window), min(len(ids), i + window + 1)):
                if j != i:
                    pairs.append((c, ids[j]))
    return pairs

def noise_distribution(sentences, vocab):
    counts = np.zeros(len(vocab))
    for s in sentences:
        for w in s.split():
            counts[vocab[w]] += 1
    p = counts ** 0.75
    return p / p.sum()

def train_skipgram(sentences, dim=8, window=2, k=5, epochs=40, lr=0.05, seed=0):
    rng = np.random.default_rng(seed)
    vocab = build_vocab(sentences)
    n = len(vocab)
    pairs = make_pairs(sentences, vocab, window)
    noise = noise_distribution(sentences, vocab)
    V = rng.normal(0, 0.1, (n, dim)); U = rng.normal(0, 0.1, (n, dim))
    losses = []
    for _ in range(epochs):
        total = 0.0
        for idx in rng.permutation(len(pairs)):
            c, o = pairs[idx]
            negs = rng.choice(n, size=k, p=noise)
            total += sgns_step(V, U, c, o, negs, lr)
        losses.append(total / len(pairs))
    return vocab, V + U, losses

def cosine(a, b):
    return float(a @ b / (np.linalg.norm(a) * np.linalg.norm(b)))

def most_similar(word, vocab, E, topn=3):
    inv = {i: w for w, i in vocab.items()}
    q = E[vocab[word]]
    sims = [(inv[i], cosine(q, E[i])) for i in range(len(vocab)) if i != vocab[word]]
    return sorted(sims, key=lambda t: -t[1])[:topn]

vocab, E, losses = train_skipgram(CORPUS)
print(round(losses[0], 3), "->", round(losses[-1], 3))
print(most_similar("king", vocab, E))
