import numpy as np

def rnn_loss_grads(params, X, y):
    Wxh, Whh, b, Wy, by = params
    B, T, n = X.shape
    H = Whh.shape[0]
    hs = [np.zeros((B, H))]
    for t in range(T):
        hs.append(np.tanh(X[:, t] @ Wxh.T + hs[-1] @ Whh.T + b))
    yhat = hs[-1] @ Wy + by
    loss = float(0.5 * np.mean((yhat - y) ** 2))
    dy = (yhat - y) / B
    dWy = hs[-1].T @ dy
    dby = np.array(dy.sum())
    dh = np.outer(dy, Wy)
    dWxh = np.zeros_like(Wxh); dWhh = np.zeros_like(Whh); db = np.zeros_like(b)
    for t in range(T, 0, -1):
        da = dh * (1.0 - hs[t] ** 2)
        dWxh += da.T @ X[:, t - 1]
        dWhh += da.T @ hs[t - 1]
        db += da.sum(axis=0)
        dh = da @ Whh
    return loss, (dWxh, dWhh, db, dWy, dby)

def grad_check(params, X, y, h=1e-6):
    _, grads = rnn_loss_grads(params, X, y)
    worst = 0.0
    for k, p in enumerate(params):
        for idx in np.ndindex(*np.shape(p)):
            a = [np.array(q, dtype=float) for q in params]; c = [np.array(q, dtype=float) for q in params]
            a[k][idx] += h; c[k][idx] -= h
            num = (rnn_loss_grads(tuple(a), X, y)[0] - rnn_loss_grads(tuple(c), X, y)[0]) / (2 * h)
            ana = grads[k][idx]
            worst = max(worst, abs(ana - num) / max(1e-8, abs(ana) + abs(num)))
    return worst

rng = np.random.default_rng(0)
B, T, n, H = 4, 6, 3, 5
params = (rng.normal(0, 0.5, (H, n)), rng.normal(0, 0.5, (H, H)), np.zeros(H), rng.normal(0, 0.5, H), np.array(0.0))
X = rng.normal(size=(B, T, n)); y = rng.normal(size=B)
print("loss:", rnn_loss_grads(params, X, y)[0])
print("grad check:", grad_check(params, X, y))
