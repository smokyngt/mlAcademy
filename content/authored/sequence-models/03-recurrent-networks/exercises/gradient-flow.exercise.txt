Measure how fast the gradient path $\partial h_T/\partial h_0$ shrinks or grows, then implement gradient clipping.

1. `make_whh(H, rho, seed)`: a random `(H, H)` matrix rescaled so that its **spectral radius** (largest eigenvalue magnitude) equals `rho`.
2. `jacobian_norm(Whh, T, linear)`: the largest singular value of $\partial h_T/\partial h_0=J_T\cdots J_1$. With `linear=True` use $J_t=W_{hh}$ (no activation). 4. `clip_by_global_norm(grads, max_norm)`: returns `(clipped_list, total_norm_before)`; all arrays are scaled by the same factor, only if the total norm exceeds `max_norm`.

import numpy as np

def make_whh(H, rho, seed=0):
    G = np.random.default_rng(seed).normal(size=(H, H))
    return G * (rho / np.max(np.abs(np.linalg.eigvals(G))))

def jacobian_norm(Whh, T, linear=False, seed=1, in_scale=0.1):
    H = Whh.shape[0]
    if linear:
        return float(np.linalg.norm(np.linalg.matrix_power(Whh, T), 2))
    rng = np.random.default_rng(seed)
    Wxh = rng.normal(0, 1.0, (H, 3))
    h = np.zeros(H)
    M = np.eye(H)
    for _ in range(T):
        x = rng.normal(0, in_scale, 3)
        h = np.tanh(Wxh @ x + Whh @ h)
        M = (np.diag(1 - h ** 2) @ Whh) @ M
    return float(np.linalg.norm(M, 2))

def orthogonal_init(H, seed=0):
    Q, _ = np.linalg.qr(np.random.default_rng(seed).normal(size=(H, H)))
    return Q

def clip_by_global_norm(grads, max_norm):
    total = float(np.sqrt(sum((g ** 2).sum() for g in grads)))
    scale = max_norm / total if total > max_norm else 1.0
    return [g * scale for g in grads], total

for rho in (0.5, 0.9, 1.0, 1.5):
    W = make_whh(16, rho)
    print(rho, ["%.1e" % jacobian_norm(W, T, linear=True) for T in (10, 50)])
