Build the forward pass of a small encoder-decoder with Bahdanau attention (no training, just the computation).

- Encoder: embedding + tanh RNN, returns all states `Henc (Ts, H)`.
- Decoder: one tanh RNN cell whose input is `[embedding(previous token); context]`. The context comes from additive attention with the previous decoder state as the query (`attention` is provided). With `use_attention=False` the context is the last encoder state at every step (the classic bottleneck).
- `decode_step` returns `(s, logits, alpha)`; `decode(P, Henc, tgt_in)` runs teacher forcing and returns `(logits (Tt, Vt), alphas (Tt, Ts) or None)`.
- `seq2seq_loss(P, src, tgt_in, tgt_out, use_attention)` returns the mean cross-entropy.
- `greedy_decode(P, src, bos, max_len)` feeds back its own arg-max predictions (inference, no teacher forcing).

import numpy as np

def softmax(x):
    z = x - x.max(axis=-1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def attention(s, H, Wa, Ua, v):
    u = np.tanh(s @ Wa.T + H @ Ua.T)
    alpha = softmax(u @ v)
    return alpha @ H, alpha

def init_params(Vs, Vt, d=6, H=8, A=7, seed=0):
    r = np.random.default_rng(seed)
    g = lambda *shape: r.normal(0, 0.1, shape)
    return dict(Es=g(Vs, d), Et=g(Vt, d), Wxh=g(H, d), Whh=g(H, H), bh=np.zeros(H),
                Wd=g(H, d + H), Ud=g(H, H), bd=np.zeros(H),
                Wa=g(A, H), Ua=g(A, H), v=g(A),
                Wo=g(Vt, 2 * H), bo=np.zeros(Vt))

def encode(P, src):
    h = np.zeros(P["Whh"].shape[0])
    states = []
    for tok in src:
        h = np.tanh(P["Wxh"] @ P["Es"][tok] + P["Whh"] @ h + P["bh"])
        states.append(h)
    return np.array(states)

def decode_step(P, Henc, s_prev, y_prev, use_attention=True):
    if use_attention:
        ctx, alpha = attention(s_prev, Henc, P["Wa"], P["Ua"], P["v"])
    else:
        ctx, alpha = Henc[-1], None
    s = np.tanh(P["Wd"] @ np.concatenate([P["Et"][y_prev], ctx]) + P["Ud"] @ s_prev + P["bd"])
    logits = P["Wo"] @ np.concatenate([s, ctx]) + P["bo"]
    return s, logits, alpha

def decode(P, Henc, tgt_in, use_attention=True):
    s = Henc[-1]
    all_logits, all_alpha = [], []
    for y_prev in tgt_in:
        s, logits, alpha = decode_step(P, Henc, s, y_prev, use_attention)
        all_logits.append(logits); all_alpha.append(alpha)
    return np.array(all_logits), (np.array(all_alpha) if use_attention else None)

def seq2seq_loss(P, src, tgt_in, tgt_out, use_attention=True):
    logits, _ = decode(P, encode(P, src), tgt_in, use_attention)
    z = logits - logits.max(axis=1, keepdims=True)
    logp = z - np.log(np.exp(z).sum(axis=1, keepdims=True))
    return float(-np.mean(logp[np.arange(len(tgt_out)), tgt_out]))

def greedy_decode(P, src, bos, max_len, use_attention=True):
    Henc = encode(P, src)
    s, y, out = Henc[-1], bos, []
    for _ in range(max_len):
        s, logits, _ = decode_step(P, Henc, s, y, use_attention)
        y = int(np.argmax(logits))
        out.append(y)
    return np.array(out)

P = init_params(10, 10)
src = np.array([3, 1, 4, 1, 5]); tgt_in = np.array([0, 2, 7, 1]); tgt_out = np.array([2, 7, 1, 9])
print(round(seq2seq_loss(P, src, tgt_in, tgt_out), 3), "vs ln(10) =", round(np.log(10), 3))
