import numpy as np

def softmax(x):
    z = x - x.max(axis=-1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def additive_attention(s, H, Wa, Ua, v, mask=None):
    Z = s @ Wa.T + H @ Ua.T
    u = np.tanh(Z)
    e = u @ v
    if mask is not None:
        e = np.where(mask, e, -1e9)
    alpha = softmax(e)
    ctx = alpha @ H
    return ctx, alpha, (s, H, Wa, Ua, v, u, alpha)

def additive_attention_backward(dctx, cache):
    s, H, Wa, Ua, v, u, alpha = cache
    dalpha = H @ dctx
    dH = np.outer(alpha, dctx)
    de = alpha * (dalpha - alpha @ dalpha)
    dv = u.T @ de
    dZ = np.outer(de, v) * (1.0 - u ** 2)
    dWa = np.outer(dZ.sum(axis=0), s)
    ds = Wa.T @ dZ.sum(axis=0)
    dUa = dZ.T @ H
    dH = dH + dZ @ Ua
    return ds, dH, dWa, dUa, dv

def scaled_dot_attention(Q, K, Vv):
    W = softmax(Q @ K.T / np.sqrt(Q.shape[-1]))
    return W @ Vv, W

H = np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]])
s = np.array([0.5, -0.5])
ctx, alpha, _ = additive_attention(s, H, np.eye(2), np.eye(2), np.array([1.0, 1.0]))
print(alpha.round(4), ctx.round(4))
