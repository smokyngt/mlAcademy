Implement the three ingredients of DQN for a **linear** Q-function $Q(s,a;W)=W[a]\cdot\phi(s)$, where `W` has shape `(A, d)` and `phi(s)` is a feature vector of length `d`.

- `q_values(W, Phi)` -> `(B, A)` action values for a batch of feature rows `Phi` `(B, d)`.
- `dqn_loss_grad(W, W_target, batch, gamma)` -> `(loss, grad)` where `batch = (S, A, R, S2, D)` (features `(B,d)`, actions `(B,)` ints, rewards `(B,)`, next features `(B,d)`, done flags `(B,)` in {0,1}). Loss: $\frac1B\sum_i\frac12(y_i-Q(s_i,a_i;W))^2$ with the target $y_i=r_i+\gamma(1-d_i)\max_{a'}Q(s'_i,a';W_{\text{target}})$. `grad` has the shape of `W` and treats `y` as a constant (semi-gradient).
- `ReplayBuffer(capacity, state_dim)` with `push(s, a, r, s2, done)`, `__len__`, and `sample(batch_size, rng)` returning the tuple `(S, A, R, S2, D)`. It is a ring buffer: when full, the oldest transition is overwritten.
- `hard_update(W_target, W)` -> a copy of `W`; `soft_update(W_target, W, tau)` -> `(1-tau)*W_target + tau*W`.

`train` (given) runs DQN on a 6-state chain (reward 1 for reaching the right end, $\gamma=0.9$) with your pieces; the exact optimal values are $\gamma^{4-s}$ for "right" in state $s$.

import numpy as np

def q_values(W, Phi):
    return Phi @ W.T

def dqn_loss_grad(W, W_target, batch, gamma):
    S, A, R, S2, D = batch
    B = len(R)
    y = R + gamma * (1.0 - D) * q_values(W_target, S2).max(axis=1)
    q = q_values(W, S)[np.arange(B), A]
    delta = y - q
    loss = 0.5 * np.mean(delta ** 2)
    grad = np.zeros_like(W)
    for i in range(B):
        grad[A[i]] -= delta[i] * S[i] / B
    return loss, grad

class ReplayBuffer:
    def __init__(self, capacity, state_dim):
        self.cap = capacity
        self.S = np.zeros((capacity, state_dim)); self.S2 = np.zeros((capacity, state_dim))
        self.A = np.zeros(capacity, dtype=int); self.R = np.zeros(capacity); self.D = np.zeros(capacity)
        self.i = 0; self.n = 0
    def push(self, s, a, r, s2, done):
        i = self.i
        self.S[i] = s; self.A[i] = a; self.R[i] = r; self.S2[i] = s2; self.D[i] = float(done)
        self.i = (i + 1) % self.cap
        self.n = min(self.n + 1, self.cap)
    def __len__(self):
        return self.n
    def sample(self, batch_size, rng):
        idx = rng.integers(0, self.n, size=batch_size)
        return self.S[idx], self.A[idx], self.R[idx], self.S2[idx], self.D[idx]

def hard_update(W_target, W):
    return W.copy()

def soft_update(W_target, W, tau):
    return (1.0 - tau) * W_target + tau * W

N_STATES = 6

def feat(s, kind="onehot"):
    if kind == "onehot":
        v = np.zeros(N_STATES); v[s] = 1.0
        return v
    x = s / (N_STATES - 1)
    return np.array([1.0, x, x * x])

        a = int(rng.integers(2)) if rng.random() < eps else int(np.argmax(q + 1e-9 * rng.random(2)))
        s2 = min(max(s + (1 if a == 1 else -1), 0), N_STATES - 1)
        done = s2 == N_STATES - 1
        buf.push(feat(s, kind), a, 1.0 if done else 0.0, feat(s2, kind), done)
        s, t_ep = s2, t_ep + 1
