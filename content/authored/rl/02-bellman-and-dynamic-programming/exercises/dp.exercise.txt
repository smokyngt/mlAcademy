`make_gridworld` builds the classic 4x4 gridworld as an MDP: states 0..15 in row-major order, actions 0..3 = up, down, left, right, states 0 and 15 are terminal (absorbing, reward 0), every other move costs reward -1, and walking into a wall leaves you where you are. `P` has shape `(S, A, S)` and `R` has shape `(S, A, S)`.

Implement the dynamic-programming toolbox:

- `q_from_v(P, R, V, gamma)` -> `Q` of shape `(S, A)`, `Q[s,a] = sum_s' P[s,a,s'] (R[s,a,s'] + gamma V[s'])`.
- `policy_evaluation(P, R, pi, gamma, tol)` -> `V` for a stochastic policy `pi` of shape `(S, A)` (iterate the Bellman expectation backup until `max|V_new - V| < tol`).
- `greedy_policy(P, R, V, gamma)` -> one-hot policy `(S, A)` that is greedy with respect to `V` (ties: lowest action index).
- `policy_iteration(P, R, gamma)` -> `(pi, V, n_iterations)` starting from the uniform random policy.
- `value_iteration(P, R, gamma, tol)` -> `(V, n_sweeps)` using the Bellman optimality backup.

import numpy as np

def make_gridworld(n=4):
    S = n * n
    P = np.zeros((S, 4, S)); R = np.zeros((S, 4, S))
    moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    for s in range(S):
        if s in (0, S - 1):
            P[s, :, s] = 1.0
            continue
        r, c = divmod(s, n)
        for a, (dr, dc) in enumerate(moves):
            r2, c2 = r + dr, c + dc
            if not (0 <= r2 < n and 0 <= c2 < n):
                r2, c2 = r, c
            s2 = r2 * n + c2
            P[s, a, s2] = 1.0
            R[s, a, s2] = -1.0
    return P, R

def q_from_v(P, R, V, gamma):
    return (P * R).sum(axis=2) + gamma * (P @ V)

def policy_evaluation(P, R, pi, gamma, tol=1e-10):
    V = np.zeros(P.shape[0])
    while True:
        V_new = (pi * q_from_v(P, R, V, gamma)).sum(axis=1)
        delta = np.abs(V_new - V).max()
        V = V_new
        if delta < tol:
            return V

def greedy_policy(P, R, V, gamma):
    Q = q_from_v(P, R, V, gamma)
    pi = np.zeros_like(Q)
    pi[np.arange(Q.shape[0]), Q.argmax(axis=1)] = 1.0
    return pi

def policy_iteration(P, R, gamma):
    S, A = P.shape[:2]
    pi = np.ones((S, A)) / A
    n = 0
    while True:
        V = policy_evaluation(P, R, pi, gamma)
        new_pi = greedy_policy(P, R, V, gamma)
        n += 1
        if np.array_equal(new_pi, pi):
            return pi, V, n
        pi = new_pi

def value_iteration(P, R, gamma, tol=1e-10):
    V = np.zeros(P.shape[0])
    n = 0
    while True:
        V_new = q_from_v(P, R, V, gamma).max(axis=1)
        n += 1
        delta = np.abs(V_new - V).max()
        V = V_new
        if delta < tol:
            return V, n
