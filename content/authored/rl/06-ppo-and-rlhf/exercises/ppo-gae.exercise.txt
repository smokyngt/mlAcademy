Implement two core pieces of PPO.

**1. Generalised advantage estimation.** `gae(rewards, values, last_value, gamma, lam, dones=None)` returns advantages of shape `(T,)` from the TD residuals $\delta_t=r_t+\gamma(1-d_t)V_{t+1}-V_t$ (with $V_T=$ `last_value`) as $A_t=\delta_t+\gamma\lambda(1-d_t)A_{t+1}$, computed backwards. `dones` marks steps after which the episode ended (default: none).

**2. The clipped surrogate for a softmax policy** over actions (a bandit-style setting: no state, logits `theta` of shape `(A,)`). With sampled `acts` `(N,)`, advantages `adv` `(N,)` and old log-probabilities `logp_old` `(N,)`:

- `ppo_objective(theta, acts, adv, logp_old, eps)` -> $\frac1N\sum_i\min\big(\rho_iA_i,\;\operatorname{clip}(\rho_i,1-\varepsilon,1+\varepsilon)A_i\big)$ with $\rho_i=\pi_\theta(a_i)/\pi_{\text{old}}(a_i)$.
- `ppo_grad(theta, acts, adv, logp_old, eps)` -> its gradient with respect to `theta` (analytic; clipped samples contribute zero).
- `ppo_update(theta, acts, adv, logp_old, eps, lr, epochs, clip=True)` -> repeated gradient ascent on the same batch. With `clip=False` use the plain surrogate $\rho_iA_i$ (no clipping) to see why the clip matters.

import numpy as np

def softmax(z):
    e = np.exp(z - np.max(z))
    return e / e.sum()

def gae(rewards, values, last_value, gamma, lam, dones=None):
    T = len(rewards)
    vals = np.append(values, last_value)
    adv = np.zeros(T)
    running = 0.0
    for t in reversed(range(T)):
        nd = 1.0 if dones is None else 1.0 - dones[t]
        delta = rewards[t] + gamma * vals[t + 1] * nd - vals[t]
        running = delta + gamma * lam * nd * running
        adv[t] = running
    return adv

def ppo_objective(theta, acts, adv, logp_old, eps):
    logp = np.log(softmax(theta))[acts]
    ratio = np.exp(logp - logp_old)
    return float(np.mean(np.minimum(ratio * adv, np.clip(ratio, 1 - eps, 1 + eps) * adv)))

def ppo_grad(theta, acts, adv, logp_old, eps):
    p = softmax(theta)
    g = np.zeros_like(theta)
    for a, A, lo in zip(acts, adv, logp_old):
        ratio = p[a] / np.exp(lo)
        if (A > 0 and ratio > 1 + eps) or (A < 0 and ratio < 1 - eps):
            continue
        gl = -p.copy(); gl[a] += 1.0
        g += ratio * A * gl
    return g / len(acts)

def ppo_update(theta, acts, adv, logp_old, eps, lr, epochs, clip=True):
    th = theta.copy()
    for _ in range(epochs):
        if clip:
            th = th + lr * ppo_grad(th, acts, adv, logp_old, eps)
        else:
            p = softmax(th)
            g = np.zeros_like(th)
            for a, A, lo in zip(acts, adv, logp_old):
                ratio = p[a] / np.exp(lo)
                gl = -p.copy(); gl[a] += 1.0
                g += ratio * A * gl
            th = th + lr * g / len(acts)
    return th
