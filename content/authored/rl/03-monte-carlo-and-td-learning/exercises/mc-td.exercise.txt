A random walk has 5 states (0..4); you start in the middle (state 2) and move left or right with probability 1/2 each. Stepping off the left end ends the episode with reward 0, stepping off the right end ends it with reward 1; all other rewards are 0 and $\gamma=1$. The true values are $(1,2,3,4,5)/6$. `walk_episode` (given) returns the visited states `traj` (ending with `None`) and the rewards `rew`.

- `mc_returns(rewards, gamma)` -> list with the return $G_t$ for every step.

- `mc_prediction(episodes, alpha, seed)` and `td0_prediction(episodes, alpha, seed)` -> value vector (5,) learned with constant step size, initial values 0.5.

import numpy as np

def walk_episode(rng, n=5):
    s = n // 2
    traj, rew = [s], []
    while True:
        s2 = s + (1 if rng.random() < 0.5 else -1)
        if s2 < 0:
            rew.append(0.0); traj.append(None); break
        if s2 >= n:
            rew.append(1.0); traj.append(None); break
        rew.append(0.0); traj.append(s2); s = s2
    return traj, rew

def mc_returns(rewards, gamma):
    out = [0.0] * len(rewards)
    G = 0.0
    for t in reversed(range(len(rewards))):
        G = rewards[t] + gamma * G
        out[t] = G
    return out

def td0_update(V, s, r, s_next, alpha, gamma):
    target = r if s_next is None else r + gamma * V[s_next]
    return V[s] + alpha * (target - V[s])

def mc_prediction(episodes, alpha, seed, n=5):
    rng = np.random.default_rng(seed)
    V = np.full(n, 0.5)
    for _ in range(episodes):
        traj, rew = walk_episode(rng)
        G = mc_returns(rew, 1.0)
        for t in range(len(rew)):
            V[traj[t]] += alpha * (G[t] - V[traj[t]])
    return V

def td0_prediction(episodes, alpha, seed, n=5):
    rng = np.random.default_rng(seed)
    V = np.full(n, 0.5)
    for _ in range(episodes):
        traj, rew = walk_episode(rng)
        for t in range(len(rew)):
            V[traj[t]] = td0_update(V, traj[t], rew[t], traj[t + 1], alpha, 1.0)
    return V
