A 4x8 cliff-walking grid: you start at the bottom-left corner and must reach the bottom-right corner. Every step costs -1; stepping onto the cliff (bottom row, columns 1..6) costs -100 and sends you back to the start. The agent explores with epsilon-greedy ($\varepsilon=0.1$).

`step` and `train` are given. Implement the two TD control updates (modify `Q` in place and return the TD error):

- `sarsa_update(Q, s, a, r, s2, a2, done, alpha, gamma)` -> on-policy: bootstrap from the action `a2` actually chosen next.
- `q_learning_update(Q, s, a, r, s2, done, alpha, gamma)` -> off-policy: bootstrap from the best action in `s2`.

The tests train both on the same seeds and compare the greedy paths and the online reward.

import numpy as np

ROWS, COLS = 4, 8
START, GOAL = (ROWS - 1) * COLS, ROWS * COLS - 1

def step(s, a):
    r, c = divmod(s, COLS)
    dr, dc = [(-1, 0), (0, 1), (1, 0), (0, -1)][a]
    r2 = min(max(r + dr, 0), ROWS - 1); c2 = min(max(c + dc, 0), COLS - 1)
    if r2 == ROWS - 1 and 1 <= c2 <= COLS - 2:
        return START, -100.0, False
    s2 = r2 * COLS + c2
    return s2, -1.0, s2 == GOAL

def sarsa_update(Q, s, a, r, s2, a2, done, alpha, gamma):
    raise NotImplementedError

def q_learning_update(Q, s, a, r, s2, done, alpha, gamma):
    raise NotImplementedError

def train(algo, n_episodes, seed, eps=0.1, alpha=0.5, gamma=1.0):
    rng = np.random.default_rng(seed)
    Q = np.zeros((ROWS * COLS, 4)); returns = []
    def act(s):
        if rng.random() < eps:
            return int(rng.integers(4))
        return int(np.argmax(Q[s] + 1e-9 * rng.random(4)))
    for _ in range(n_episodes):
        s = START; a = act(s); G = 0.0
        for _ in range(500):
            s2, r, done = step(s, a)
            a2 = act(s2)
            G += r
            if algo == "sarsa":
                sarsa_update(Q, s, a, r, s2, a2, done, alpha, gamma)
            else:
                q_learning_update(Q, s, a, r, s2, done, alpha, gamma)
            s, a = s2, a2
            if done:
                break
        returns.append(G)
    return Q, returns

def greedy_path(Q):
    s = START; path = [s]
    for _ in range(60):
        s, _, done = step(s, int(np.argmax(Q[s]))); path.append(s)
        if done:
            break
    return path

import numpy as np

ROWS, COLS = 4, 8
START, GOAL = (ROWS - 1) * COLS, ROWS * COLS - 1

def step(s, a):
    r, c = divmod(s, COLS)
    dr, dc = [(-1, 0), (0, 1), (1, 0), (0, -1)][a]
    r2 = min(max(r + dr, 0), ROWS - 1); c2 = min(max(c + dc, 0), COLS - 1)
    if r2 == ROWS - 1 and 1 <= c2 <= COLS - 2:
        return START, -100.0, False
    s2 = r2 * COLS + c2
    return s2, -1.0, s2 == GOAL

def sarsa_update(Q, s, a, r, s2, a2, done, alpha, gamma):
    target = r if done else r + gamma * Q[s2, a2]
    td_error = target - Q[s, a]
    Q[s, a] += alpha * td_error
    return td_error

def q_learning_update(Q, s, a, r, s2, done, alpha, gamma):
    target = r if done else r + gamma * Q[s2].max()
    td_error = target - Q[s, a]
    Q[s, a] += alpha * td_error
    return td_error

def train(algo, n_episodes, seed, eps=0.1, alpha=0.5, gamma=1.0):
    rng = np.random.default_rng(seed)
    Q = np.zeros((ROWS * COLS, 4)); returns = []
    def act(s):
        if rng.random() < eps:
            return int(rng.integers(4))
        return int(np.argmax(Q[s] + 1e-9 * rng.random(4)))
    for _ in range(n_episodes):
        s = START; a = act(s); G = 0.0
        for _ in range(500):
            s2, r, done = step(s, a)
            a2 = act(s2)
            G += r
            if algo == "sarsa":
                sarsa_update(Q, s, a, r, s2, a2, done, alpha, gamma)
            else:
                q_learning_update(Q, s, a, r, s2, done, alpha, gamma)
            s, a = s2, a2
            if done:
                break
        returns.append(G)
    return Q, returns

def greedy_path(Q):
    s = START; path = [s]
    for _ in range(60):
        s, _, done = step(s, int(np.argmax(Q[s]))); path.append(s)
        if done:
            break
    return path
