import numpy as np

def softmax(z):
    e = np.exp(z - np.max(z))
    return e / e.sum()

def grad_log_pi(th_s, a):
    g = -softmax(th_s)
    g[a] += 1.0
    return g

def returns_to_go(rewards, gamma):
    out = [0.0] * len(rewards)
    G = 0.0
    for t in reversed(range(len(rewards))):
        G = rewards[t] + gamma * G
        out[t] = G
    return out

def reinforce_baseline_update(theta, V, traj, gamma, lr_pi, lr_v):
    G = returns_to_go([step[2] for step in traj], gamma)
    for (s, a, r, s2, done), g in zip(traj, G):
        adv = g - V[s]
        V[s] += lr_v * adv
        theta[s] += lr_pi * adv * grad_log_pi(theta[s], a)

def a2c_update(theta, V, s, a, r, s2, done, gamma, lr_pi, lr_v):
    target = r + (0.0 if done else gamma * V[s2])
    delta = target - V[s]
    V[s] += lr_v * delta
    theta[s] += lr_pi * delta * grad_log_pi(theta[s], a)
    return delta

N = 5

def run(mode, seed, episodes=400, lr_pi=0.5, lr_v=0.2, gamma=0.95):
    rng = np.random.default_rng(seed)
    theta = np.zeros((N, 2)); V = np.zeros(N)
    for _ in range(episodes):
        s = 0; traj = []
        for _ in range(30):
            a = int(rng.choice(2, p=softmax(theta[s])))
            s2 = min(max(s + (1 if a == 1 else -1), 0), N - 1)
            done = s2 == N - 1
            r = 1.0 if done else 0.0
            traj.append((s, a, r, s2, done))
            if mode == "a2c":
                a2c_update(theta, V, s, a, r, s2, done, gamma, lr_pi, lr_v)
            s = s2
            if done:
                break
        if mode == "reinforce":
            reinforce_baseline_update(theta, V, traj, gamma, lr_pi, lr_v)
    return theta, V
