`I` is `(N, d)` image embeddings and `T` is `(N, d)` embeddings of the matching captions (`T[i]` is the caption of `I[i]`).

- `l2_normalize(x)`: unit-length rows.
- `clip_logits(I, T, tau)`: `(N, N)` matrix of cosine similarities divided by the temperature `tau`.
- `clip_loss(I, T, tau)`: symmetric InfoNCE loss, the mean of image-to-text and text-to-image cross-entropies with the identity as target.
- `clip_logit_grad(I, T, tau)`: the gradient of `clip_loss` with respect to the **logit matrix** (shape `(N, N)`).
- `zero_shot_predict(image_emb, class_emb)`: for image embeddings `(M, d)` and class-prompt embeddings `(C, d)` return the predicted class index per image.
- `ensemble_prompts(embs)`: `embs` has shape `(C, K, d)` (K prompt templates per class); return `(C, d)` class embeddings: mean over templates, then normalised.
- `recall_at_k(I, T, k)`: image-to-text retrieval recall@k: the fraction of images whose own caption is among the `k` most similar captions.

import numpy as np

def l2_normalize(x):
    return x / np.linalg.norm(x, axis=1, keepdims=True)

def clip_logits(I, T, tau):
    return l2_normalize(I) @ l2_normalize(T).T / tau

def _softmax(L):
    Z = L - L.max(axis=1, keepdims=True)
    P = np.exp(Z)
    return P / P.sum(axis=1, keepdims=True)

def clip_loss(I, T, tau):
    L = clip_logits(I, T, tau)
    N = len(L)
    i2t = -np.log(_softmax(L)[np.arange(N), np.arange(N)]).mean()
    t2i = -np.log(_softmax(L.T)[np.arange(N), np.arange(N)]).mean()
    return float(0.5 * (i2t + t2i))

def clip_logit_grad(I, T, tau):
    L = clip_logits(I, T, tau)
    N = len(L)
    E = np.eye(N)
    return 0.5 * ((_softmax(L) - E) / N + (_softmax(L.T) - E).T / N)

def zero_shot_predict(image_emb, class_emb):
    return np.argmax(l2_normalize(image_emb) @ l2_normalize(class_emb).T, axis=1)

def ensemble_prompts(embs):
    C, K, d = embs.shape
    normed = embs / np.linalg.norm(embs, axis=2, keepdims=True)
    return l2_normalize(normed.mean(axis=1))

def recall_at_k(I, T, k):
    S = l2_normalize(I) @ l2_normalize(T).T
    top = np.argsort(-S, axis=1)[:, :k]
    return float(np.mean([i in top[i] for i in range(len(S))]))
