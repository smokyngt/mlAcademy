- `patchify(img, p)`: image `(C, H, W)` with `H, W` divisible by `p` -> `(N, C*p*p)` with `N = (H/p)*(W/p)`. Patches are in row-major grid order; inside a patch the pixels are flattened in `(C, p, p)` order.

Normalise to unit length, use cosine similarity divided by `tau`, exclude self-similarity, positive pair = the other view, negatives = all other `2N - 2` views; return the mean loss over all `2N` anchors.

import numpy as np

def patchify(img, p):
    C, H, W = img.shape
    x = img.reshape(C, H // p, p, W // p, p)
    x = x.transpose(1, 3, 0, 2, 4)
    return x.reshape((H // p) * (W // p), C * p * p)

def unpatchify(patches, C, H, W, p):
    x = patches.reshape(H // p, W // p, C, p, p)
    x = x.transpose(2, 0, 3, 1, 4)
    return x.reshape(C, H, W)

def patch_embed(img, p, Wp, bp):
    return patchify(img, p) @ Wp + bp

def n_tokens(H, W, p, cls=True):
    return (H // p) * (W // p) + (1 if cls else 0)

def nt_xent(z1, z2, tau):
    N = len(z1)
    Z = np.concatenate([z1, z2], axis=0)
    Z = Z / np.linalg.norm(Z, axis=1, keepdims=True)
    S = Z @ Z.T / tau
    np.fill_diagonal(S, -np.inf)
    pos = (np.arange(2 * N) + N) % (2 * N)
    m = S.max(axis=1, keepdims=True)
    lse = (m + np.log(np.exp(S - m).sum(axis=1, keepdims=True))).ravel()
    return float(np.mean(lse - S[np.arange(2 * N), pos]))
