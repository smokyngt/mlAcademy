import numpy as np

def dice_score(p, y, eps=1e-9):
    return 2 * np.sum(p * y) / (np.sum(p) + np.sum(y) + eps)

def dice_loss(p, y, eps=1e-9):
    return 1 - dice_score(p, y, eps)

def iou_score(p, y, eps=1e-9):
    inter = np.sum(p * y)
    return inter / (np.sum(p) + np.sum(y) - inter + eps)

def confusion_matrix(pred, target, n_classes):
    cm = np.zeros((n_classes, n_classes), dtype=int)
    np.add.at(cm, (np.asarray(target).ravel(), np.asarray(pred).ravel()), 1)
    return cm

def miou(cm):
    diag = np.diag(cm).astype(float)
    denom = cm.sum(axis=1) + cm.sum(axis=0) - diag
    ious = np.where(denom > 0, diag / np.maximum(denom, 1), np.nan)
    return ious, float(np.nanmean(ious))

def unet_shapes(H, W, base, depth):
    if H % (2 ** depth) or W % (2 ** depth):
        raise ValueError("H and W must be divisible by 2**depth")
    enc = [(base * 2 ** k, H // 2 ** k, W // 2 ** k) for k in range(depth)]
    bottleneck = [(base * 2 ** depth, H // 2 ** depth, W // 2 ** depth)]
    return enc + bottleneck + enc[::-1]
