- `match_detections(det_boxes, det_scores, gt_boxes, thr)`: return a list of 0/1 flags, one per detection, **ordered by decreasing score**: 1 = true positive, 0 = false positive. Each ground-truth box may be claimed by at most one detection (the highest-scoring one); a detection matches its **best-IoU** ground truth, and counts only if that IoU is `>= thr` and the box is unclaimed.
- `average_precision(tp_flags, n_gt)`: area under the precision-recall curve using the all-points rule: precision made monotone non-increasing from the right (the "envelope"), then summed over recall increments.
- `ap_over_thresholds(det_boxes, det_scores, gt_boxes, thrs)`: mean AP over a list of IoU thresholds.

import numpy as np

def iou(a, b):
    iw = max(0.0, min(a[2], b[2]) - max(a[0], b[0]))
    ih = max(0.0, min(a[3], b[3]) - max(a[1], b[1]))
    inter = iw * ih
    ua = (a[2] - a[0]) * (a[3] - a[1]) + (b[2] - b[0]) * (b[3] - b[1]) - inter
    return inter / ua if ua > 0 else 0.0

def match_detections(det_boxes, det_scores, gt_boxes, thr):
    order = np.argsort(-np.asarray(det_scores, dtype=float), kind="stable")
    claimed = set()
    flags = []
    for i in order:
        ious = [iou(det_boxes[i], g) for g in gt_boxes]
        if ious:
            j = int(np.argmax(ious))
            if ious[j] >= thr and j not in claimed:
                claimed.add(j)
                flags.append(1)
                continue
        flags.append(0)
    return flags

def average_precision(tp_flags, n_gt):
    if len(tp_flags) == 0 or n_gt == 0:
        return 0.0
    tp = np.cumsum(np.asarray(tp_flags, dtype=float))
    precision = tp / np.arange(1, len(tp) + 1)
    recall = tp / n_gt
    envelope = np.maximum.accumulate(precision[::-1])[::-1]
    prev = np.concatenate([[0.0], recall[:-1]])
    return float(np.sum((recall - prev) * envelope))

def ap_over_thresholds(det_boxes, det_scores, gt_boxes, thrs):
    return float(np.mean([average_precision(match_detections(det_boxes, det_scores, gt_boxes, t), len(gt_boxes)) for t in thrs]))
