Boxes are `(x1, y1, x2, y2)` with `x1 < x2`, `y1 < y2`.

- `iou(a, b)`: intersection over union of two boxes (0 when they do not overlap).
- `iou_matrix(A, B)`: `(len(A), len(B))` matrix of pairwise IoU.

import numpy as np

def iou(a, b):
    iw = max(0.0, min(a[2], b[2]) - max(a[0], b[0]))
    ih = max(0.0, min(a[3], b[3]) - max(a[1], b[1]))
    inter = iw * ih
    ua = (a[2] - a[0]) * (a[3] - a[1]) + (b[2] - b[0]) * (b[3] - b[1]) - inter
    return inter / ua if ua > 0 else 0.0

def iou_matrix(A, B):
    return np.array([[iou(a, b) for b in B] for a in A]).reshape(len(A), len(B))

def nms(boxes, scores, thr):
    order = list(np.argsort(-np.asarray(scores, dtype=float), kind="stable"))
    keep = []
    while order:
        i = order.pop(0)
        keep.append(int(i))
        order = [j for j in order if iou(boxes[i], boxes[j]) <= thr]
    return keep

def decode_cell(t, cell_x, cell_y, stride, anchor):
    sig = lambda z: 1.0 / (1.0 + np.exp(-z))
    cx = (sig(t[0]) + cell_x) * stride
    cy = (sig(t[1]) + cell_y) * stride
    w = anchor[0] * np.exp(t[2])
    h = anchor[1] * np.exp(t[3])
    return (cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2)
