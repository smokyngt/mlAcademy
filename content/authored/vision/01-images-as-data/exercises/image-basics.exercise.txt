import numpy as np

def to_gray(rgb):
    return rgb.astype(float) @ np.array([0.299, 0.587, 0.114])

def standardize(img_uint8, mean, std):
    x = img_uint8.astype(float) / 255.0
    x = (x - np.asarray(mean)) / np.asarray(std)
    return x.transpose(2, 0, 1)

def conv2d_valid(img, kernel):
    H, W = img.shape
    kh, kw = kernel.shape
    out = np.zeros((H - kh + 1, W - kw + 1))
    for i in range(out.shape[0]):
        for j in range(out.shape[1]):
            out[i, j] = np.sum(img[i:i + kh, j:j + kw] * kernel)
    return out

def sobel_magnitude(gray):
    Kx = np.array([[-1., 0, 1], [-2, 0, 2], [-1, 0, 1]])
    gx = conv2d_valid(gray, Kx)
    gy = conv2d_valid(gray, Kx.T)
    return np.sqrt(gx ** 2 + gy ** 2)

def histogram(gray, bins):
    idx = (np.asarray(gray).astype(int) * bins) // 256
    return np.bincount(idx.ravel(), minlength=bins)
