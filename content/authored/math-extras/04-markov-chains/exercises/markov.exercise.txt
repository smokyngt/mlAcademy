import numpy as np

def n_step(P, dist, n):
    d = np.asarray(dist, dtype=float)
    for _ in range(n):
        d = d @ P
    return d

def stationary(P):
    w, V = np.linalg.eig(P.T)
    i = int(np.argmin(np.abs(w - 1)))
    v = np.real(V[:, i])
    return v / v.sum()

def is_ergodic(P):
    n = P.shape[0]
    Q = np.eye(n)
    for _ in range((n - 1) ** 2 + 1):
        Q = Q @ P
        if np.all(Q > 1e-12):
            return True
    return False

def second_eigenvalue_modulus(P):
    w = np.sort(np.abs(np.linalg.eigvals(P)))[::-1]
    return float(w[1])

def simulate_chain(P, start, steps, rng):
    cum = np.cumsum(P, axis=1)
    n = P.shape[0]
    u = rng.random(steps)
    path = np.empty(steps, dtype=int)
    s = start
    for t in range(steps):
        s = min(int(np.searchsorted(cum[s], u[t], side="right")), n - 1)
        path[t] = s
    return path

def empirical_distribution(path, n_states):
    return np.bincount(path, minlength=n_states) / len(path)

def mh_matrix(target):
    target = np.asarray(target, dtype=float)
    n = len(target)
    Q = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if i != j:
                Q[i, j] = min(1.0, target[j] / target[i]) / n
        Q[i, i] = 1.0 - Q[i].sum()
    return Q

def poisson_counts(rate, T, n, rng):
    m = int(rate * T * 3 + 20)
    gaps = rng.exponential(1.0 / rate, size=(n, m))
    times = np.cumsum(gaps, axis=1)
    return (times <= T).sum(axis=1)
