import numpy as np

def adjacency(n, edges):
    A = np.zeros((n, n))
    for i, j in edges:
        A[i, j] = 1
        A[j, i] = 1
    return A

def laplacian(A):
    return np.diag(A.sum(axis=1)) - A

def n_components(A):
    w = np.linalg.eigvalsh(laplacian(A))
    return int(np.sum(w < 1e-9))

def count_walks(A, i, j, k):
    return int(round(np.linalg.matrix_power(A, k)[i, j]))

def spectral_bisection(A):
    w, V = np.linalg.eigh(laplacian(A))
    lab = (V[:, 1] > 0).astype(int)
    if lab[0] == 1:
        lab = 1 - lab
    return lab

def pagerank(M, d=0.85, tol=1e-12):
    n = M.shape[0]
    out = M.sum(axis=1, keepdims=True)
    P = np.where(out > 0, M / np.where(out > 0, out, 1), 1.0 / n)
    r = np.ones(n) / n
    for _ in range(10000):
        r_new = d * (r @ P) + (1 - d) / n
        if np.abs(r_new - r).sum() < tol:
            r = r_new
            break
        r = r_new
    return r / r.sum()
