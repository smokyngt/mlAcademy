import numpy as np

def dft(x):
    x = np.asarray(x, dtype=complex)
    N = len(x)
    k = np.arange(N)
    F = np.exp(-2j * np.pi * np.outer(k, k) / N)
    return F @ x

def idft(X):
    X = np.asarray(X, dtype=complex)
    N = len(X)
    k = np.arange(N)
    F = np.exp(2j * np.pi * np.outer(k, k) / N)
    return F @ X / N

def fft(x):
    x = np.asarray(x, dtype=complex)
    N = len(x)
    if N == 1:
        return x
    E = fft(x[0::2])
    O = fft(x[1::2])
    t = np.exp(-2j * np.pi * np.arange(N // 2) / N) * O
    return np.concatenate([E + t, E - t])

def bin_frequencies(N, fs):
    return np.arange(N) * fs / N

def alias_frequency(f, fs):
    g = f % fs
    return min(g, fs - g)

def peak_frequency(x, fs):
    X = fft(x)
    N = len(X)
    k = int(np.argmax(np.abs(X[: N // 2 + 1])))
    return k * fs / N

def circular_convolve(a, b):
    A, B = fft(a), fft(b)
    prod = A * B
    out = np.conj(fft(np.conj(prod))) / len(prod)
    return out.real
