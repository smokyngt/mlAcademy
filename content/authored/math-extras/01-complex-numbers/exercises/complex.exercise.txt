import numpy as np

def to_polar(z):
    return float(abs(z)), float(np.arctan2(z.imag, z.real))

def from_polar(r, theta):
    return r * np.exp(1j * theta)

def mul_polar(p1, p2):
    return p1[0] * p2[0], p1[1] + p2[1]

def euler_series(theta, terms):
    total, term = 0j, 1 + 0j
    for k in range(terms):
        total += term
        term = term * (1j * theta) / (k + 1)
    return total

def roots_of_unity(n):
    return np.exp(2j * np.pi * np.arange(n) / n)

def rotation_eigenvalues(theta):
    tr, det = 2 * np.cos(theta), 1.0
    disc = np.sqrt(complex(tr * tr - 4 * det))
    a, b = (tr + disc) / 2, (tr - disc) / 2
    return (a, b) if a.imag >= b.imag else (b, a)
