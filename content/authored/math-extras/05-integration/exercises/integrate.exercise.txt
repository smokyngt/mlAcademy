import numpy as np

def left_riemann(f, a, b, n):
    x = a + (b - a) * np.arange(n) / n
    return (b - a) / n * np.sum(f(x))

def trapezoid(f, a, b, n):
    x = np.linspace(a, b, n + 1)
    y = f(x)
    h = (b - a) / n
    return h * (y[0] / 2 + y[1:-1].sum() + y[-1] / 2)

def simpson(f, a, b, n):
    x = np.linspace(a, b, n + 1)
    y = f(x)
    h = (b - a) / n
    return h / 3 * (y[0] + 4 * y[1:-1:2].sum() + 2 * y[2:-1:2].sum() + y[-1])

def mc_integrate(f, a, b, n, rng):
    u = rng.uniform(a, b, n)
    vals = (b - a) * f(u)
    return float(vals.mean()), float(vals.std(ddof=1) / np.sqrt(n))

def gaussian_integral(n):
    return trapezoid(lambda x: np.exp(-x ** 2 / 2), -10, 10, n)

def expectation(pdf, g, a, b, n=400):
    return simpson(lambda x: g(x) * pdf(x), a, b, n)

def mc_ball_volume(d, n, rng):
    pts = rng.uniform(-1, 1, size=(n, d))
    inside = (pts ** 2).sum(axis=1) <= 1.0
    return float(inside.mean() * 2 ** d)
