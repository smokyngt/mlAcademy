import numpy as np

def sum_of_squares(n):
    total = 0
    for i in range(1, n + 1):
        total += i * i
    return total

def classify(x):
    if x < 0:
        return "negative"
    elif x == 0:
        return "zero"
    return "positive"

def standardize(a):
    return (a - a.mean()) / a.std()

def center_columns(M):
    return M - M.mean(axis=0)

def times_table(n):
    r = np.arange(1, n + 1)
    return r[:, None] * r[None, :]
