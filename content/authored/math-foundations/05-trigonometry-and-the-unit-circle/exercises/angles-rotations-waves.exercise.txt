import numpy as np

def deg2rad(d):
    return d * np.pi / 180

def polar(x, y):
    return float(np.hypot(x, y)), float(np.arctan2(y, x))

def rotate(points, theta):
    c, s = np.cos(theta), np.sin(theta)
    R = np.array([[c, -s], [s, c]])
    return points @ R.T

def wave(t, A, f, phi):
    return A * np.sin(2 * np.pi * f * t + phi)

def fourier_amplitude(x, k):
    N = len(x)
    n = np.arange(N)
    ang = 2 * np.pi * k * n / N
    return float(2 / N * np.sqrt((x @ np.cos(ang)) ** 2 + (x @ np.sin(ang)) ** 2))

def pos_encoding(pos, d):
    out = np.zeros(d)
    for i in range(d // 2):
        angle = pos / 10000 ** (2 * i / d)
        out[2 * i] = np.sin(angle)
        out[2 * i + 1] = np.cos(angle)
    return out
