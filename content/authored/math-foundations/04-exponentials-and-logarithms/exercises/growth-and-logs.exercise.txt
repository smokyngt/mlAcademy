import math
import numpy as np

def compound(P, r, t):
    return P * (1 + r) ** t

def doubling_time(r):
    return math.log(2) / math.log(1 + r)

def log_base(x, b):
    return np.log(x) / np.log(b)

def exp_series(x, terms):
    total = 0.0
    for k in range(terms):
        total += x ** k / math.factorial(k)
    return total

def e_approx(n):
    return (1 + 1 / n) ** n

def cross_entropy(p, q):
    p = np.asarray(p, dtype=float)
    q = np.asarray(q, dtype=float)
    mask = p > 0
    return float(-np.sum(p[mask] * np.log(q[mask])))

def log_sum_exp(z):
    z = np.asarray(z, dtype=float)
    m = z.max()
    return float(m + np.log(np.sum(np.exp(z - m))))
