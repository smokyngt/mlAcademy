import math
import numpy as np

def one_sided(f, a, h=1e-6):
    return (f(a - h), f(a + h))

def limit_estimate(f, a):
    left, right = one_sided(f, a, 1e-6)
    if abs(left - right) < 1e-4:
        return (left + right) / 2
    return None

def is_continuous_at(f, a):
    try:
        value = f(a)
    except (ZeroDivisionError, ValueError):
        return False
    if not math.isfinite(value):
        return False
    lim = limit_estimate(f, a)
    return lim is not None and abs(value - lim) < 1e-4

def delta_for_epsilon(f, a, L, eps):
    delta = 1.0
    while delta > 1e-12:
        xs = np.linspace(a - delta, a + delta, 2001)
        xs = xs[xs != a]
        if all(abs(f(x) - L) < eps for x in xs):
            return delta
        delta /= 2
    return delta

def diff_quotient(f, x, h):
    return (f(x + h) - f(x)) / h

def derivative_estimate(f, x):
    return diff_quotient(f, x, 1e-6)
