import math

def poly_eval(coeffs, x):
    result = 0
    for c in coeffs:
        result = result * x + c
    return result

def quadratic_roots(a, b, c):
    d = b * b - 4 * a * c
    if d < 0:
        return []
    s = math.sqrt(d)
    return sorted({(-b - s) / (2 * a), (-b + s) / (2 * a)})

def shift(f, h):
    def g(x):
        return f(x - h)
    return g

def scale_y(f, c):
    def g(x):
        return c * f(x)
    return g

def compose(f, g):
    def h(x):
        return f(g(x))
    return h

def inverse_by_bisection(f, y, lo, hi):
    for _ in range(60):
        mid = (lo + hi) / 2
        if f(mid) < y:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2
