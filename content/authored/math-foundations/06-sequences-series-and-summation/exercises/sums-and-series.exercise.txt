import numpy as np

def arith_sum(a, d, n):
    return n * (2 * a + (n - 1) * d) / 2

def geom_sum(a, r, n):
    if r == 1:
        return a * n
    return a * (1 - r ** n) / (1 - r)

def geom_infinite(a, r):
    if abs(r) < 1:
        return a / (1 - r)
    return None

def loop_sum_squares(n):
    total = 0
    for k in range(1, n + 1):
        total += k * k
    return total

def closed_sum_squares(n):
    return n * (n + 1) * (2 * n + 1) // 6

def harmonic(n):
    return sum(1 / k for k in range(1, n + 1))

def terms_to_exceed(target):
    total, n = 0.0, 0
    while total <= target:
        n += 1
        total += 1 / n
    return n

def partial_sums(terms):
    return np.cumsum(np.asarray(terms))
