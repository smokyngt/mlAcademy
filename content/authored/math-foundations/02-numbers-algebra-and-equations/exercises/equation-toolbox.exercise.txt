import math

def evaluate(a, b, c):
    return a + b * c ** 2

def solve_linear(a, b, c):
    if a == 0:
        raise ValueError("a must not be 0")
    return (c - b) / a

def solve_inequality(a, b, c):
    bound = (c - b) / a
    return ("<", bound) if a > 0 else (">", bound)

def solve_2x2(a1, b1, c1, a2, b2, c2):
    det = a1 * b2 - a2 * b1
    if det == 0:
        return None
    return ((c1 * b2 - c2 * b1) / det, (a1 * c2 - a2 * c1) / det)

def percent_change(old, new):
    return (new - old) / old * 100

def add_fractions(p, q, r, s):
    num, den = p * s + r * q, q * s
    g = math.gcd(num, den)
    return (num // g, den // g)
