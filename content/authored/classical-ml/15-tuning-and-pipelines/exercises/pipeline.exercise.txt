Build a miniature scikit-learn-style pipeline and use it to expose a classic leak.

- `StandardScaler`: `fit(X, y=None)` stores column means and standard deviations (a zero std becomes 1); `transform(X)` returns `(X - mean) / std`.
- `SelectTopCorr(k)`: `fit(X, y)` keeps the `k` columns with the largest absolute Pearson correlation with `y` (use `argsort` on `-|corr|`); `transform(X)` returns those columns.
- `Ridge(lam)`: `fit(X, y)` centres X and y and solves `(Xc.T Xc + lam I) w = Xc.T (y - ybar)`; `predict(X)` returns `(X - xbar) @ w + ybar`.
- `Pipeline(steps)` with `steps = [(name, obj), ..., (name, model)]`, `fit(X, y)` and `predict(X)` as in the hints.
- `kfold(n, k)` -> list of `(train_idx, test_idx)` using `np.array_split(np.arange(n), k)` as the test folds (no shuffling).
- `r2_score(y, pred)` and `cross_val_r2(make, X, y, k)`: mean R-squared over the folds, where `make()` returns a new unfitted model or pipeline.

import numpy as np

class StandardScaler:
    def fit(self, X, y=None):
        self.m = X.mean(axis=0)
        self.s = X.std(axis=0)
        self.s = np.where(self.s == 0, 1.0, self.s)
        return self
    def transform(self, X):
        return (X - self.m) / self.s

class SelectTopCorr:
    def __init__(self, k):
        self.k = k
    def fit(self, X, y):
        Xc = X - X.mean(axis=0)
        yc = y - y.mean()
        r = np.abs(Xc.T @ yc) / (np.linalg.norm(Xc, axis=0) * np.linalg.norm(yc) + 1e-12)
        self.idx = np.argsort(-r)[: self.k]
        return self
    def transform(self, X):
        return X[:, self.idx]

class Ridge:
    def __init__(self, lam=1.0):
        self.lam = lam
    def fit(self, X, y):
        self.xm = X.mean(axis=0)
        self.ym = y.mean()
        Xc = X - self.xm
        self.w = np.linalg.solve(Xc.T @ Xc + self.lam * np.eye(X.shape[1]), Xc.T @ (y - self.ym))
        return self
    def predict(self, X):
        return (X - self.xm) @ self.w + self.ym

class Pipeline:
    def __init__(self, steps):
        self.steps = steps
    def fit(self, X, y):
        for _, step in self.steps[:-1]:
            step.fit(X, y)
            X = step.transform(X)
        self.steps[-1][1].fit(X, y)
        return self
    def predict(self, X):
        for _, step in self.steps[:-1]:
            X = step.transform(X)
        return self.steps[-1][1].predict(X)

def kfold(n, k):
    idx = np.arange(n)
    return [(np.setdiff1d(idx, f), f) for f in np.array_split(idx, k)]

def r2_score(y, pred):
    return 1 - np.sum((y - pred) ** 2) / np.sum((y - y.mean()) ** 2)

def cross_val_r2(make, X, y, k=5):
    scores = []
    for tr, te in kfold(len(y), k):
        model = make().fit(X[tr], y[tr])
        scores.append(r2_score(y[te], model.predict(X[te])))
    return float(np.mean(scores))
