import numpy as np
import math

def grid_search(f, axis_a, axis_b):
    best = (-np.inf, None)
    for a in axis_a:
        for b in axis_b:
            v = f(a, b)
            if v > best[0]:
                best = (v, (a, b))
    return best

def random_search(f, n, rng):
    best = (-np.inf, None)
    for _ in range(n):
        a, b = rng.uniform(0, 1, 2)
        v = f(a, b)
        if v > best[0]:
            best = (v, (a, b))
    return best

def successive_halving(n_configs, evaluate, eta=3, min_budget=1):
    alive = np.arange(n_configs)
    budget = min_budget
    schedule = []
    while True:
        scores = np.array([evaluate(int(i), budget) for i in alive])
        schedule.append((len(alive), budget))
        if len(alive) == 1:
            return int(alive[0]), schedule
        keep = max(1, len(alive) // eta)
        alive = alive[np.argsort(-scores)[:keep]]
        budget *= eta

def rbf(A, B, length, sigma):
    A, B = np.asarray(A, float), np.asarray(B, float)
    return sigma ** 2 * np.exp(-0.5 * (A[:, None] - B[None, :]) ** 2 / length ** 2)

def gp_posterior(Xtr, ytr, Xte, length, sigma, noise):
    K = rbf(Xtr, Xtr, length, sigma) + noise * np.eye(len(Xtr))
    L = np.linalg.cholesky(K)
    Ks = rbf(Xtr, Xte, length, sigma)
    alpha = np.linalg.solve(L.T, np.linalg.solve(L, ytr))
    mean = Ks.T @ alpha
    v = np.linalg.solve(L, Ks)
    var = sigma ** 2 - np.sum(v ** 2, axis=0)
    return mean, np.sqrt(np.maximum(var, 0))

def expected_improvement(mu, sd, best):
    out = np.zeros(len(mu))
    for i, (m, s) in enumerate(zip(mu, sd)):
        if s < 1e-12:
            out[i] = max(m - best, 0.0)
        else:
            z = (m - best) / s
            Phi = 0.5 * (1 + math.erf(z / math.sqrt(2)))
            phi = math.exp(-z * z / 2) / math.sqrt(2 * math.pi)
            out[i] = (m - best) * Phi + s * phi
    return out

def bayes_opt(fun, n_init, n_iter, seed):
    rng = np.random.default_rng(seed)
    X = list(rng.uniform(0, 1, n_init))
    y = [fun(x) for x in X]
    cand = np.linspace(0, 1, 201)
    for _ in range(n_iter):
        ym = np.mean(y)
        mu, sd = gp_posterior(np.array(X), np.array(y) - ym, cand, 0.1, 1.0, 1e-6)
        ei = expected_improvement(mu + ym, sd, max(y))
        x = cand[int(np.argmax(ei))]
        X.append(x)
        y.append(fun(x))
    return max(y)
