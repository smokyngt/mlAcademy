import numpy as np

def pca_fit(X, k):
    mean = X.mean(axis=0)
    Xc = X - mean
    C = Xc.T @ Xc / (len(X) - 1)
    vals, vecs = np.linalg.eigh(C)
    order = np.argsort(vals)[::-1]
    return mean, vecs[:, order[:k]].T, vals[order]

def transform(X, mean, components):
    return (X - mean) @ components.T

def inverse_transform(Z, mean, components):
    return Z @ components + mean

def explained_variance_ratio(eigenvalues, k):
    return float(np.sum(eigenvalues[:k]) / np.sum(eigenvalues))
