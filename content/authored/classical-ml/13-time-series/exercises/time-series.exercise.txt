Implement four small tools with NumPy.

- `acf(y, max_lag)` -> array of length `max_lag + 1`, the sample autocorrelation (centre by the overall mean; divide every lag by `sum((y - mean)**2)`). `acf[0]` is 1.
- `fit_ar(y, p)` -> `(c, phi)` where `c` is the intercept and `phi` has shape `(p,)`, fitted by ordinary least squares on the model `y[t] = c + phi[0]*y[t-1] + ... + phi[p-1]*y[t-p]`.

import numpy as np

def acf(y, max_lag):
    y = np.asarray(y, float)
    d = y - y.mean()
    den = d @ d
    return np.array([d[k:] @ d[:len(d) - k] / den for k in range(max_lag + 1)])

def fit_ar(y, p):
    y = np.asarray(y, float)
    n = len(y)
    X = np.column_stack([np.ones(n - p)] + [y[p - j:n - j] for j in range(1, p + 1)])
    w = np.linalg.lstsq(X, y[p:], rcond=None)[0]
    return w[0], w[1:]

def forecast_ar(history, c, phi, h):
    hist = list(np.asarray(history, float))
    p = len(phi)
    out = []
    for _ in range(h):
        nxt = c + sum(phi[j] * hist[-1 - j] for j in range(p))
        out.append(nxt)
        hist.append(nxt)
    return np.array(out)

def walk_forward_ar(y, start, p):
    y = np.asarray(y, float)
    preds = []
    for t in range(start, len(y)):
        c, phi = fit_ar(y[:t], p)
        preds.append(forecast_ar(y[:t], c, phi, 1)[0])
    return np.array(preds)
