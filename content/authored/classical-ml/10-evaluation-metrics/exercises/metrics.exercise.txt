Implement, with NumPy only:

- `confusion_counts(y, pred)` -> `(tp, fp, fn, tn)` for 0/1 arrays.

- `auc_pairs(y, scores)` -> $P(S^+ > S^-) + \tfrac12 P(S^+ = S^-)$ computed over all (positive, negative) pairs.

The two AUC functions must agree.

import numpy as np

def confusion_counts(y, pred):
    y = np.asarray(y); pred = np.asarray(pred)
    tp = int(np.sum((pred == 1) & (y == 1)))
    fp = int(np.sum((pred == 1) & (y == 0)))
    fn = int(np.sum((pred == 0) & (y == 1)))
    tn = int(np.sum((pred == 0) & (y == 0)))
    return tp, fp, fn, tn

def prf(tp, fp, fn, tn):
    p = tp / (tp + fp) if tp + fp else 0.0
    r = tp / (tp + fn) if tp + fn else 0.0
    f1 = 2 * p * r / (p + r) if p + r else 0.0
    acc = (tp + tn) / (tp + fp + fn + tn)
    return p, r, f1, acc

def roc_curve(y, scores):
    y = np.asarray(y); scores = np.asarray(scores, dtype=float)
    order = np.argsort(-scores, kind="stable")
    y_s, s_s = y[order], scores[order]
    last = np.r_[np.where(np.diff(s_s) != 0)[0], len(s_s) - 1]
    tps = np.cumsum(y_s)[last]
    fps = np.cumsum(1 - y_s)[last]
    tpr = np.r_[0.0, tps / y.sum()]
    fpr = np.r_[0.0, fps / (1 - y).sum()]
    return fpr, tpr

def auc_trapz(fpr, tpr):
    total = 0.0
    for i in range(1, len(fpr)):
        total += (fpr[i] - fpr[i - 1]) * (tpr[i] + tpr[i - 1]) / 2
    return float(total)

def auc_pairs(y, scores):
    y = np.asarray(y); scores = np.asarray(scores, dtype=float)
    pos = scores[y == 1]; neg = scores[y == 0]
    wins = (pos[:, None] > neg[None, :]) + 0.5 * (pos[:, None] == neg[None, :])
    return float(wins.mean())
