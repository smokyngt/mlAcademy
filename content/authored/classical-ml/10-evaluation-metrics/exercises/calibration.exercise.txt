import numpy as np

def calibration_bins(p, y, n_bins=10):
    p = np.asarray(p, dtype=float); y = np.asarray(y, dtype=float)
    b = np.minimum((p * n_bins).astype(int), n_bins - 1)
    conf, freq, cnt = [], [], []
    for k in range(n_bins):
        m = b == k
        if m.any():
            conf.append(p[m].mean()); freq.append(y[m].mean()); cnt.append(int(m.sum()))
    return np.array(conf), np.array(freq), np.array(cnt)

def ece(p, y, n_bins=10):
    conf, freq, cnt = calibration_bins(p, y, n_bins)
    return float(np.sum(cnt / cnt.sum() * np.abs(freq - conf)))

def brier(p, y):
    p = np.asarray(p, dtype=float); y = np.asarray(y, dtype=float)
    return float(np.mean((p - y) ** 2))

def mae(y, yhat):
    return float(np.mean(np.abs(np.asarray(y) - np.asarray(yhat))))

def rmse(y, yhat):
    return float(np.sqrt(np.mean((np.asarray(y) - np.asarray(yhat)) ** 2)))

def r2(y, yhat):
    y = np.asarray(y, dtype=float); yhat = np.asarray(yhat, dtype=float)
    return float(1 - np.sum((y - yhat) ** 2) / np.sum((y - y.mean()) ** 2))

def cost_threshold(c_fp, c_fn):
    return c_fp / (c_fp + c_fn)
