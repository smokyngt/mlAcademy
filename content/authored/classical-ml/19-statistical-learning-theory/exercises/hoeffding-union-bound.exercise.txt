import numpy as np

def hoeffding_bound(n, eps):
    return 2 * np.exp(-2 * n * eps ** 2)

def union_eps(n, M, delta):
    return np.sqrt(np.log(2 * M / delta) / (2 * n))

def union_n(eps, M, delta):
    return int(np.ceil(np.log(2 * M / delta) / (2 * eps ** 2)))

def finite_class_experiment(n, M, trials, seed, d=5):
    rng = np.random.default_rng(seed)
    W = rng.normal(size=(M, d))
    wstar = rng.normal(size=d)
    def draw(m):
        X = rng.normal(size=(m, d))
        y = np.where(X @ wstar + 0.8 * rng.normal(size=m) > 0, 1, -1)
        return X, y
    Xb, yb = draw(20000)
    true = np.mean(np.sign(Xb @ W.T) != yb[:, None], axis=0)
    max_gap, erm_gap = [], []
    for _ in range(trials):
        X, y = draw(n)
        emp = np.mean(np.sign(X @ W.T) != y[:, None], axis=0)
        max_gap.append(np.max(np.abs(true - emp)))
        j = int(np.argmin(emp))
        erm_gap.append(true[j] - emp[j])
    max_gap = np.array(max_gap)
    return max_gap.mean(), np.quantile(max_gap, 0.95), float(np.mean(erm_gap))
