import numpy as np

def softmax(z):
    z = z - z.max(axis=1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=1, keepdims=True)

def nll(T, z, y):
    p = softmax(z / T)
    return -np.mean(np.log(p[np.arange(len(y)), y] + 1e-12))

def fit_temperature(z, y, lo=0.05, hi=10.0):
    f = lambda t: nll(t, z, y)
    g = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c, d = b - g * (b - a), a + g * (b - a)
    for _ in range(60):
        if f(c) < f(d):
            b = d
        else:
            a = c
        c, d = b - g * (b - a), a + g * (b - a)
    return (a + b) / 2

def ece(P, y, bins=10):
    conf = P.max(axis=1)
    acc = (P.argmax(axis=1) == y)
    edges = np.linspace(0, 1, bins + 1)
    e = 0.0
    for lo, hi in zip(edges[:-1], edges[1:]):
        m = (conf > lo) & (conf <= hi)
        if m.any():
            e += m.mean() * abs(acc[m].mean() - conf[m].mean())
    return e

def blr_posterior(Phi, t, alpha, beta):
    SN = np.linalg.inv(alpha * np.eye(Phi.shape[1]) + beta * Phi.T @ Phi)
    return beta * SN @ Phi.T @ t, SN

def blr_predict(Phi_star, mN, SN, beta):
    epi = np.einsum('ij,jk,ik->i', Phi_star, SN, Phi_star)
    return Phi_star @ mN, epi, epi + 1.0 / beta

def bootstrap_std(Phi, t, Phi_star, B, seed):
    rg = np.random.default_rng(seed)
    N, M = Phi.shape
    out = []
    for _ in range(B):
        i = rg.integers(0, N, N)
        F = Phi[i]
        w = np.linalg.solve(F.T @ F + 1e-3 * np.eye(M), F.T @ t[i])
        out.append(Phi_star @ w)
    return np.array(out).std(axis=0)
