import numpy as np

def make_data(m, rng):
    x = rng.uniform(-1, 1, m)
    y = 2 * x + (0.2 + 0.4 * np.abs(x)) * rng.standard_t(3, m)
    return x, y

def conformal_q(scores, alpha):
    n = len(scores)
    k = int(np.ceil((n + 1) * (1 - alpha)))
    if k > n:
        return np.inf
    return np.sort(scores)[k - 1]

def conformal_interval(pred, q):
    return pred - q, pred + q

def coverage_trials(n_train, n_cal, alpha, trials, seed):
    rng = np.random.default_rng(seed)
    out = []
    for _ in range(trials):
        x, y = make_data(n_train, rng)
        A = np.c_[np.ones(n_train), x]
        w = np.linalg.lstsq(A, y, rcond=None)[0]
        xc, yc = make_data(n_cal, rng)
        q = conformal_q(np.abs(yc - (w[0] + w[1] * xc)), alpha)
        xt, yt = make_data(2000, rng)
        pred = w[0] + w[1] * xt
        lo, hi = conformal_interval(pred, q)
        out.append(np.mean((yt >= lo) & (yt <= hi)))
    return np.array(out)
