- `linear_kernel(A, B)`, `poly_kernel(A, B)` $=(\langle a,b\rangle+1)^2$ and `rbf_kernel(A, B, gamma)` $=\exp(-\gamma\|a-b\|^2)$ all return an `(len(A), len(B))` matrix.

import numpy as np

def linear_kernel(A, B):
    return A @ B.T

def poly_kernel(A, B):
    return (A @ B.T + 1.0) ** 2

def rbf_kernel(A, B, gamma):
    d2 = (A ** 2).sum(1)[:, None] + (B ** 2).sum(1)[None, :] - 2.0 * A @ B.T
    return np.exp(-gamma * np.maximum(d2, 0.0))

def poly_features(X):
    x1, x2 = X[:, 0], X[:, 1]
    s = np.sqrt(2.0)
    return np.stack([np.ones(len(X)), s * x1, s * x2, x1 ** 2, s * x1 * x2, x2 ** 2], axis=1)

def primal_w(alpha, y, X):
    return ((alpha * y)[:, None] * X).sum(axis=0)

def dual_value(alpha, y, K):
    ay = alpha * y
    return float(alpha.sum() - 0.5 * ay @ K @ ay)

def dual_decision(Kxt, alpha, y, b):
    return Kxt @ (alpha * y) + b
