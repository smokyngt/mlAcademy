Train a linear SVM in the primal. Labels are in $\{-1,+1\}$. The objective is
$$J(w,b)=\frac\lambda2\|w\|^2+\frac1N\sum_n\max\{0,\,1-y_n(\langle w,x_n\rangle+b)\}.$$

- `hinge(y, f)` elementwise hinge loss for labels `y` and scores `f`.
- `svm_objective(w, b, X, y, lam)` returns $J$.
- `svm_subgradient(w, b, X, y, lam)` returns `(gw, gb)` as given in the lesson (only margin violators, i.e. $y f<1$, contribute to the data term).

- `margin(w)` returns $1/\|w\|$.

import numpy as np

def hinge(y, f):
    return np.maximum(0.0, 1.0 - y * f)

def svm_objective(w, b, X, y, lam):
    return float(lam / 2 * (w @ w) + hinge(y, X @ w + b).mean())

def svm_subgradient(w, b, X, y, lam):
    viol = y * (X @ w + b) < 1
    gw = lam * w - (y[viol, None] * X[viol]).sum(axis=0) / len(y)
    gb = -y[viol].sum() / len(y)
    return gw, float(gb)

def svm_fit(X, y, lam=0.01, lr=0.5, epochs=500):
    w = np.zeros(X.shape[1])
    b = 0.0
    for t in range(epochs):
        gw, gb = svm_subgradient(w, b, X, y, lam)
        step = lr / (1 + 0.05 * t)
        w = w - step * gw
        b = b - step * gb
    return w, b

def margin(w):
    return float(1.0 / np.linalg.norm(w))
