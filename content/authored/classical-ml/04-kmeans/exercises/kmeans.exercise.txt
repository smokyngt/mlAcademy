import numpy as np

def assign(X, centers):
    d2 = ((X[:, None, :] - centers[None, :, :]) ** 2).sum(-1)
    return d2.argmin(axis=1)

def update(X, labels, centers):
    new = centers.astype(float).copy()
    for j in range(len(centers)):
        pts = X[labels == j]
        if len(pts):
            new[j] = pts.mean(axis=0)
    return new

def inertia(X, labels, centers):
    return float(np.sum((X - centers[labels]) ** 2))

def lloyd(X, centers, iters=100):
    centers = centers.astype(float).copy()
    labels = assign(X, centers)
    history = []
    for _ in range(iters):
        centers = update(X, labels, centers)
        new_labels = assign(X, centers)
        history.append(inertia(X, new_labels, centers))
        if np.array_equal(new_labels, labels):
            labels = new_labels
            break
        labels = new_labels
    return centers, labels, history

def kmeans(X, k, n_init=10, seed=0):
    rng = np.random.default_rng(seed)
    best = None
    for _ in range(n_init):
        init = X[rng.choice(len(X), k, replace=False)]
        centers, labels, hist = lloyd(X, init)
        J = inertia(X, labels, centers)
        if best is None or J < best[2]:
            best = (centers, labels, J)
    return best
