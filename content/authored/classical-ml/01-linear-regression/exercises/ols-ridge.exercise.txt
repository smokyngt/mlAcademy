import numpy as np

def add_intercept(X):
    return np.column_stack([np.ones(len(X)), X])

def fit_ols(X, y):
    return np.linalg.solve(X.T @ X, X.T @ y)

def fit_ridge(X, y, lam):
    d = X.shape[1]
    return np.linalg.solve(X.T @ X + lam * np.eye(d), X.T @ y)

def predict(X, w):
    return X @ w

def r2(y, y_hat):
    return 1.0 - np.sum((y - y_hat) ** 2) / np.sum((y - np.mean(y)) ** 2)
