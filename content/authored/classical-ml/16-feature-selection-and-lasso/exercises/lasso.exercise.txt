We minimise the lasso objective `(1/(2n)) * ||y - X w||^2 + lam * ||w||_1` (no intercept: y is centred, X is standardised).

- `soft_threshold(z, t)`: works on scalars and arrays.
- `lasso_cd(X, y, lam, n_iter=500, tol=1e-10)` -> `w` of shape `(d,)`: cyclic coordinate descent starting from zeros, stopping when the largest change in a sweep is below `tol`.
- `lambda_max(X, y)` -> smallest `lam` giving the all-zero solution.
- `lasso_path(X, y, lams)` -> array `(len(lams), d)`.
- `stability_selection(X, y, lam, B, seed)` -> selection frequency of every feature over `B` half-samples, with `rng = np.random.default_rng(seed)` and `rng.choice(n, n // 2, replace=False)` for each subsample.
- `mutual_information(x, y, bins)` -> mutual information in nats between two continuous 1-D arrays, estimated from a `bins x bins` histogram whose edges are the interior quantiles `np.quantile(v, np.linspace(0, 1, bins + 1)[1:-1])` of each variable (use `np.digitize`).

import numpy as np

def soft_threshold(z, t):
    return np.sign(z) * np.maximum(np.abs(z) - t, 0.0)

def lasso_cd(X, y, lam, n_iter=500, tol=1e-10):
    n, d = X.shape
    w = np.zeros(d)
    r = y.astype(float).copy()
    z = (X ** 2).sum(axis=0) / n
    for _ in range(n_iter):
        w_old = w.copy()
        for j in range(d):
            r += X[:, j] * w[j]
            rho = X[:, j] @ r / n
            w[j] = soft_threshold(rho, lam) / z[j] if z[j] > 0 else 0.0
            r -= X[:, j] * w[j]
        if np.max(np.abs(w - w_old)) < tol:
            break
    return w

def lambda_max(X, y):
    return float(np.max(np.abs(X.T @ y)) / len(y))

def lasso_path(X, y, lams):
    return np.array([lasso_cd(X, y, lam) for lam in lams])

def stability_selection(X, y, lam, B, seed):
    rng = np.random.default_rng(seed)
    n, d = X.shape
    counts = np.zeros(d)
    for _ in range(B):
        idx = rng.choice(n, n // 2, replace=False)
        ys = y[idx] - y[idx].mean()
        counts += np.abs(lasso_cd(X[idx], ys, lam)) > 1e-12
    return counts / B

def mutual_information(x, y, bins=8):
    qs = np.linspace(0, 1, bins + 1)[1:-1]
    a = np.digitize(x, np.quantile(x, qs))
    b = np.digitize(y, np.quantile(y, qs))
    P = np.zeros((bins, bins))
    for i, j in zip(a, b):
        P[i, j] += 1
    P /= P.sum()
    px = P.sum(axis=1, keepdims=True)
    py = P.sum(axis=0, keepdims=True)
    m = P > 0
    return float(np.sum(P[m] * np.log(P[m] / (px @ py)[m])))
