import numpy as np

def mnb_fit(X, y, alpha=1.0):
    classes = np.unique(y)
    log_prior = np.log(np.array([(y == c).mean() for c in classes]))
    counts = np.array([X[y == c].sum(axis=0) for c in classes]).astype(float) + alpha
    with np.errstate(divide="ignore"):
        log_theta = np.log(counts / counts.sum(axis=1, keepdims=True))
    return log_prior, log_theta

def mnb_scores(X, log_prior, log_theta):
    return X @ log_theta.T + log_prior

def softmax_rows(S):
    E = np.exp(S - S.max(axis=1, keepdims=True))
    return E / E.sum(axis=1, keepdims=True)

def mnb_predict(X, log_prior, log_theta):
    return mnb_scores(X, log_prior, log_theta).argmax(axis=1)

def gnb_fit(X, y):
    classes = np.unique(y)
    log_prior = np.log(np.array([(y == c).mean() for c in classes]))
    mu = np.array([X[y == c].mean(axis=0) for c in classes])
    var = np.array([X[y == c].var(axis=0) for c in classes]) + 1e-9
    return log_prior, mu, var

def gnb_scores(X, log_prior, mu, var):
    ll = -0.5 * np.log(2 * np.pi * var)[None, :, :] - (X[:, None, :] - mu[None, :, :]) ** 2 / (2 * var[None, :, :])
    return ll.sum(axis=2) + log_prior

def gnb_predict(X, log_prior, mu, var):
    return gnb_scores(X, log_prior, mu, var).argmax(axis=1)
