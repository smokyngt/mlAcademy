import numpy as np

def pairwise_d2(A, B):
    return ((A[:, None, :] - B[None, :, :]) ** 2).sum(-1)

def _nn(Xtr, Xte, k):
    return np.argsort(pairwise_d2(Xte, Xtr), axis=1, kind="stable")[:, :k]

def knn_predict(Xtr, ytr, Xte, k):
    nn = _nn(Xtr, Xte, k)
    return np.array([np.bincount(ytr[row]).argmax() for row in nn])

def knn_regress(Xtr, ytr, Xte, k):
    return ytr[_nn(Xtr, Xte, k)].mean(axis=1)

def standardize(Xtr, Xte):
    mu, sd = Xtr.mean(0), Xtr.std(0)
    return (Xtr - mu) / sd, (Xte - mu) / sd

def relative_contrast(d, n, seed):
    rng = np.random.default_rng(seed)
    X = rng.uniform(size=(n, d))
    q = rng.uniform(size=d)
    dist = np.linalg.norm(X - q, axis=1)
    return float((dist.max() - dist.min()) / dist.min())
