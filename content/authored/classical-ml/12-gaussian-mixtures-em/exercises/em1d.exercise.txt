Implement EM for $p(x)=\sum_{k=1}^K \pi_k\,\mathcal N(x\mid\mu_k,\sigma_k^2)$ on 1-D data `x` of shape `(N,)`. Parameters are arrays `pi, mu, var` of shape `(K,)` (`var` is the variance $\sigma^2$).

- `gauss_pdf(x, mu, var)` -> `(N, K)` matrix of $\mathcal N(x_n\mid\mu_k,\sigma_k^2)$.
- `e_step(x, pi, mu, var)` -> responsibilities `r` of shape `(N, K)` (rows sum to 1).
- `m_step(x, r, min_var=1e-6)` -> `(pi, mu, var)`: $N_k=\sum_n r_{nk}$, $\mu_k=\frac1{N_k}\sum_n r_{nk}x_n$, $\sigma_k^2=\frac1{N_k}\sum_n r_{nk}(x_n-\mu_k)^2$ using the new $\mu_k$, $\pi_k=N_k/N$; variances are floored at `min_var`.
- `loglik(x, pi, mu, var)` -> $\sum_n\log\sum_k\pi_k\mathcal N(x_n\mid\mu_k,\sigma_k^2)$.
- `em_1d(x, pi, mu, var, iters=50, min_var=1e-6)` -> `(pi, mu, var, history)` where `history[t]` is the log-likelihood before iteration `t`'s update.

import numpy as np

def gauss_pdf(x, mu, var):
    x = np.asarray(x, dtype=float)[:, None]
    return np.exp(-(x - mu[None, :]) ** 2 / (2 * var[None, :])) / np.sqrt(2 * np.pi * var[None, :])

def e_step(x, pi, mu, var):
    w = pi[None, :] * gauss_pdf(x, mu, var)
    return w / w.sum(axis=1, keepdims=True)

def m_step(x, r, min_var=1e-6):
    x = np.asarray(x, dtype=float)
    Nk = r.sum(axis=0)
    mu = (r * x[:, None]).sum(axis=0) / Nk
    var = (r * (x[:, None] - mu[None, :]) ** 2).sum(axis=0) / Nk
    var = np.maximum(var, min_var)
    pi = Nk / len(x)
    return pi, mu, var

def loglik(x, pi, mu, var):
    return float(np.log((pi[None, :] * gauss_pdf(x, mu, var)).sum(axis=1)).sum())

def em_1d(x, pi, mu, var, iters=50, min_var=1e-6):
    hist = []
    for _ in range(iters):
        hist.append(loglik(x, pi, mu, var))
        r = e_step(x, pi, mu, var)
        pi, mu, var = m_step(x, r, min_var)
    return pi, mu, var, hist
