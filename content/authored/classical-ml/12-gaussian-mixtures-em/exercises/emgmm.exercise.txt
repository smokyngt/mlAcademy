import numpy as np

def log_gauss(X, mu, S):
    d = X.shape[1]
    L = np.linalg.cholesky(S)
    z = np.linalg.solve(L, (X - mu).T)
    return -0.5 * (z ** 2).sum(axis=0) - np.log(np.diag(L)).sum() - 0.5 * d * np.log(2 * np.pi)

def _logsumexp(a):
    m = a.max(axis=1, keepdims=True)
    return (m + np.log(np.exp(a - m).sum(axis=1, keepdims=True)))[:, 0]

def em_gmm(X, K, iters=100, seed=0, reg=1e-6):
    rng = np.random.default_rng(seed)
    N, d = X.shape
    mu = X[rng.choice(N, K, replace=False)].astype(float).copy()
    S = np.stack([np.cov(X.T) + reg * np.eye(d)] * K)
    pi = np.ones(K) / K
    hist = []
    for _ in range(iters):
        log_w = np.stack([np.log(pi[k]) + log_gauss(X, mu[k], S[k]) for k in range(K)], axis=1)
        lse = _logsumexp(log_w)
        hist.append(float(lse.sum()))
        r = np.exp(log_w - lse[:, None])
        Nk = r.sum(axis=0)
        mu = (r.T @ X) / Nk[:, None]
        for k in range(K):
            D = X - mu[k]
            S[k] = (r[:, k, None] * D).T @ D / Nk[k] + reg * np.eye(d)
        pi = Nk / N
    log_w = np.stack([np.log(pi[k]) + log_gauss(X, mu[k], S[k]) for k in range(K)], axis=1)
    r = np.exp(log_w - _logsumexp(log_w)[:, None])
    return pi, mu, S, r, hist
