A depth-limited regression tree (`fit_reg_tree`, `pred_reg_tree`) is provided. Build bagging around it.

- `bootstrap_indices(n, rng)` returns `n` indices drawn with replacement.
- `oob_mask(idx, n)` returns a boolean `(n,)` array, True for rows **not** in `idx`.
- `bagging_fit(X, y, B, seed, max_depth=4)` returns `(trees, masks)` with `B` trees, each trained on its own bootstrap sample (one `np.random.default_rng(seed)` shared across the loop), and the matching OOB masks.
- `bagging_predict(trees, X)` returns the mean prediction over trees, shape `(len(X),)`.
- `oob_predict(trees, masks, X)` returns for each row the mean prediction of only the trees for which that row is out-of-bag (NaN if none).
- `oob_mse(y, pred)` is the mean squared error ignoring NaN entries.

import numpy as np

def fit_reg_tree(X, y, max_depth, min_leaf=3):
    if max_depth == 0 or len(y) < 2 * min_leaf:
        return float(y.mean())
    best = None
    for j in range(X.shape[1]):
        u = np.unique(X[:, j])
        for t in (u[:-1] + u[1:]) / 2:
            L = X[:, j] <= t
            if L.sum() < min_leaf or (~L).sum() < min_leaf:
                continue
            sse = ((y[L] - y[L].mean()) ** 2).sum() + ((y[~L] - y[~L].mean()) ** 2).sum()
            if best is None or sse < best[0]:
                best = (sse, j, t)
    if best is None:
        return float(y.mean())
    _, j, t = best
    L = X[:, j] <= t
    return (j, t, fit_reg_tree(X[L], y[L], max_depth - 1, min_leaf), fit_reg_tree(X[~L], y[~L], max_depth - 1, min_leaf))

import numpy as np

def fit_reg_tree(X, y, max_depth, min_leaf=3):
    if max_depth == 0 or len(y) < 2 * min_leaf:
        return float(y.mean())
    best = None
    for j in range(X.shape[1]):
        u = np.unique(X[:, j])
        for t in (u[:-1] + u[1:]) / 2:
            L = X[:, j] <= t
            if L.sum() < min_leaf or (~L).sum() < min_leaf:
                continue
            sse = ((y[L] - y[L].mean()) ** 2).sum() + ((y[~L] - y[~L].mean()) ** 2).sum()
            if best is None or sse < best[0]:
                best = (sse, j, t)
    if best is None:
        return float(y.mean())
    _, j, t = best
    L = X[:, j] <= t
    return (j, t, fit_reg_tree(X[L], y[L], max_depth - 1, min_leaf), fit_reg_tree(X[~L], y[~L], max_depth - 1, min_leaf))

def pred_reg_tree(tree, X):
    out = np.empty(len(X))
    for i, row in enumerate(X):
        node = tree
        while isinstance(node, tuple):
            node = node[2] if row[node[0]] <= node[1] else node[3]
        out[i] = node
    return out

def bootstrap_indices(n, rng):
    return rng.integers(0, n, size=n)

def oob_mask(idx, n):
    mask = np.ones(n, dtype=bool)
    mask[idx] = False
    return mask

def bagging_fit(X, y, B, seed, max_depth=4):
    rng = np.random.default_rng(seed)
    n = len(X)
    trees, masks = [], []
    for _ in range(B):
        idx = bootstrap_indices(n, rng)
        trees.append(fit_reg_tree(X[idx], y[idx], max_depth))
        masks.append(oob_mask(idx, n))
    return trees, masks

def bagging_predict(trees, X):
    return np.mean([pred_reg_tree(t, X) for t in trees], axis=0)

def oob_predict(trees, masks, X):
    s = np.zeros(len(X))
    c = np.zeros(len(X))
    for t, m in zip(trees, masks):
        p = pred_reg_tree(t, X)
        s[m] += p[m]
        c[m] += 1
    out = np.full(len(X), np.nan)
    ok = c > 0
    out[ok] = s[ok] / c[ok]
    return out

def oob_mse(y, pred):
    ok = ~np.isnan(pred)
    return float(np.mean((y[ok] - pred[ok]) ** 2))
