import numpy as np

def _p(y):
    return np.bincount(y) / len(y)

def gini(y):
    if len(y) == 0:
        return 0.0
    return float(1.0 - np.sum(_p(y) ** 2))

def entropy(y):
    if len(y) == 0:
        return 0.0
    p = _p(y)
    p = p[p > 0]
    return float(-np.sum(p * np.log2(p)))

def best_split(X, y, criterion="gini", min_samples_leaf=1):
    imp = gini if criterion == "gini" else entropy
    n, d = X.shape
    parent = imp(y)
    best = (None, None, 0.0)
    for j in range(d):
        u = np.unique(X[:, j])
        for t in (u[:-1] + u[1:]) / 2:
            L = X[:, j] <= t
            nl = int(L.sum())
            nr = n - nl
            if nl < min_samples_leaf or nr < min_samples_leaf:
                continue
            gain = parent - (nl * imp(y[L]) + nr * imp(y[~L])) / n
            if gain > best[2] + 1e-12:
                best = (j, float(t), float(gain))
    return best

def _leaf(y):
    return {"leaf": int(np.bincount(y).argmax())}

def fit_tree(X, y, max_depth=3, min_samples_leaf=1, criterion="gini"):
    if max_depth == 0 or len(np.unique(y)) == 1 or len(y) < 2 * min_samples_leaf:
        return _leaf(y)
    j, t, g = best_split(X, y, criterion, min_samples_leaf)
    if j is None:
        return _leaf(y)
    L = X[:, j] <= t
    return {"f": j, "t": t,
            "l": fit_tree(X[L], y[L], max_depth - 1, min_samples_leaf, criterion),
            "r": fit_tree(X[~L], y[~L], max_depth - 1, min_samples_leaf, criterion)}

def predict_tree(tree, X):
    out = np.empty(len(X), dtype=int)
    for i, row in enumerate(X):
        node = tree
        while "leaf" not in node:
            node = node["l"] if row[node["f"]] <= node["t"] else node["r"]
        out[i] = node["leaf"]
    return out
