import numpy as np

def fit_stump(X, r):
    best = None
    for j in range(X.shape[1]):
        u = np.unique(X[:, j])
        for t in (u[:-1] + u[1:]) / 2:
            L = X[:, j] <= t
            lv, rv = r[L].mean(), r[~L].mean()
            sse = ((r[L] - lv) ** 2).sum() + ((r[~L] - rv) ** 2).sum()
            if best is None or sse < best[0] - 1e-12:
                best = (sse, j, float(t), float(lv), float(rv))
    return best[1:]

def predict_stump(s, X):
    j, t, lv, rv = s
    return np.where(X[:, j] <= t, lv, rv)

def gb_fit(X, y, n_rounds, lr):
    f0 = float(y.mean())
    F = np.full(len(y), f0)
    stumps, history = [], []
    for _ in range(n_rounds):
        s = fit_stump(X, y - F)
        F = F + lr * predict_stump(s, X)
        stumps.append(s)
        history.append(float(np.mean((y - F) ** 2)))
    return {"f0": f0, "stumps": stumps}, history

def gb_predict(model, X, lr):
    F = np.full(len(X), model["f0"])
    for s in model["stumps"]:
        F = F + lr * predict_stump(s, X)
    return F
