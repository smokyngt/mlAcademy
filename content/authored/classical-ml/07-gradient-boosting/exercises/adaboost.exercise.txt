import numpy as np

def ada_round(w, wrong):
    wrong = np.asarray(wrong, float)
    eps = float(np.sum(w * wrong) / np.sum(w))
    alpha = float(np.log((1 - eps) / eps))
    nw = w * np.exp(alpha * wrong)
    return eps, alpha, nw / nw.sum()

def weighted_stump(X, y, w):
    best = None
    for j in range(X.shape[1]):
        u = np.unique(X[:, j])
        for t in (u[:-1] + u[1:]) / 2:
            for s in (1, -1):
                pred = np.where(X[:, j] <= t, s, -s)
                err = np.sum(w[pred != y]) / np.sum(w)
                if best is None or err < best[0] - 1e-12:
                    best = (err, j, float(t), s)
    return best[1:]

def stump_predict(st, X):
    j, t, s = st
    return np.where(X[:, j] <= t, s, -s)

def ada_fit(X, y, n_rounds):
    w = np.full(len(y), 1.0 / len(y))
    model = []
    for _ in range(n_rounds):
        st = weighted_stump(X, y, w)
        wrong = (stump_predict(st, X) != y).astype(float)
        eps = np.sum(w * wrong) / np.sum(w)
        if eps <= 1e-12:
            model.append((10.0, st))
            break
        if eps >= 0.5:
            break
        eps, alpha, w = ada_round(w, wrong)
        model.append((alpha, st))
    return model

def ada_predict(model, X):
    score = sum(a * stump_predict(st, X) for a, st in model)
    return np.where(score >= 0, 1, -1)
