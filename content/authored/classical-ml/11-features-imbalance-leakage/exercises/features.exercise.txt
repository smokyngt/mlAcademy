Implement a tiny preprocessing toolkit. Every object has `fit` (learn statistics from the **training** data only) and `transform` (apply them to any data), so that test data never influences the statistics.

- `Standardizer`: `fit(X)` stores per-column mean and standard deviation (population std, a zero std becomes 1); `transform(X)` returns $(X-\mu)/\sigma$.
- `OneHot`: `fit(values)` learns the sorted list of categories; `transform(values)` returns an `(n, n_categories)` 0/1 float matrix. Unseen categories give an all-zero row.
- `MedianImputer`: `fit(X)` stores column medians ignoring NaN; `transform(X)` returns the matrix with NaN replaced by the train median, followed by one indicator column per original column (1 if it was missing).
- `cross(a, b)`: list of strings `"a_i|b_i"` (a feature cross of two categorical features).
- `hash_index(s, n_buckets)`: deterministic hashing trick, $\left(\sum_i \mathrm{ord}(s_i)\,31^{\,i}\right) \bmod n_{\text{buckets}}$ using Python integers.

import numpy as np

class Standardizer:
    def fit(self, X):
        X = np.asarray(X, dtype=float)
        self.mean_ = X.mean(axis=0)
        std = X.std(axis=0)
        self.std_ = np.where(std == 0, 1.0, std)
        return self
    def transform(self, X):
        return (np.asarray(X, dtype=float) - self.mean_) / self.std_

class OneHot:
    def fit(self, values):
        self.categories_ = sorted(set(values))
        self.index_ = {c: i for i, c in enumerate(self.categories_)}
        return self
    def transform(self, values):
        M = np.zeros((len(values), len(self.categories_)))
        for r, v in enumerate(values):
            j = self.index_.get(v)
            if j is not None:
                M[r, j] = 1.0
        return M

class MedianImputer:
    def fit(self, X):
        self.median_ = np.nanmedian(np.asarray(X, dtype=float), axis=0)
        return self
    def transform(self, X):
        X = np.asarray(X, dtype=float)
        miss = np.isnan(X)
        filled = np.where(miss, self.median_[None, :], X)
        return np.hstack([filled, miss.astype(float)])

def cross(a, b):
    return [f"{x}|{y}" for x, y in zip(a, b)]

def hash_index(s, n_buckets):
    h = 0
    for i, ch in enumerate(s):
        h += ord(ch) * 31 ** i
    return h % n_buckets
