Labels that are pure coin flips cannot be predicted. Yet a pipeline that looks at the test rows while choosing features will appear to predict them.

Implement:

- `select_top_k(X, y, k)` -> indices of the `k` columns with the largest absolute correlation with `y`.
- `centroid_fit(X, y)` -> `(2, d)` matrix of class means; `centroid_predict(means, X)` -> nearest-mean labels.

If `leak` select features on all rows, otherwise only on the train rows. Fit the centroid classifier on the train rows and return the test accuracy.
- `class_weights(y)` -> `{0: w0, 1: w1}` with $w_c = N/(2\,n_c)$.
- `oversample(X, y, seed)` -> training set where minority rows are duplicated (random draws with replacement from the minority rows) until classes are balanced. Apply it to the train split only.
- `best_f1_threshold(y, p)` -> `(threshold, f1)` maximising F1 over the distinct predicted values, predicting positive when `p >= threshold`.

import numpy as np

def select_top_k(X, y, k):
    Xc = X - X.mean(axis=0)
    yc = y - y.mean()
    corr = np.abs(Xc.T @ yc) / (np.sqrt((Xc ** 2).sum(axis=0)) * np.sqrt((yc ** 2).sum()) + 1e-12)
    return np.argsort(-corr, kind="stable")[:k]

def centroid_fit(X, y):
    return np.stack([X[y == 0].mean(axis=0), X[y == 1].mean(axis=0)])

def centroid_predict(means, X):
    d2 = ((X[:, None, :] - means[None, :, :]) ** 2).sum(-1)
    return d2.argmin(axis=1)

def experiment(seed, leak, n=40, d=500, k=10):
    rng = np.random.default_rng(seed)
    X = rng.normal(size=(n, d))
    y = np.r_[np.zeros(n // 2), np.ones(n // 2)].astype(int)
    rng.shuffle(y)
    idx = rng.permutation(n)
    tr, te = idx[: n // 2], idx[n // 2:]
    if leak:
        cols = select_top_k(X, y, k)
    else:
        cols = select_top_k(X[tr], y[tr], k)
    means = centroid_fit(X[tr][:, cols], y[tr])
    pred = centroid_predict(means, X[te][:, cols])
    return float((pred == y[te]).mean())

def class_weights(y):
    y = np.asarray(y)
    n = len(y)
    return {c: n / (2 * int((y == c).sum())) for c in (0, 1)}

def oversample(X, y, seed):
    rng = np.random.default_rng(seed)
    n0, n1 = int((y == 0).sum()), int((y == 1).sum())
    minority = 1 if n1 < n0 else 0
    idx_min = np.where(y == minority)[0]
    extra = rng.choice(idx_min, abs(n0 - n1), replace=True)
    keep = np.r_[np.arange(len(y)), extra]
    return X[keep], y[keep]

def best_f1_threshold(y, p):
    y = np.asarray(y); p = np.asarray(p, dtype=float)
    best_t, best_f = None, -1.0
    for t in sorted(set(p.tolist()), reverse=True):
        pred = (p >= t).astype(int)
        tp = int(((pred == 1) & (y == 1)).sum())
        fp = int(((pred == 1) & (y == 0)).sum())
        fn = int(((pred == 0) & (y == 1)).sum())
        f = 2 * tp / (2 * tp + fp + fn) if tp else 0.0
        if f > best_f + 1e-12:
            best_t, best_f = t, f
    return best_t, best_f
