import numpy as np

def fit_ridge(X, y, lam):
    return np.linalg.solve(X.T @ X + lam * np.eye(X.shape[1]), X.T @ y)

def poly_features(x, degree):
    return np.vander(x, degree + 1, increasing=True)

def kfold_indices(n, k, seed=0):
    idx = np.random.default_rng(seed).permutation(n)
    return np.array_split(idx, k)

def cv_mse(x, y, degree, lam=1e-8, k=5, seed=0):
    errs = []
    for val in kfold_indices(len(x), k, seed):
        tr = np.setdiff1d(np.arange(len(x)), val)
        w = fit_ridge(poly_features(x[tr], degree), y[tr], lam)
        errs.append(np.mean((poly_features(x[val], degree) @ w - y[val]) ** 2))
    return float(np.mean(errs))

def select_degree(x, y, degrees, k=5, seed=0):
    degrees = list(degrees)
    scores = [cv_mse(x, y, d, k=k, seed=seed) for d in degrees]
    return degrees[int(np.argmin(scores))]
