import numpy as np

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -500, 500)))

def loss(w, X, y, lam=0.0):
    z = X @ w
    return float(np.mean(np.logaddexp(0.0, z) - y * z) + 0.5 * lam * np.sum(w[1:] ** 2))

def grad(w, X, y, lam=0.0):
    g = X.T @ (sigmoid(X @ w) - y) / len(y)
    reg = lam * w
    reg[0] = 0.0
    return g + reg

def fit(X, y, lr=0.5, steps=500, lam=0.0):
    w = np.zeros(X.shape[1])
    for _ in range(steps):
        w = w - lr * grad(w, X, y, lam)
    return w

def predict(X, w):
    return (X @ w > 0).astype(int)
