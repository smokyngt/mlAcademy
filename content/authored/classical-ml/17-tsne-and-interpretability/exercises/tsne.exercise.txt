import numpy as np

def sq_dists(X):
    s = (X ** 2).sum(axis=1)
    D = s[:, None] + s[None, :] - 2 * X @ X.T
    D = np.maximum(D, 0)
    np.fill_diagonal(D, 0)
    return D

def conditional_p(D, perplexity):
    n = len(D)
    P = np.zeros((n, n))
    target = np.log(perplexity)
    for i in range(n):
        lo, hi, beta = 0.0, np.inf, 1.0
        di = np.delete(D[i], i)
        for _ in range(60):
            p = np.exp(-(di - di.min()) * beta)
            p /= p.sum()
            H = -np.sum(p * np.log(p + 1e-300))
            if H > target:
                lo = beta
                beta = beta * 2 if hi == np.inf else (beta + hi) / 2
            else:
                hi = beta
                beta = (beta + lo) / 2
        P[i, np.arange(n) != i] = p
    return P

def joint_p(X, perplexity):
    Pc = conditional_p(sq_dists(X), perplexity)
    return (Pc + Pc.T) / (2 * len(X))

def q_matrix(Y):
    W = 1.0 / (1.0 + sq_dists(Y))
    np.fill_diagonal(W, 0)
    return W / W.sum(), W

def kl_divergence(P, Q):
    m = P > 0
    return float(np.sum(P[m] * np.log(P[m] / Q[m])))

def tsne_grad(P, Y):
    Q, W = q_matrix(Y)
    PQ = (P - Q) * W
    return 4 * (np.diag(PQ.sum(axis=1)) - PQ) @ Y

def tsne_step(P, Y, lr):
    return Y - lr * tsne_grad(P, Y)

def kernel_pca(X, gamma, k):
    n = len(X)
    K = np.exp(-gamma * sq_dists(X))
    J = np.eye(n) - np.ones((n, n)) / n
    vals, vecs = np.linalg.eigh(J @ K @ J)
    order = np.argsort(vals)[::-1][:k]
    return vecs[:, order] * np.sqrt(np.maximum(vals[order], 0)), vals[order]
