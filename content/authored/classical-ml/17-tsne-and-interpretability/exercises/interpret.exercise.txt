import numpy as np

rng = np.random.default_rng(0)
n = 400
X = rng.normal(size=(n, 4))
y = 2 * X[:, 0] + X[:, 1] ** 2 + 0.3 * rng.normal(size=n)
def F(A):
    return np.column_stack([np.ones(len(A)), A[:, 0], A[:, 1], A[:, 1] ** 2, A[:, 2], A[:, 3]])
beta = np.linalg.lstsq(F(X), y, rcond=None)[0]
predict = lambda A: F(A) @ beta

check("irrelevant columns have importance ~ 0", lambda: abs(imp[2]) < 0.01 and abs(imp[3]) < 0.01)
check("ranking x0 > x1 > others", lambda: list(np.argsort(-imp)[:2]) == [0, 1])

import numpy as np
import itertools
import math

def permutation_importance(predict, X, y, n_repeats, rng):
    base = np.mean((y - predict(X)) ** 2)
    imp = np.zeros(X.shape[1])
    for j in range(X.shape[1]):
        vals = []
        for _ in range(n_repeats):
            Xp = X.copy()
            Xp[:, j] = rng.permutation(Xp[:, j])
            vals.append(np.mean((y - predict(Xp)) ** 2) - base)
        imp[j] = np.mean(vals)
    return base, imp

def partial_dependence(predict, X, j, grid):
    out = []
    for g in grid:
        Xg = X.copy()
        Xg[:, j] = g
        out.append(predict(Xg).mean())
    return np.array(out)

def shapley_values(predict, x, background):
    d = len(x)
    def v(S):
        Z = background.copy()
        S = list(S)
        Z[:, S] = x[S]
        return predict(Z).mean()
    phi = np.zeros(d)
    for j in range(d):
        others = [k for k in range(d) if k != j]
        for r in range(d):
            w = math.factorial(r) * math.factorial(d - r - 1) / math.factorial(d)
            for S in itertools.combinations(others, r):
                phi[j] += w * (v(S + (j,)) - v(S))
    return phi
