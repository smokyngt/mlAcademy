Build three small anomaly tools.

1. `zscores(x)` -> classical z-scores (population std, `np.std`), and `robust_zscores(x)` -> `0.6745 * (x - median) / MAD` with `MAD = median(|x - median|)`.
2. `mahalanobis(X)` -> array of length N: distance of every row from the sample mean using the sample covariance `np.cov(X, rowvar=False)`.
3. `c_factor(n)`: the average path length of an unsuccessful search in a binary search tree: `0` for `n <= 1`, `1` for `n == 2`, otherwise `2*(ln(n-1) + 0.5772156649) - 2*(n-1)/n`.
4. `IsolationForest(n_trees, psi, seed)` with `fit(X)` and `score(X)`. Use `rng = np.random.default_rng(seed)`; every tree is grown on `psi` rows sampled WITHOUT replacement (`rng.choice(len(X), psi, replace=False)`), with depth limit `ceil(log2(psi))`. At each node choose a column with `rng.integers(d)`; if the column is constant, stop; otherwise threshold `rng.uniform(min, max)`, left = `x < threshold`. A leaf with `s` rows contributes `depth + c_factor(s)`. `score` returns `2 ** (-mean_path / c_factor(psi))` for each row: near 1 means anomalous.
5. `flag_top_fraction(scores, contamination)` -> boolean mask flagging the `ceil(contamination * n)` highest scores, and `precision_at_k(scores, labels, k)` -> share of true anomalies (`labels == 1`) among the `k` highest scores.

import numpy as np

def zscores(x):
    x = np.asarray(x, float)
    return (x - x.mean()) / x.std()

def robust_zscores(x):
    x = np.asarray(x, float)
    med = np.median(x)
    mad = np.median(np.abs(x - med))
    return 0.6745 * (x - med) / mad

def mahalanobis(X):
    X = np.asarray(X, float)
    Xc = X - X.mean(axis=0)
    Sinv = np.linalg.inv(np.cov(X, rowvar=False))
    return np.sqrt(np.einsum("ij,jk,ik->i", Xc, Sinv, Xc))

def c_factor(n):
    if n <= 1:
        return 0.0
    if n == 2:
        return 1.0
    return 2 * (np.log(n - 1) + 0.5772156649) - 2 * (n - 1) / n

class _Tree:
    def __init__(self, rng, max_depth):
        self.rng, self.max_depth = rng, max_depth

    def fit(self, X, depth=0):
        n = len(X)
        if depth >= self.max_depth or n <= 1:
            self.leaf, self.size = True, n
            return self
        j = int(self.rng.integers(X.shape[1]))
        lo, hi = X[:, j].min(), X[:, j].max()
        if lo == hi:
            self.leaf, self.size = True, n
            return self
        self.leaf, self.j, self.t = False, j, self.rng.uniform(lo, hi)
        m = X[:, j] < self.t
        self.left = _Tree(self.rng, self.max_depth).fit(X[m], depth + 1)
        self.right = _Tree(self.rng, self.max_depth).fit(X[~m], depth + 1)
        return self

    def path(self, x, depth=0):
        if self.leaf:
            return depth + c_factor(self.size)
        return (self.left if x[self.j] < self.t else self.right).path(x, depth + 1)

class IsolationForest:
    def __init__(self, n_trees=100, psi=64, seed=0):
        self.n_trees, self.psi, self.seed = n_trees, psi, seed

    def fit(self, X):
        X = np.asarray(X, float)
        rng = np.random.default_rng(self.seed)
        self.psi_ = min(self.psi, len(X))
        depth = int(np.ceil(np.log2(self.psi_)))
        self.trees = [_Tree(rng, depth).fit(X[rng.choice(len(X), self.psi_, replace=False)])
                      for _ in range(self.n_trees)]
        return self

    def score(self, X):
        X = np.asarray(X, float)
        return np.array([2 ** (-np.mean([t.path(x) for t in self.trees]) / c_factor(self.psi_)) for x in X])

def flag_top_fraction(scores, contamination):
    scores = np.asarray(scores)
    k = int(np.ceil(contamination * len(scores)))
    thr = np.sort(scores)[::-1][k - 1]
    return scores >= thr

def precision_at_k(scores, labels, k):
    idx = np.argsort(-np.asarray(scores))[:k]
    return float(np.mean(np.asarray(labels)[idx] == 1))
