import numpy as np

def entropy(p):
    p = np.asarray(p, float)
    p = p[p > 0]
    return -np.sum(p * np.log2(p))

def cross_entropy(p, q):
    p = np.asarray(p, float); q = np.asarray(q, float)
    m = p > 0
    return -np.sum(p[m] * np.log2(q[m]))

def kl(p, q):
    return cross_entropy(p, q) - entropy(p)

def softmax_xent(Z, y):
    Z = np.asarray(Z, float)
    Zs = Z - Z.max(axis=1, keepdims=True)
    lse = np.log(np.exp(Zs).sum(axis=1))
    return np.mean(lse - Zs[np.arange(len(y)), y])

def perplexity(probs):
    return float(np.exp(-np.mean(np.log(probs))))

def mutual_information(J):
    J = np.asarray(J, float)
    px = J.sum(axis=1, keepdims=True); py = J.sum(axis=0, keepdims=True)
    m = J > 0
    return np.sum(J[m] * np.log2(J[m] / (px @ py)[m]))
