import numpy as np

def eigvals2(A):
    a, b, c, d = A[0, 0], A[0, 1], A[1, 0], A[1, 1]
    tr, det = a + d, a * d - b * c
    disc = np.sqrt(tr * tr - 4 * det)
    return np.array(sorted([(tr + disc) / 2, (tr - disc) / 2], reverse=True))

def eigvec2(A, lam):
    a, b, c, d = A[0, 0], A[0, 1], A[1, 0], A[1, 1]
    for v in (np.array([b, lam - a]), np.array([lam - d, c])):
        n = np.linalg.norm(v)
        if n > 1e-12:
            return v / n
    return np.array([1.0, 0.0])

def power_iteration(A, iters=200):
    v = np.ones(A.shape[0]) + np.arange(A.shape[0]) * 0.1
    v = v / np.linalg.norm(v)
    for _ in range(iters):
        w = A @ v
        v = w / np.linalg.norm(w)
    return float(v @ A @ v), v

def matrix_power_diag(A, k):
    vals, P = np.linalg.eig(A)
    return np.real(P @ np.diag(vals ** k) @ np.linalg.inv(P))

def svd_via_eigh(A):
    lam, V = np.linalg.eigh(A.T @ A)
    order = np.argsort(lam)[::-1]
    lam, V = lam[order], V[:, order]
    S = np.sqrt(np.clip(lam, 0, None))
    U = A @ V / S
    return U, S, V.T

def low_rank(U, S, Vt, k):
    return (U[:, :k] * S[:k]) @ Vt[:k]
