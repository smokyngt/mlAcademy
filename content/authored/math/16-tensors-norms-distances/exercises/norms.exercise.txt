import numpy as np

def lp_norm(x, p):
    a = np.abs(x)
    if np.isinf(p):
        return a.max()
    return (a ** p).sum() ** (1.0 / p)

def frobenius(A):
    return np.sqrt((A ** 2).sum())

def spectral(A):
    return np.linalg.svd(A, compute_uv=False)[0]

def cosine(a, b):
    return a @ b / (np.linalg.norm(a) * np.linalg.norm(b))

def pairwise_sq_dists(X, Y):
    D2 = (X ** 2).sum(1)[:, None] + (Y ** 2).sum(1)[None, :] - 2 * X @ Y.T
    return np.maximum(D2, 0)

def mahalanobis(x, mu, S):
    d = x - mu
    return np.sqrt(d @ np.linalg.solve(S, d))

def distance_contrast(D, n, seed):
    rng = np.random.default_rng(seed)
    P = rng.random((n, D))
    q = rng.random(D)
    d = np.linalg.norm(P - q, axis=1)
    return (d.max() - d.min()) / d.min()

def shell_fraction(eps, D):
    return 1 - (1 - eps) ** D
