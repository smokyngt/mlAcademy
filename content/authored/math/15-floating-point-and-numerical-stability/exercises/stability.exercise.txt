NumPy only. Implement:

- `softmax_naive(z)` - the textbook formula `exp(z) / sum(exp(z))`; `softmax_stable(z)` - the max-shifted version.
- `logsumexp_stable(z)` - `c + log(sum(exp(z - c)))` with `c = max(z)`.
- `var_raw(x)` - population variance by `mean(x**2) - mean(x)**2`.
- `welford(xs)` - one pass over an iterable, returning `(mean, population_variance, sample_variance)`.
- `kahan_sum(v)` - Kahan compensated sum of a float32 array, computed in float32.
- `central_diff_error(h)` - absolute error of the central difference of `exp` at `x=1` with step `h` (compare with `np.e`).
- `to_bfloat16(x)` - round a float32 value to the nearest bfloat16 (keep the top 16 bits of the float32 pattern, round to nearest even) and return it as a float32.

import numpy as np

def softmax_naive(z):
    e = np.exp(z)
    return e / e.sum()

def softmax_stable(z):
    e = np.exp(z - np.max(z))
    return e / e.sum()

def logsumexp_stable(z):
    c = np.max(z)
    return c + np.log(np.sum(np.exp(z - c)))

def var_raw(x):
    return np.mean(x ** 2) - np.mean(x) ** 2

def welford(xs):
    n, m, M2 = 0, 0.0, 0.0
    for x in xs:
        n += 1
        d = x - m
        m += d / n
        M2 += d * (x - m)
    return m, M2 / n, M2 / (n - 1)

def kahan_sum(v):
    s = np.float32(0)
    c = np.float32(0)
    for t in v:
        y = np.float32(t - c)
        tt = np.float32(s + y)
        c = np.float32((tt - s) - y)
        s = tt
    return s

def central_diff_error(h):
    return abs((np.exp(1.0 + h) - np.exp(1.0 - h)) / (2 * h) - np.e)

def to_bfloat16(x):
    b = np.array([x], dtype=np.float32).view(np.uint32)
    b = (b + np.uint32(0x7FFF) + ((b >> np.uint32(16)) & np.uint32(1))) & np.uint32(0xFFFF0000)
    return b.view(np.float32)[0]
