Implement, with NumPy only:

- `discrete_moments(x, p)` - mean and variance of a discrete random variable with values `x` and probabilities `p` (shape `(K,)`). Compute the variance from the centred values.
- `var_raw(x)` and `var_two_pass(x)` - variance of a sample (divide by `N`) via `mean(x**2) - mean(x)**2` and via centring first.
- `emp_cov(X, ddof=0)` - empirical covariance of `X` with shape `(N, D)`, returning `(D, D)`, dividing by `N - ddof`.
- `affine_gaussian(mu, Sigma, A, b)` - mean and covariance of `A x + b` for `x ~ N(mu, Sigma)`.
- `sample_mvn(mu, Sigma, n, rng)` - `n` samples of `N(mu, Sigma)` as an `(n, D)` array, using the Cholesky factor of `Sigma` and `rng.standard_normal((n, D))`.
- `gaussian_mle(x)` - `(mu_ML, var_ML)` of a 1-D sample (divide by `N`).
- `gaussian_loglik(x, mu, var)` - total log-likelihood of a 1-D sample.

import numpy as np

def discrete_moments(x, p):
    x = np.asarray(x, float); p = np.asarray(p, float)
    mean = np.sum(x * p)
    var = np.sum((x - mean) ** 2 * p)
    return mean, var

def var_raw(x):
    x = np.asarray(x, float)
    return np.mean(x ** 2) - np.mean(x) ** 2

def var_two_pass(x):
    x = np.asarray(x, float)
    return np.mean((x - x.mean()) ** 2)

def emp_cov(X, ddof=0):
    Xc = X - X.mean(axis=0)
    return Xc.T @ Xc / (X.shape[0] - ddof)

def affine_gaussian(mu, Sigma, A, b):
    return A @ mu + b, A @ Sigma @ A.T

def sample_mvn(mu, Sigma, n, rng):
    L = np.linalg.cholesky(Sigma)
    z = rng.standard_normal((n, len(mu)))
    return mu + z @ L.T

def gaussian_mle(x):
    x = np.asarray(x, float)
    mu = x.mean()
    return mu, np.mean((x - mu) ** 2)

def gaussian_loglik(x, mu, var):
    x = np.asarray(x, float)
    return -0.5 * len(x) * np.log(2 * np.pi * var) - np.sum((x - mu) ** 2) / (2 * var)
