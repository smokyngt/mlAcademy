import numpy as np

def variance_ml(x):
    x = np.asarray(x, dtype=float)
    return np.sum((x - x.mean()) ** 2) / len(x)

def variance_unbiased(x):
    x = np.asarray(x, dtype=float)
    return np.sum((x - x.mean()) ** 2) / (len(x) - 1)

def bias_of_variance(N, ntrials, seed):
    data = np.random.default_rng(seed).normal(size=(ntrials, N))
    return data.var(axis=1).mean(), data.var(axis=1, ddof=1).mean()

def ridge_1d(x, y, lam):
    return (x @ y) / (x @ x + lam)

def lam_from_prior(sigma2, tau2):
    return sigma2 / tau2

def t_interval(x, tcrit):
    x = np.asarray(x, dtype=float)
    se = x.std(ddof=1) / np.sqrt(len(x))
    return x.mean() - tcrit * se, x.mean() + tcrit * se

def paired_t(a, b):
    d = np.asarray(b) - np.asarray(a)
    return d.mean() / (d.std(ddof=1) / np.sqrt(len(d)))

def bootstrap_ci(d, B, seed):
    rng = np.random.default_rng(seed)
    means = np.array([rng.choice(d, size=len(d)).mean() for _ in range(B)])
    return np.percentile(means, [2.5, 97.5])
