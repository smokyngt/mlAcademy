import numpy as np

def dot(a, b):
    return np.sum(a * b)

def norm(a):
    return np.sqrt(dot(a, a))

def cosine_similarity(a, b):
    return dot(a, b) / (norm(a) * norm(b))

def project(a, b):
    return dot(a, b) / dot(b, b) * b
