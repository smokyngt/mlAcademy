import numpy as np

def marginal_x(P):
    return P.sum(axis=1)

def marginal_y(P):
    return P.sum(axis=0)

def conditional_y_given_x(P):
    return P / P.sum(axis=1, keepdims=True)

def is_independent(P, tol=1e-9):
    return bool(np.allclose(P, np.outer(marginal_x(P), marginal_y(P)), atol=tol))

def posterior(prior, likelihood):
    joint = prior * likelihood
    return joint / joint.sum()

def sequential_posterior(prior, likelihoods):
    p = prior
    for lk in likelihoods:
        p = posterior(p, lk)
    return p
