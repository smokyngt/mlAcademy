import numpy as np

def gauss_solve(A, b):
    A = np.array(A, float); b = np.array(b, float)
    n = len(b)
    for k in range(n):
        p = k + np.argmax(np.abs(A[k:, k]))
        if p != k:
            A[[k, p]] = A[[p, k]]
            b[[k, p]] = b[[p, k]]
        for i in range(k + 1, n):
            m = A[i, k] / A[k, k]
            A[i, k:] -= m * A[k, k:]
            b[i] -= m * b[k]
    x = np.zeros(n)
    for i in range(n - 1, -1, -1):
        x[i] = (b[i] - A[i, i + 1:] @ x[i + 1:]) / A[i, i]
    return x

def rref(A, tol=1e-10):
    R = np.array(A, float)
    m, n = R.shape
    piv = []
    r = 0
    for c in range(n):
        if r >= m:
            break
        p = r + np.argmax(np.abs(R[r:, c]))
        if abs(R[p, c]) <= tol:
            continue
        R[[r, p]] = R[[p, r]]
        R[r] = R[r] / R[r, c]
        for i in range(m):
            if i != r:
                R[i] = R[i] - R[i, c] * R[r]
        piv.append(c)
        r += 1
    R[np.abs(R) < tol] = 0.0
    return R, piv

def rank(A):
    return len(rref(A)[1])

def null_space(A):
    R, piv = rref(A)
    n = R.shape[1]
    free = [c for c in range(n) if c not in piv]
    N = np.zeros((n, len(free)))
    for k, f in enumerate(free):
        N[f, k] = 1.0
        for j, pc in enumerate(piv):
            N[pc, k] = -R[j, f]
    return N

def solve_general(A, b):
    A = np.array(A, float)
    aug = np.c_[A, np.asarray(b, float)]
    R, piv = rref(aug)
    n = A.shape[1]
    if n in piv:
        return None
    x = np.zeros(n)
    for j, pc in enumerate(piv):
        x[pc] = R[j, -1]
    return x, null_space(A)

def lstsq_normal(A, b):
    return gauss_solve(A.T @ A, A.T @ b)

def cond2(A):
    s = np.linalg.svd(A, compute_uv=False)
    return s.max() / s.min()
