import numpy as np

def hessian_fd(f, x, h=1e-3):
    x = np.asarray(x, float)
    n = len(x)
    H = np.zeros((n, n))
    E = np.eye(n) * h
    for i in range(n):
        for j in range(n):
            H[i, j] = (f(x + E[i] + E[j]) - f(x + E[i] - E[j]) - f(x - E[i] + E[j]) + f(x - E[i] - E[j])) / (4 * h * h)
    return (H + H.T) / 2

def classify(H, tol=1e-8):
    ev = np.linalg.eigvalsh(H)
    if np.all(ev > tol):
        return "min"
    if np.all(ev < -tol):
        return "max"
    if ev.min() < -tol and ev.max() > tol:
        return "saddle"
    return "degenerate"

def is_convex_quadratic(Q, tol=1e-10):
    return bool(np.linalg.eigvalsh((Q + Q.T) / 2).min() >= -tol)

def gradient_descent(grad, x0, lr, iters):
    x = np.array(x0, float)
    for _ in range(iters):
        x = x - lr * grad(x)
    return x

def newton(grad, hess, x0, iters):
    x = np.array(x0, float)
    for _ in range(iters):
        x = x + np.linalg.solve(hess(x), -grad(x))
    return x

def equality_qp(Q, c, A, b):
    n, m = Q.shape[0], A.shape[0]
    K = np.block([[Q, A.T], [A, np.zeros((m, m))]])
    rhs = np.concatenate([c, b])
    sol = np.linalg.solve(K, rhs)
    return sol[:n], sol[n:]

def dual_value(lam):
    return lam - lam ** 2 / 4
