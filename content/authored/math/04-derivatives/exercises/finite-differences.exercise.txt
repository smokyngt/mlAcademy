import numpy as np

def forward_diff(f, x, h):
    return (f(x + h) - f(x)) / h

def central_diff(f, x, h):
    return (f(x + h) - f(x - h)) / (2 * h)

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def sigmoid_prime(x):
    s = sigmoid(x)
    return s * (1.0 - s)

def best_step(f, true_derivative, x, hs):
    errs = np.array([abs(central_diff(f, x, h) - true_derivative) for h in hs])
    return hs[int(np.argmin(errs))]

hs = np.logspace(-12, 0, 49)
print("best h for sin at 1:", best_step(np.sin, np.cos(1.0), 1.0, hs))
