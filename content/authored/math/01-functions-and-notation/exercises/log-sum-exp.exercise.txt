#@@ solution
import numpy as np

def log_likelihood(p):
    return float(np.sum(np.log(p)))

def log_sum_exp(x):
    x = np.asarray(x, dtype=float)
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(x - m))))
