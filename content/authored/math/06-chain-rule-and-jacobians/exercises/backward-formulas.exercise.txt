import numpy as np

def linear_backward(W, x, g):
    dx = W.T @ g
    dW = np.outer(g, x)
    return dx, dW

def relu_backward(z, g):
    return g * (z > 0)

def numerical_jacobian(f, x, h=1e-5):
    x = np.asarray(x, dtype=float)
    m = np.asarray(f(x)).size
    J = np.zeros((m, x.size))
    for j in range(x.size):
        xp = x.copy(); xm = x.copy()
        xp[j] += h; xm[j] -= h
        J[:, j] = (np.asarray(f(xp)) - np.asarray(f(xm))) / (2 * h)
    return J
