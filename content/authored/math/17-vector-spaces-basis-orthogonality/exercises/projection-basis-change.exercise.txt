import numpy as np

def projection_matrix(B):
    return B @ np.linalg.solve(B.T @ B, B.T)

def project(B, x):
    return B @ np.linalg.solve(B.T @ B, B.T @ x)

def is_projection(P):
    return bool(np.allclose(P @ P, P)), bool(np.allclose(P, P.T))

def to_coords(S, x):
    return np.linalg.solve(S, x)

def from_coords(S, c):
    return S @ c

def matrix_in_basis(S, M):
    return np.linalg.solve(S, M @ S)
