import numpy as np

def exp_inverse(u, lam):
    return -np.log(1 - u) / lam

def rejection_beta22(n, seed):
    rng = np.random.default_rng(seed)
    x = rng.random(n)
    u = rng.random(n)
    return x[u < 6 * x * (1 - x) / 1.5]

def mc_pi(n, seed):
    p = np.random.default_rng(seed).random((n, 2))
    return 4 * np.mean((p ** 2).sum(axis=1) <= 1)

def tail_naive(n, seed):
    z = np.random.default_rng(seed).normal(size=n)
    return np.mean(z > 3)

def tail_importance(n, seed):
    q = np.random.default_rng(seed).normal(3, 1, size=n)
    w = np.exp(-q ** 2 / 2 + (q - 3) ** 2 / 2)
    return np.mean((q > 3) * w)

def metropolis(logp, n, step, seed):
    rng = np.random.default_rng(seed)
    x, acc = 0.0, 0
    out = np.empty(n)
    for t in range(n):
        xp = x + step * rng.normal()
        if np.log(rng.random()) < logp(xp) - logp(x):
            x = xp
            acc += 1
        out[t] = x
    return out, acc / n
