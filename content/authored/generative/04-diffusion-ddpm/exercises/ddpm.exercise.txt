import numpy as np

def linear_betas(T, b0=1e-4, b1=0.02):
    return np.linspace(b0, b1, T)

def alpha_bars(betas):
    return np.cumprod(1 - betas)

def q_sample(x0, t, abar, eps):
    return np.sqrt(abar[t]) * x0 + np.sqrt(1 - abar[t]) * eps

def analytic_x0_hat(xt, t, abar, m0, s0):
    a = abar[t]
    return m0 + np.sqrt(a) * s0 ** 2 / (a * s0 ** 2 + 1 - a) * (xt - np.sqrt(a) * m0)

def analytic_eps(xt, t, abar, m0, s0):
    return (xt - np.sqrt(abar[t]) * analytic_x0_hat(xt, t, abar, m0, s0)) / np.sqrt(1 - abar[t])

def posterior_mean_var(x0, xt, t, betas):
    ab = alpha_bars(betas)
    al = 1 - betas
    abprev = ab[t - 1] if t > 0 else 1.0
    mean = np.sqrt(abprev) * betas[t] / (1 - ab[t]) * x0 + np.sqrt(al[t]) * (1 - abprev) / (1 - ab[t]) * xt
    var = (1 - abprev) / (1 - ab[t]) * betas[t]
    return mean, var

def sample(n, betas, eps_fn, rng):
    al = 1 - betas
    ab = alpha_bars(betas)
    x = rng.normal(size=n)
    for t in range(len(betas) - 1, -1, -1):
        e = eps_fn(x, t)
        x = (x - betas[t] / np.sqrt(1 - ab[t]) * e) / np.sqrt(al[t])
        if t > 0:
            x = x + np.sqrt(betas[t]) * rng.normal(size=n)
    return x
