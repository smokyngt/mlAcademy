import numpy as np

def sigmoid(s):
    return 1 / (1 + np.exp(-s))

def optimal_discriminator(pd, pg):
    return pd / (pd + pg)

def kl(p, q):
    m = p > 0
    return np.sum(p[m] * np.log(p[m] / q[m]))

def js(p, q):
    m = 0.5 * (p + q)
    return 0.5 * kl(p, m) + 0.5 * kl(q, m)

def value(pd, pg, D):
    return np.sum(pd * np.log(D)) + np.sum(pg * np.log(1 - D))

def w1_samples(a, b):
    return np.mean(np.abs(np.sort(a) - np.sort(b)))

def grad_saturating(s):
    return -sigmoid(s)

def grad_nonsaturating(s):
    return -(1 - sigmoid(s))

def d_loss(a, b, xr, xf):
    return -np.mean(np.log(sigmoid(a * xr + b))) - np.mean(np.log(1 - sigmoid(a * xf + b)))

def d_grads(a, b, xr, xf):
    sr = sigmoid(a * xr + b)
    sf = sigmoid(a * xf + b)
    ga = np.mean(-(1 - sr) * xr) + np.mean(sf * xf)
    gb = np.mean(-(1 - sr)) + np.mean(sf)
    return ga, gb

def g_loss(theta, a, b, z):
    return -np.mean(np.log(sigmoid(a * (theta + z) + b)))

def g_grad(theta, a, b, z):
    return np.mean(-(1 - sigmoid(a * (theta + z) + b)) * a)

def train_toy(steps=300, lr=0.1, batch=128, seed=0):
    rng = np.random.default_rng(seed)
    a = b = theta = 0.0
    for _ in range(steps):
        xr = rng.normal(3, 1, batch)
        z = rng.normal(0, 1, batch)
        ga, gb = d_grads(a, b, xr, theta + z)
        a -= lr * ga
        b -= lr * gb
        theta -= lr * g_grad(theta, a, b, z)
    return a, b, theta
