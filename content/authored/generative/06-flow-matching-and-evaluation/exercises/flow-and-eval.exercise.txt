import numpy as np

def fm_interp(x0, x1, t):
    return (1 - t) * x0 + t * x1

def fm_target(x0, x1):
    return x1 - x0

def optimal_velocity(x, t, m, s):
    var = (1 - t) ** 2 * s ** 2 + t ** 2
    mean = (1 - t) * m
    cov = t - (1 - t) * s ** 2
    return -m + cov / var * (x - mean)

def euler_sample(n, v_fn, steps, rng):
    x = rng.normal(size=n)
    dt = 1.0 / steps
    for k in range(steps):
        t = 1 - k * dt
        x = x - dt * v_fn(x, t)
    return x

def sqrtm_psd(S):
    w, V = np.linalg.eigh((S + S.T) / 2)
    return (V * np.sqrt(np.clip(w, 0, None))) @ V.T

def frechet_distance(mu1, S1, mu2, S2):
    r = sqrtm_psd(S1)
    M = r @ S2 @ r
    tr = np.sqrt(np.clip(np.linalg.eigvalsh((M + M.T) / 2), 0, None)).sum()
    return float(np.sum((mu1 - mu2) ** 2) + np.trace(S1) + np.trace(S2) - 2 * tr)

def fid_from_features(a, b):
    return frechet_distance(a.mean(0), np.cov(a, rowvar=False), b.mean(0), np.cov(b, rowvar=False))

def knn_radii(X, k):
    D = np.linalg.norm(X[:, None] - X[None], axis=-1)
    return np.sort(D, axis=1)[:, k]

def precision_recall(real, fake, k=3):
    rr = knn_radii(real, k)
    rf = knn_radii(fake, k)
    D = np.linalg.norm(fake[:, None] - real[None], axis=-1)
    prec = np.mean((D <= rr[None]).any(1))
    rec = np.mean((D.T <= rf[None]).any(1))
    return prec, rec

def elo_expected(ra, rb):
    return 1.0 / (1.0 + 10 ** ((rb - ra) / 400.0))

def elo_update(ra, rb, score_a, k=32):
    e = elo_expected(ra, rb)
    return ra + k * (score_a - e), rb + k * ((1 - score_a) - (1 - e))
