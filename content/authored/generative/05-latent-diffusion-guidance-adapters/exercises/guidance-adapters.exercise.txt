import numpy as np

def linear_betas(T, b0=1e-4, b1=0.02):
    return np.linspace(b0, b1, T)

def alpha_bars(betas):
    return np.cumprod(1 - betas)

def x0_hat(xt, t, abar, m, s):
    a = abar[t]
    return m + np.sqrt(a) * s ** 2 / (a * s ** 2 + 1 - a) * (xt - np.sqrt(a) * m)

def eps_cond(xt, t, abar, m, s):
    return (xt - np.sqrt(abar[t]) * x0_hat(xt, t, abar, m, s)) / np.sqrt(1 - abar[t])

def npdf(x, m, s):
    return np.exp(-(x - m) ** 2 / (2 * s ** 2)) / np.sqrt(2 * np.pi * s ** 2)

def eps_uncond(xt, t, abar, ms, s):
    a = abar[t]
    sd = np.sqrt(a * s ** 2 + 1 - a)
    lik = np.stack([npdf(xt, np.sqrt(a) * m, sd) for m in ms])
    r = lik / lik.sum(0)
    xh = sum(r[i] * x0_hat(xt, t, abar, m, s) for i, m in enumerate(ms))
    return (xt - np.sqrt(a) * xh) / np.sqrt(1 - a)

def cfg(eps_u, eps_c, w):
    return eps_u + w * (eps_c - eps_u)

def sample(n, betas, eps_fn, rng):
    al = 1 - betas
    ab = alpha_bars(betas)
    x = rng.normal(size=n)
    for t in range(len(betas) - 1, -1, -1):
        e = eps_fn(x, t)
        x = (x - betas[t] / np.sqrt(1 - ab[t]) * e) / np.sqrt(al[t])
        if t > 0:
            x = x + np.sqrt(betas[t]) * rng.normal(size=n)
    return x

def cross_attention(F, C, Wq, Wk, Wv):
    Q, K, V = F @ Wq, C @ Wk, C @ Wv
    S = Q @ K.T / np.sqrt(K.shape[1])
    P = np.exp(S - S.max(1, keepdims=True))
    P = P / P.sum(1, keepdims=True)
    return P @ V, P

def lora_delta(A, B, alpha=1.0):
    r = A.shape[0]
    return (alpha / r) * (B @ A)

def lora_params(d_out, d_in, r):
    return r * (d_out + d_in)
