Implement the pieces of a tiny VAE with a linear decoder and verify the gradients of the reparameterised sample.

- `kl_to_std_normal(mu, logvar)`: $\tfrac12\sum(\mu^2+e^{\text{logvar}}-\text{logvar}-1)$ over the last axis.
- `kl_gauss_1d(m1, s1, m2, s2)`: KL between two 1-D Gaussians (standard deviations, not variances).

import numpy as np

def kl_to_std_normal(mu, logvar):
    return 0.5 * np.sum(np.exp(logvar) + mu ** 2 - logvar - 1, axis=-1)

def kl_gauss_1d(m1, s1, m2, s2):
    return np.log(s2 / s1) + (s1 ** 2 + (m1 - m2) ** 2) / (2 * s2 ** 2) - 0.5

def reparameterize(mu, logvar, eps):
    return mu + np.exp(0.5 * logvar) * eps

def vae_loss(x, mu, logvar, eps, W, b, beta=1.0):
    z = reparameterize(mu, logvar, eps)
    xh = W @ z + b
    rec = np.sum((x - xh) ** 2)
    kl = kl_to_std_normal(mu, logvar)
    return rec + beta * kl, rec, kl

def vae_grads(x, mu, logvar, eps, W, b, beta=1.0):
    z = reparameterize(mu, logvar, eps)
    xh = W @ z + b
    dz = 2 * W.T @ (xh - x)
    dmu = dz + beta * mu
    dlv = dz * 0.5 * np.exp(0.5 * logvar) * eps + beta * 0.5 * (np.exp(logvar) - 1)
    return dmu, dlv
