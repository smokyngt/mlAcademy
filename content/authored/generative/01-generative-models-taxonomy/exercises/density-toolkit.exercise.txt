import numpy as np

def gauss_logpdf(x, mu, sigma):
    return -0.5 * np.log(2 * np.pi * sigma ** 2) - (x - mu) ** 2 / (2 * sigma ** 2)

def gmm_logpdf(x, w, mu, sigma):
    x = np.asarray(x, float)[:, None]
    lp = np.log(np.asarray(w)) + gauss_logpdf(x, np.asarray(mu), np.asarray(sigma))
    m = lp.max(1, keepdims=True)
    return (m + np.log(np.exp(lp - m).sum(1, keepdims=True)))[:, 0]

def gmm_sample(n, w, mu, sigma, rng):
    k = rng.choice(len(w), size=n, p=w)
    return rng.normal(np.asarray(mu)[k], np.asarray(sigma)[k])

def mle_gauss(x):
    return x.mean(), x.std()

def avg_nll(logp):
    return -np.mean(logp)

def generative_classify(x, priors, mus, sigmas):
    x = np.asarray(x, float)[:, None]
    lp = np.log(np.asarray(priors)) + gauss_logpdf(x, np.asarray(mus), np.asarray(sigmas))
    lp = lp - lp.max(1, keepdims=True)
    p = np.exp(lp)
    return p / p.sum(1, keepdims=True)
