import numpy as np

def power_spectrogram(x, n_fft, hop):
    n = 1 + (len(x) - n_fft) // hop
    idx = np.arange(n_fft)[None, :] + hop * np.arange(n)[:, None]
    w = 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(n_fft) / n_fft)
    return np.abs(np.fft.rfft(x[idx] * w, axis=1)) ** 2

def hz_to_mel(f):
    return 2595.0 * np.log10(1.0 + np.asarray(f, float) / 700.0)

def mel_to_hz(m):
    return 700.0 * (10.0 ** (np.asarray(m, float) / 2595.0) - 1.0)

def mel_filterbank(n_mels, n_fft, sr, fmin=0.0, fmax=None):
    fmax = sr / 2 if fmax is None else fmax
    pts = mel_to_hz(np.linspace(hz_to_mel(fmin), hz_to_mel(fmax), n_mels + 2))
    freqs = np.arange(n_fft // 2 + 1) * sr / n_fft
    fb = np.zeros((n_mels, len(freqs)))
    for j in range(n_mels):
        lo, mid, hi = pts[j], pts[j + 1], pts[j + 2]
        fb[j] = np.maximum(0, np.minimum((freqs - lo) / (mid - lo), (hi - freqs) / (hi - mid)))
    return fb

def dct_matrix(n_in, n_out):
    k = np.arange(n_out)[:, None]
    m = np.arange(n_in)[None, :]
    D = np.sqrt(2.0 / n_in) * np.cos(np.pi * k * (2 * m + 1) / (2 * n_in))
    D[0] /= np.sqrt(2.0)
    return D

def mfcc_from_power(P, sr, n_fft, n_mels=26, n_mfcc=13):
    logmel = np.log(P @ mel_filterbank(n_mels, n_fft, sr).T + 1e-10)
    return logmel @ dct_matrix(n_mels, n_mfcc).T

def deltas(c, N=2):
    T = len(c)
    pad = np.pad(c, ((N, N), (0, 0)), mode="edge")
    out = np.zeros_like(c, dtype=float)
    for n in range(1, N + 1):
        out += n * (pad[N + n:N + n + T] - pad[N - n:N - n + T])
    return out / (2 * sum(n * n for n in range(1, N + 1)))
