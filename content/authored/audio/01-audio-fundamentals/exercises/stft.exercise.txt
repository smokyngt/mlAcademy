Implement the first stage of every audio pipeline with NumPy only:

- `frame_signal(x, frame_len, hop)` -> array of shape `(n_frames, frame_len)` with `n_frames = 1 + (len(x) - frame_len) // hop` (no padding; drop incomplete frames).

- `alias_freq(f, sr)` -> the frequency at which a tone of `f` Hz appears after sampling at `sr` Hz.
- `bin_freq(k, n_fft, sr)` -> frequency in Hz of DFT bin `k`.

import numpy as np

def frame_signal(x, frame_len, hop):
    n = 1 + (len(x) - frame_len) // hop
    idx = np.arange(frame_len)[None, :] + hop * np.arange(n)[:, None]
    return x[idx]

def hann(N):
    n = np.arange(N)
    return 0.5 - 0.5 * np.cos(2 * np.pi * n / N)

def stft(x, n_fft, hop):
    frames = frame_signal(x, n_fft, hop) * hann(n_fft)
    return np.fft.rfft(frames, axis=1)

def quantize(x, bits):
    L = 2 ** bits
    step = 2.0 / L
    q = np.clip(np.floor(x / step), -L // 2, L // 2 - 1)
    return (q + 0.5) * step

def alias_freq(f, sr):
    r = f % sr
    return min(r, sr - r)

def bin_freq(k, n_fft, sr):
    return k * sr / n_fft
