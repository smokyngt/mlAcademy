Implement the augmentation and bookkeeping tools of an audio CNN pipeline:

- `time_mask(spec, t, t0, fill=0.0)` and `freq_mask(spec, f, f0, fill=0.0)`: return a **copy** of the `(T, F)` spectrogram with `t` rows starting at `t0` (or `f` columns starting at `f0`) replaced by `fill`.
- `spec_augment(spec, rng, n_time=2, T_max=20, n_freq=2, F_max=10, fill=0.0)`: apply `n_freq` frequency masks, then `n_time` time masks; each width is drawn uniformly from `0..max_width` (use `rng.integers(0, max_width + 1)`) and each start uniformly so the mask fits inside the array.

import numpy as np

def time_mask(spec, t, t0, fill=0.0):
    out = spec.copy()
    out[t0:t0 + t, :] = fill
    return out

def freq_mask(spec, f, f0, fill=0.0):
    out = spec.copy()
    out[:, f0:f0 + f] = fill
    return out

def spec_augment(spec, rng, n_time=2, T_max=20, n_freq=2, F_max=10, fill=0.0):
    out = spec.copy()
    T, F = spec.shape
    for _ in range(n_freq):
        f = int(rng.integers(0, F_max + 1))
        f0 = int(rng.integers(0, F - f + 1))
        out = freq_mask(out, f, f0, fill)
    for _ in range(n_time):
        t = int(rng.integers(0, T_max + 1))
        t0 = int(rng.integers(0, T - t + 1))
        out = time_mask(out, t, t0, fill)
    return out

def mixup(x1, y1, x2, y2, lam):
    return lam * x1 + (1 - lam) * x2, lam * y1 + (1 - lam) * y2

def conv_out_len(i, k, s=1, p=0):
    return (i + 2 * p - k) // s + 1

def conv_params(c_in, c_out, k):
    return c_in * c_out * k * k + c_out

def linear_params(d_in, d_out):
    return d_in * d_out + d_out

def audio_cnn_audit(T, F, n_classes):
    shapes = []
    t, f = T, F
    shapes.append((32, conv_out_len(t, 3, 1, 1), conv_out_len(f, 3, 1, 1)))
    t, f = conv_out_len(t, 2, 2), conv_out_len(f, 2, 2)
    shapes.append((32, t, f))
    shapes.append((64, conv_out_len(t, 3, 1, 1), conv_out_len(f, 3, 1, 1)))
    t, f = conv_out_len(t, 2, 2), conv_out_len(f, 2, 2)
    shapes.append((64, t, f))
    shapes.append((128, conv_out_len(t, 3, 1, 1), conv_out_len(f, 3, 1, 1)))
    n = conv_params(1, 32, 3) + conv_params(32, 64, 3) + conv_params(64, 128, 3) + linear_params(128, n_classes)
    return shapes, n
