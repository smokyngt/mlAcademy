import numpy as np

def make_clip(label, rng, sr=8000, n=2000):
    t = np.arange(n) / sr
    if label == 0:
        x = np.sin(2 * np.pi * 440 * t)
    elif label == 1:
        x = np.sin(2 * np.pi * 520 * t)
    elif label == 2:
        x = np.sin(2 * np.pi * (300 * t + 2000 * t ** 2))
    else:
        x = 0.7 * rng.normal(size=n)
    return x + 3.0 * rng.normal(size=n)

def clip_features(x, frame=200, hop=100, bands=16):
    n = 1 + (len(x) - frame) // hop
    idx = np.arange(frame)[None, :] + hop * np.arange(n)[:, None]
    fr = x[idx] * (0.5 - 0.5 * np.cos(2 * np.pi * np.arange(frame) / frame))
    P = np.abs(np.fft.rfft(fr, axis=1)) ** 2
    P = P[:, :bands * 6].reshape(n, bands, 6).sum(2)
    L = np.log(P + 1e-8)
    return np.concatenate([L.mean(0), L.std(0)])

def knn_predict(Xtr, ytr, Xte, k=3):
    a = Xtr / np.linalg.norm(Xtr, axis=1, keepdims=True)
    b = Xte / np.linalg.norm(Xte, axis=1, keepdims=True)
    sim = b @ a.T
    out = []
    for row in sim:
        nn = np.argsort(-row)[:k]
        out.append(np.bincount(ytr[nn]).argmax())
    return np.array(out)

def confusion_matrix(y, p, K):
    C = np.zeros((K, K), dtype=int)
    for a, b in zip(y, p):
        C[a, b] += 1
    return C

def per_class_recall(C):
    return np.diag(C) / np.maximum(C.sum(1), 1)

def per_class_precision(C):
    return np.diag(C) / np.maximum(C.sum(0), 1)

def macro_f1(C):
    p = per_class_precision(C)
    r = per_class_recall(C)
    f1 = np.where(p + r > 0, 2 * p * r / np.maximum(p + r, 1e-12), 0.0)
    return float(f1.mean())

def accuracy(C):
    return float(np.trace(C) / C.sum())
