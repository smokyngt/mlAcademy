#@@ prompt
Rebuild a waveform from a magnitude spectrogram.

- `stft(x, n_fft=256, hop=64)` -> complex `(frames, n_fft//2+1)`; the signal is reflect-padded by `n_fft//2` on both sides (centred frames) and a **periodic** Hann window is used.
- `istft(S, n_fft=256, hop=64, length=None)` -> the least-squares inverse: weighted overlap-add (window the inverse-FFT frames again, sum, divide by the summed squared windows), trimmed to the original length (`n_fft//2` samples removed from each end).
- `spectral_convergence(S_hat, mag)` = `||abs(S_hat) - mag||_F / ||mag||_F`.
- `griffin_lim(mag, n_fft=256, hop=64, n_iter=100, seed=0, length=None)`: start from random phase `mag * exp(2j*pi*rng.random(mag.shape))` with `rng = np.random.default_rng(seed)`, iterate (inverse STFT, STFT, keep the phase, restore the magnitude) and return the waveform of the final spectrogram.
- `invert_mel(M, fb)` -> estimate of the linear magnitude from a mel **power** spectrogram `M` of shape `(T, n_mels)`: `sqrt(max(0, M @ pinv(fb).T))`.

import numpy as np

def hann(N):
    return 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(N) / N)

def hz_to_mel(f):
    return 2595.0 * np.log10(1.0 + np.asarray(f, float) / 700.0)

def mel_to_hz(m):
    return 700.0 * (10.0 ** (np.asarray(m, float) / 2595.0) - 1.0)

def mel_filterbank(n_mels, n_fft, sr):
    pts = mel_to_hz(np.linspace(0.0, hz_to_mel(sr / 2), n_mels + 2))
    freqs = np.arange(n_fft // 2 + 1) * sr / n_fft
    fb = np.zeros((n_mels, len(freqs)))
    for j in range(n_mels):
        lo, mid, hi = pts[j], pts[j + 1], pts[j + 2]
        fb[j] = np.maximum(0, np.minimum((freqs - lo) / (mid - lo), (hi - freqs) / (hi - mid)))
    return fb

def stft(x, n_fft=256, hop=64):
    xp = np.pad(x, (n_fft // 2, n_fft // 2), mode="reflect")
    n = 1 + (len(xp) - n_fft) // hop
    idx = np.arange(n_fft)[None, :] + hop * np.arange(n)[:, None]
    return np.fft.rfft(xp[idx] * hann(n_fft), axis=1)

def istft(S, n_fft=256, hop=64, length=None):
    n = S.shape[0]
    fr = np.fft.irfft(S, n=n_fft, axis=1) * hann(n_fft)
    L = n_fft + hop * (n - 1)
    y = np.zeros(L)
    wsum = np.zeros(L)
    w2 = hann(n_fft) ** 2
    for i in range(n):
        y[i * hop:i * hop + n_fft] += fr[i]
        wsum[i * hop:i * hop + n_fft] += w2
    y = y / np.maximum(wsum, 1e-8)
    y = y[n_fft // 2:L - n_fft // 2]
    return y if length is None else y[:length]

def spectral_convergence(S_hat, mag):
    return float(np.linalg.norm(np.abs(S_hat) - mag) / np.linalg.norm(mag))

def griffin_lim(mag, n_fft=256, hop=64, n_iter=100, seed=0, length=None):
    rng = np.random.default_rng(seed)
    S = mag * np.exp(2j * np.pi * rng.random(mag.shape))
    for _ in range(n_iter):
        x = istft(S, n_fft, hop, length)
        S = mag * np.exp(1j * np.angle(stft(x, n_fft, hop)))
    return istft(S, n_fft, hop, length)

def invert_mel(M, fb):
    return np.sqrt(np.maximum(0.0, M @ np.linalg.pinv(fb).T))
