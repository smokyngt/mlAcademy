Small, checkable tools from the TTS and voice-cloning toolbox:

- `length_regulator(H, durations)` -> expand phoneme hidden states `H` of shape `(U, d)` to `(sum(durations), d)`.
- `mel_frames_to_samples(T, hop)` and `frame_seconds(hop, sr)`: number of waveform samples produced by `T` mel frames and the duration of one frame.
- `upsampling_ok(rates, hop)` -> True if the product of the vocoder's upsampling factors equals the hop length.
- `codec_bitrate(frame_rate, n_codebooks, codebook_size)` -> bits per second of a residual-VQ codec.
- `mos_ci(scores, z=1.96)` -> `(mean, lower, upper)` normal-approximation confidence interval (sample standard deviation with `ddof=1`).
- `secs(e1, e2)` -> cosine similarity between two speaker embeddings.
- `consent_ok(record, speaker_id, now)` -> True only if `record` is a dict with a `speaker_id` equal to the one requested, a boolean `signed` that is True, and `expires` (a number) greater than `now`.

import numpy as np

def length_regulator(H, durations):
    return np.repeat(H, durations, axis=0)

def mel_frames_to_samples(T, hop):
    return T * hop

def frame_seconds(hop, sr):
    return hop / sr

def upsampling_ok(rates, hop):
    return int(np.prod(rates)) == hop

def codec_bitrate(frame_rate, n_codebooks, codebook_size):
    return frame_rate * n_codebooks * np.log2(codebook_size)

def mos_ci(scores, z=1.96):
    s = np.asarray(scores, float)
    m = s.mean()
    se = s.std(ddof=1) / np.sqrt(len(s))
    return float(m), float(m - z * se), float(m + z * se)

def secs(e1, e2):
    a, b = np.asarray(e1, float), np.asarray(e2, float)
    return float(a @ b / (np.linalg.norm(a) * np.linalg.norm(b)))

def consent_ok(record, speaker_id, now):
    if not isinstance(record, dict):
        return False
    if record.get("speaker_id") != speaker_id:
        return False
    if record.get("signed") is not True:
        return False
    exp = record.get("expires")
    if not isinstance(exp, (int, float)):
        return False
    return exp > now
