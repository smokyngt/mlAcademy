#@@ prompt
Implement CTC for a single utterance. `logp` is a `(T, V)` array of per-frame **log**-probabilities (each row is a log-softmax; symbol 0 is the blank).

- `collapse(path, blank=0)` -> list: merge consecutive repeats, then drop blanks.
- `ctc_forward(logp, target, blank=0)` -> `log P(target | x)` (float; `-inf` when impossible) using the forward algorithm in log space. `target` is a list of non-blank symbols (it may be empty).
- `ctc_greedy(logp, blank=0)` -> best-path decoding (argmax per frame, then collapse).
- `ctc_prefix_beam_search(logp, beam=8, blank=0)` -> `(label_list, log_prob)` of the best prefix found. With a beam wide enough to keep everything the result must be the exact most probable transcript, and `log_prob` must equal `ctc_forward` of that transcript.

def brute(logp, target, blank=0):
    T, V = logp.shape
    total = 0.0
    for path in itertools.product(range(V), repeat=T):
        if collapse(path, blank) == list(target):
            total += np.exp(sum(logp[t, v] for t, v in enumerate(path)))
    return total

import numpy as np
from collections import defaultdict

NEG = -np.inf

def collapse(path, blank=0):
    out, prev = [], None
    for p in path:
        if p != prev and p != blank:
            out.append(p)
        prev = p
    return out

def ctc_forward(logp, target, blank=0):
    T = logp.shape[0]
    ext = [blank]
    for c in target:
        ext += [c, blank]
    S = len(ext)
    alpha = np.full((T, S), NEG)
    alpha[0, 0] = logp[0, blank]
    if S > 1:
        alpha[0, 1] = logp[0, ext[1]]
    for t in range(1, T):
        for s in range(S):
            terms = [alpha[t - 1, s]]
            if s >= 1:
                terms.append(alpha[t - 1, s - 1])
            if s >= 2 and ext[s] != blank and ext[s] != ext[s - 2]:
                terms.append(alpha[t - 1, s - 2])
            alpha[t, s] = np.logaddexp.reduce(terms) + logp[t, ext[s]]
    end = alpha[T - 1, S - 1]
    if S > 1:
        end = np.logaddexp(end, alpha[T - 1, S - 2])
    return float(end)

def ctc_greedy(logp, blank=0):
    return collapse(list(logp.argmax(axis=1)), blank)

def ctc_prefix_beam_search(logp, beam=8, blank=0):
    T, V = logp.shape
    beams = {(): (0.0, NEG)}
    for t in range(T):
        new = defaultdict(lambda: [NEG, NEG])
        for pre, (pb, pnb) in beams.items():
            tot = np.logaddexp(pb, pnb)
            e = new[pre]
            e[0] = np.logaddexp(e[0], tot + logp[t, blank])
            for c in range(V):
                if c == blank:
                    continue
                p = logp[t, c]
                if pre and pre[-1] == c:
                    e = new[pre]
                    e[1] = np.logaddexp(e[1], pnb + p)
                    n = new[pre + (c,)]
                    n[1] = np.logaddexp(n[1], pb + p)
                else:
                    n = new[pre + (c,)]
                    n[1] = np.logaddexp(n[1], tot + p)
        items = sorted(new.items(), key=lambda kv: -np.logaddexp(kv[1][0], kv[1][1]))[:beam]
        beams = {k: (v[0], v[1]) for k, v in items}
    best = max(beams.items(), key=lambda kv: np.logaddexp(kv[1][0], kv[1][1]))
    return [int(c) for c in best[0]], float(np.logaddexp(best[1][0], best[1][1]))
