Implement the standard ASR metrics with NumPy/plain Python:

- `edit_distance_table(ref, hyp)` -> integer array `(len(ref)+1, len(hyp)+1)` (works on lists of words or strings of characters).
- `edit_ops(ref, hyp)` -> `(S, D, I)` found by back-tracing the table; prefer the diagonal move first, then deletion, then insertion. Only the total `S+D+I` is unique; the tests check the split only on unambiguous cases.

- `wer(ref, hyp)` -> `(S + D + I) / N` on normalised word lists (`N` = number of reference words, at least 1).

import numpy as np

def edit_distance_table(ref, hyp):
    n, m = len(ref), len(hyp)
    d = np.zeros((n + 1, m + 1), dtype=int)
    d[:, 0] = np.arange(n + 1)
    d[0, :] = np.arange(m + 1)
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            d[i, j] = min(d[i - 1, j] + 1, d[i, j - 1] + 1, d[i - 1, j - 1] + (ref[i - 1] != hyp[j - 1]))
    return d

def edit_ops(ref, hyp):
    d = edit_distance_table(ref, hyp)
    i, j = len(ref), len(hyp)
    S = D = I = 0
    while i > 0 or j > 0:
        if i > 0 and j > 0 and d[i, j] == d[i - 1, j - 1] + (ref[i - 1] != hyp[j - 1]):
            S += int(ref[i - 1] != hyp[j - 1])
            i -= 1
            j -= 1
        elif i > 0 and d[i, j] == d[i - 1, j] + 1:
            D += 1
            i -= 1
        else:
            I += 1
            j -= 1
    return S, D, I

def normalize(text):
    keep = "".join(ch if (ch.isalnum() or ch.isspace()) else " " for ch in text.lower())
    return keep.split()

def wer(ref, hyp):
    r, h = normalize(ref), normalize(hyp)
    return float(edit_distance_table(r, h)[len(r), len(h)] / max(1, len(r)))

def cer(ref, hyp):
    r = "".join(normalize(ref))
    h = "".join(normalize(hyp))
    return float(edit_distance_table(r, h)[len(r), len(h)] / max(1, len(r)))

def rnnt_lattice_floats(T, U, V1):
    return T * (U + 1) * V1

def ctc_matrix_floats(T, V1):
    return T * V1
