import numpy as np

def gradient_descent(grad, x0, lr, steps):
    x = np.array(x0, dtype=float)
    path = [x.copy()]
    for _ in range(steps):
        x = x - lr * grad(x)
        path.append(x.copy())
    return np.array(path)

def fit_linear(X, y, lr, steps):
    n, d = X.shape
    grad = lambda w: 2.0 / n * X.T @ (X @ w - y)
    return gradient_descent(grad, np.zeros(d), lr, steps)[-1]

def stable_lr(lam_max):
    return 1.0 / lam_max
