import numpy as np
import torch
import torch.nn.functional as F

def numpy_loss_and_grads(params, X, y):
    W1, b1, W2, b2 = params
    B = len(y)
    Z1 = X @ W1 + b1; A1 = np.maximum(0, Z1); Z2 = A1 @ W2 + b2
    Z = Z2 - Z2.max(axis=1, keepdims=True)
    P = np.exp(Z); P /= P.sum(axis=1, keepdims=True)
    loss = float(-np.mean(np.log(P[np.arange(B), y])))
    dZ2 = P.copy(); dZ2[np.arange(B), y] -= 1; dZ2 /= B
    dZ1 = (dZ2 @ W2.T) * (Z1 > 0)
    return loss, [X.T @ dZ1, dZ1.sum(0), A1.T @ dZ2, dZ2.sum(0)]

def torch_loss_and_grads(params, X, y):
    ts = [torch.tensor(p, dtype=torch.float64, requires_grad=True) for p in params]
    W1, b1, W2, b2 = ts
    Xt = torch.tensor(X, dtype=torch.float64)
    yt = torch.tensor(y, dtype=torch.long)
    logits = torch.relu(Xt @ W1 + b1) @ W2 + b2
    loss = F.cross_entropy(logits, yt)
    loss.backward()
    return float(loss.item()), [t.grad.numpy() for t in ts]

def accumulated_after_two_backwards(w0):
    w = torch.tensor(w0, dtype=torch.float64, requires_grad=True)
    loss = (w ** 2).sum()
    loss.backward(retain_graph=True)
    loss.backward()
    return w.grad.numpy()
