import numpy as np

def loss_and_grads(params, X, y):
    W1, b1, W2, b2 = params
    B = len(y)
    Z1 = X @ W1 + b1
    A1 = np.maximum(0, Z1)
    Z2 = A1 @ W2 + b2
    Z = Z2 - Z2.max(axis=1, keepdims=True)
    P = np.exp(Z); P /= P.sum(axis=1, keepdims=True)
    loss = float(-np.mean(np.log(P[np.arange(B), y])))
    dZ2 = P.copy(); dZ2[np.arange(B), y] -= 1; dZ2 /= B
    dW2 = A1.T @ dZ2
    db2 = dZ2.sum(axis=0)
    dA1 = dZ2 @ W2.T
    dZ1 = dA1 * (Z1 > 0)
    dW1 = X.T @ dZ1
    db1 = dZ1.sum(axis=0)
    return loss, (dW1, db1, dW2, db2)

def grad_check(params, X, y, h=1e-5):
    _, grads = loss_and_grads(params, X, y)
    worst = 0.0
    for k, p in enumerate(params):
        for idx in np.ndindex(*p.shape):
            plus = [q.copy() for q in params]; minus = [q.copy() for q in params]
            plus[k][idx] += h; minus[k][idx] -= h
            num = (loss_and_grads(tuple(plus), X, y)[0] - loss_and_grads(tuple(minus), X, y)[0]) / (2 * h)
            ana = grads[k][idx]
            worst = max(worst, abs(ana - num) / max(1e-8, abs(ana) + abs(num)))
    return worst

rng = np.random.default_rng(0)

print("gradient check error:", grad_check(params, X, y))
