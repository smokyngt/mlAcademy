import numpy as np

def make_moons(n=200, noise=0.1, seed=0):
    rng = np.random.default_rng(seed)
    n1 = n // 2; n2 = n - n1
    t1 = np.pi * rng.random(n1); t2 = np.pi * rng.random(n2)
    X = np.vstack([np.c_[np.cos(t1), np.sin(t1)], np.c_[1 - np.cos(t2), 0.5 - np.sin(t2)]])
    X = X + noise * rng.normal(size=X.shape)
    y = np.r_[np.zeros(n1), np.ones(n2)].astype(int)
    return X, y

def forward_backward(params, X, y):
    W1, b1, W2, b2 = params
    B = len(y)
    Z1 = X @ W1 + b1; A1 = np.maximum(0, Z1); Z2 = A1 @ W2 + b2
    Z = Z2 - Z2.max(axis=1, keepdims=True)
    P = np.exp(Z); P /= P.sum(axis=1, keepdims=True)
    loss = float(-np.mean(np.log(P[np.arange(B), y] + 1e-300)))
    dZ2 = P.copy(); dZ2[np.arange(B), y] -= 1; dZ2 /= B
    dW2 = A1.T @ dZ2; db2 = dZ2.sum(0)
    dZ1 = (dZ2 @ W2.T) * (Z1 > 0)
    return loss, (X.T @ dZ1, dZ1.sum(0), dW2, db2)

def init_params(n_in, hidden, n_out, seed=0):
    rng = np.random.default_rng(seed)
    W1 = rng.normal(0, np.sqrt(2.0 / n_in), (n_in, hidden))
    W2 = rng.normal(0, np.sqrt(2.0 / hidden), (hidden, n_out))
    return (W1, np.zeros(hidden), W2, np.zeros(n_out))

def sgd_step(params, grads, lr):
    return tuple(p - lr * g for p, g in zip(params, grads))

def train(X, y, hidden=16, lr=0.5, epochs=400, seed=0):
    params = init_params(X.shape[1], hidden, int(y.max()) + 1, seed)
    history = []
    for _ in range(epochs):
        loss, grads = forward_backward(params, X, y)
        params = sgd_step(params, grads, lr)
        history.append(loss)
    return params, history

def accuracy(params, X, y):
    W1, b1, W2, b2 = params
    logits = np.maximum(0, X @ W1 + b1) @ W2 + b2
    return float(np.mean(logits.argmax(axis=1) == y))

X, y = make_moons()
params, history = train(X, y)
print("final loss", history[-1], "accuracy", accuracy(params, X, y))
