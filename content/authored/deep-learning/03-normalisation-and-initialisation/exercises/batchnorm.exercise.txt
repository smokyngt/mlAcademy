import numpy as np

def numgrad(f, x, h=1e-6):
    g = np.zeros_like(x)
    it = np.nditer(x, flags=["multi_index"])
    for _ in it:
        i = it.multi_index
        old = x[i]
        x[i] = old + h; a = f()
        x[i] = old - h; b = f()
        x[i] = old
        g[i] = (a - b) / (2 * h)
    return g

def bn_forward(x, gamma, beta, eps=1e-5):
    mu = x.mean(0)
    var = x.var(0)
    inv = 1.0 / np.sqrt(var + eps)
    xh = (x - mu) * inv
    return gamma * xh + beta, (xh, inv, gamma)

def bn_backward(dy, cache):
    xh, inv, gamma = cache
    N = dy.shape[0]
    dxh = dy * gamma
    dx = (inv / N) * (N * dxh - dxh.sum(0) - xh * (dxh * xh).sum(0))
    return dx, (dy * xh).sum(0), dy.sum(0)

def bn_eval(x, gamma, beta, run_mean, run_var, eps=1e-5):
    return gamma * (x - run_mean) / np.sqrt(run_var + eps) + beta

def update_running(run, batch_stat, momentum=0.1):
    return (1 - momentum) * run + momentum * batch_stat

def layer_norm(x, gamma, beta, eps=1e-5):
    mu = x.mean(1, keepdims=True)
    var = x.var(1, keepdims=True)
    return gamma * (x - mu) / np.sqrt(var + eps) + beta

def he_init(fan_in, fan_out, rng):
    return rng.normal(0.0, np.sqrt(2.0 / fan_in), (fan_in, fan_out))

x = np.array([[1.0], [2.0], [3.0], [4.0]])
y, cache = bn_forward(x, np.ones(1), np.zeros(1), eps=0.0)
print(y.ravel())
