An ill-conditioned quadratic `f(x) = 0.5 x^T A x - b^T x` (condition number 100, valley rotated 30 degrees) is provided together with plain `SGD` and a runner `steps_to(opt)` that counts steps until `f(x) - f(x*) < 1e-6`.

Write:

- `Momentum(lr, mu=0.9, nesterov=False)` with `step(p, g)` returning the new parameters (never modify `p` in place). Velocity: `v = mu*v + g`. Nesterov uses the direction `g + mu*v`.
- `Adam(lr, b1=0.9, b2=0.999, eps=1e-8, wd=0.0)` with bias correction and decoupled weight decay (AdamW): `p - lr*mhat/(sqrt(vhat)+eps) - lr*wd*p`.
- `warmup_cosine(t, peak, warmup, total, lr_min=0.0)` returning the learning rate at step `t`.

Then compare: with the textbook values `lr=4/121, mu=(9/11)**2` Momentum should need far fewer steps than SGD at its best step `2/101`. Also check that Adam's very first step moves every coordinate by about `lr`, whatever the gradient scale.

import numpy as np
import math

th = np.pi / 6
R = np.array([[np.cos(th), -np.sin(th)], [np.sin(th), np.cos(th)]])
A = R @ np.diag([1.0, 100.0]) @ R.T
b = np.array([1.0, 1.0])
x_star = np.linalg.solve(A, b)
f = lambda x: 0.5 * x @ A @ x - b @ x
grad = lambda x: A @ x - b

class SGD:
    def __init__(self, lr):
        self.lr = lr
    def step(self, p, g):
        return p - self.lr * g

def steps_to(opt, tol=1e-6, max_steps=3000):
    x = np.array([-3.0, -1.0])
    for i in range(1, max_steps + 1):
        x = opt.step(x, grad(x))
        if not np.all(np.isfinite(x)):
            return None
        if f(x) - f(x_star) < tol:
            return i
    return None

class Momentum:
    def __init__(self, lr, mu=0.9, nesterov=False):
        self.lr, self.mu, self.nes, self.v = lr, mu, nesterov, None
    def step(self, p, g):
        if self.v is None:
            self.v = np.zeros_like(p)
        self.v = self.mu * self.v + g
        d = g + self.mu * self.v if self.nes else self.v
        return p - self.lr * d

class Adam:
    def __init__(self, lr, b1=0.9, b2=0.999, eps=1e-8, wd=0.0):
        self.lr, self.b1, self.b2, self.eps, self.wd = lr, b1, b2, eps, wd
        self.m = None
        self.v = None
        self.t = 0
    def step(self, p, g):
        if self.m is None:
            self.m = np.zeros_like(p)
            self.v = np.zeros_like(p)
        self.t += 1
        self.m = self.b1 * self.m + (1 - self.b1) * g
        self.v = self.b2 * self.v + (1 - self.b2) * g * g
        mh = self.m / (1 - self.b1 ** self.t)
        vh = self.v / (1 - self.b2 ** self.t)
        return p - self.lr * mh / (np.sqrt(vh) + self.eps) - self.lr * self.wd * p

def warmup_cosine(t, peak, warmup, total, lr_min=0.0):
    if t < warmup:
        return peak * t / warmup
    progress = (t - warmup) / (total - warmup)
    return lr_min + 0.5 * (peak - lr_min) * (1 + math.cos(math.pi * progress))

print("SGD:", steps_to(SGD(2 / 101)))
print("Momentum:", steps_to(Momentum(4 / 121, (9 / 11) ** 2)))
