An agent loop (model -> maybe tool -> model ...) is a **state machine**: a state dict flows through **nodes** (functions returning a partial update) and **edges** (static or conditional). Making it explicit gives you checkpoints, interrupts for human approval, and a hard step limit.

Implement `run_graph(graph, state, start=None, max_steps=20, interrupt_before=(), resume=False)` where `graph = {"nodes": {name: fn(state) -> dict}, "edges": {name: next_name | fn(state) -> next_name}, "reducers": {field: fn(old, new) -> value}, "entry": name}` and `END = "__end__"`. It returns a dict:

- `"state"`: the final (or current) state;
- `"trace"`: names of the nodes executed, in order;
- `"checkpoints"`: after each executed node a dict `{"node", "state"}` with a **deep copy** of the state;
- `"status"`: `"done"` (reached END), `"interrupted"` (about to run a node listed in `interrupt_before`, unless `resume=True` for that first node), or `"max_steps"` (executed `max_steps` nodes without reaching END);
- `"next"`: the node to run next (for resuming), or `None`.

Do not mutate the `state` you were given: work on a copy.

import copy
SCRIPT = [{"tool": "add", "args": [2, 3]}, {"final": "The sum is 5"}]

def agent(state):
    i = state["turn"]
    out = SCRIPT[i % len(SCRIPT)] if state.get("loop") else SCRIPT[i]
    return {"turn": i + 1, "messages": [("assistant", out)], "pending": out if "tool" in out else None}

def tools(state):
    call = state["pending"]
    return {"messages": [("tool", sum(call["args"]))], "pending": None}

def route(state):
    return "tools" if state["pending"] else END

G = {
    "nodes": {"agent": agent, "tools": tools},
    "edges": {"agent": route, "tools": "agent"},
    "reducers": {"messages": lambda old, new: (old or []) + new},
    "entry": "agent",
}
init = {"turn": 0, "messages": [("user", "what is 2+3?")], "pending": None}
r = run_graph(G, init)
check("react loop: agent, tools, agent", lambda: r["trace"] == ["agent", "tools", "agent"])
check("status is done and next is None", lambda: r["status"] == "done" and r["next"] is None)
check("messages accumulate through the reducer (user, assistant, tool, assistant)", lambda: len(r["state"]["messages"]) == 4)
check("final message is the model's answer", lambda: r["state"]["messages"][-1] == ("assistant", {"final": "The sum is 5"}))

import copy

END = "__end__"

def run_graph(graph, state, start=None, max_steps=20, interrupt_before=(), resume=False):
    state = copy.deepcopy(state)
    node = start or graph["entry"]
    trace, checkpoints = [], []
    first = True
    while True:
        if node == END:
            return {"state": state, "trace": trace, "checkpoints": checkpoints, "status": "done", "next": None}
        if node in interrupt_before and not (resume and first):
            return {"state": state, "trace": trace, "checkpoints": checkpoints, "status": "interrupted", "next": node}
        if len(trace) >= max_steps:
            return {"state": state, "trace": trace, "checkpoints": checkpoints, "status": "max_steps", "next": node}
        first = False
        update = graph["nodes"][node](state)
        for k, v in update.items():
            red = graph.get("reducers", {}).get(k)
            state[k] = red(state.get(k), v) if red else v
        trace.append(node)
        checkpoints.append({"node": node, "state": copy.deepcopy(state)})
        nxt = graph["edges"][node]
        node = nxt(state) if callable(nxt) else nxt
