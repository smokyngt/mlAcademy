#@@ prompt
Your application receives model outputs as **strings** and needs typed data. Build the two pieces that sit between the model and your code. No model is called: the outputs are canned.

1. `validate(value, schema, path="$")` -> list of error strings (empty list means valid). Support these JSON Schema keywords: `type` (`object`, `array`, `string`, `integer`, `number`, `boolean`, `null`), `required`, `properties`, `additionalProperties` (only the value `False`), `enum`, `minimum`, `maximum`, `minLength`, `items`. Each error string starts with the path, e.g. `$.price: expected number, got str`. Remember: a Python `bool` is not an `integer` or `number` in JSON Schema.
2. `extract_json(text)` -> the JSON text inside a model reply: if there is a fenced block (three backticks, optional `json` tag) take its content; otherwise slice from the first `{` or `[` to the last `}` or `]`; otherwise return the text unchanged.
3. `repair_loop(outputs, schema, max_attempts)` -> dict `{"ok", "value", "attempts", "feedback"}`. For attempt `i` take `outputs[i]`: extract, `json.loads` (on failure the feedback is `"invalid JSON: ..."`), then `validate` (on failure the feedback is the errors joined by `"; "`). Stop at the first valid output. `feedback` is the list of messages you would send back to the model, one per failed attempt. If `max_attempts` is exhausted, return `ok=False`, `value=None`.

import json

def _type_ok(v, t):
    if t == "object":
        return isinstance(v, dict)
    if t == "array":
        return isinstance(v, list)
    if t == "string":
        return isinstance(v, str)
    if t == "boolean":
        return isinstance(v, bool)
    if t == "null":
        return v is None
    if t == "integer":
        return isinstance(v, int) and not isinstance(v, bool)
    if t == "number":
        return isinstance(v, (int, float)) and not isinstance(v, bool)
    return True

def validate(value, schema, path="$"):
    errs = []
    t = schema.get("type")
    if t is not None and not _type_ok(value, t):
        return [f"{path}: expected {t}, got {type(value).__name__}"]
    if "enum" in schema and value not in schema["enum"]:
        errs.append(f"{path}: {value!r} not in {schema['enum']}")
    if isinstance(value, str) and len(value) < schema.get("minLength", 0):
        errs.append(f"{path}: shorter than minLength {schema['minLength']}")
    if _type_ok(value, "number") and not isinstance(value, bool):
        if "minimum" in schema and value < schema["minimum"]:
            errs.append(f"{path}: {value} < minimum {schema['minimum']}")
        if "maximum" in schema and value > schema["maximum"]:
            errs.append(f"{path}: {value} > maximum {schema['maximum']}")
    if isinstance(value, dict):
        props = schema.get("properties", {})
        for k in schema.get("required", []):
            if k not in value:
                errs.append(f"{path}: missing required key '{k}'")
        for k, v in value.items():
            if k in props:
                errs += validate(v, props[k], f"{path}.{k}")
            elif schema.get("additionalProperties") is False:
                errs.append(f"{path}: unexpected key '{k}'")
    if isinstance(value, list) and "items" in schema:
        for i, v in enumerate(value):
            errs += validate(v, schema["items"], f"{path}[{i}]")
    return errs

def extract_json(text):
    if "```" in text:
        inner = text.split("```")[1]
        if inner.startswith("json"):
            inner = inner[4:]
        return inner.strip()
    starts = [i for i in (text.find("{"), text.find("[")) if i >= 0]
    ends = [i for i in (text.rfind("}"), text.rfind("]")) if i >= 0]
    if starts and ends and max(ends) > min(starts):
        return text[min(starts):max(ends) + 1]
    return text

def repair_loop(outputs, schema, max_attempts=3):
    feedback = []
    for i, out in enumerate(outputs[:max_attempts]):
        try:
            value = json.loads(extract_json(out))
        except json.JSONDecodeError as e:
            feedback.append(f"invalid JSON: {e.msg}")
            continue
        errs = validate(value, schema)
        if not errs:
            return {"ok": True, "value": value, "attempts": i + 1, "feedback": feedback}
        feedback.append("; ".join(errs))
    return {"ok": False, "value": None, "attempts": min(max_attempts, len(outputs)), "feedback": feedback}
