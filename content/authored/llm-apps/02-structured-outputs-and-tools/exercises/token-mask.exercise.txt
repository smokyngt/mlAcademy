#@@ prompt
Constrained decoding makes invalid output **unreachable** by setting the logits of illegal tokens to $-\infty$ before the softmax or argmax. You will implement the core of it on a tiny grammar, with a deterministic fake "model".

The grammar (given) is the language of strings like `{"sentiment":"positive","score":7}`:

```
{"sentiment":"(positive|negative)","score":[0-9]{1,2}}
```

`build_dfa(pieces)` is provided and returns a character-level automaton `{"trans": [dict, ...], "start": int, "accept": int}` where `trans[state][char]` is the next state. `VOCAB` is a list of **multi-character tokens** (including junk such as `"Sure"` and the end token `"<eos>"`).

Implement:

- `dfa_step(dfa, state, token)` -> the state after feeding all characters of `token`, or `None` if some character has no transition.
- `token_mask(dfa, state, vocab)` -> list of booleans, one per token: `True` iff the token can legally be emitted now. `"<eos>"` is legal only in the accept state.
- `constrained_greedy(dfa, vocab, logits_fn, max_steps=40)` -> the generated string. At every step call `logits_fn(step)` (a list of floats), mask illegal tokens to `-inf`, take the argmax, and stop at `"<eos>"`.

import numpy as np

def build_dfa(pieces):
    trans = [{}]
    def new():
        trans.append({})
        return len(trans) - 1
    acc = new()
    nxt = acc
    for p in reversed(pieces):
        if p[0] == "lit":
            s = nxt
            for ch in reversed(p[1]):
                n = new(); trans[n][ch] = s; s = n
            nxt = s
        elif p[0] == "choice":
            start = new()
            for w in p[1]:
                s = nxt
                for ch in reversed(w[1:]):
                    n = new(); trans[n][ch] = s; s = n
                trans[start][w[0]] = s
            nxt = start
        else:
            lo, hi = p[1], p[2]
            states = [new() for _ in range(hi + 1)]
            for k in range(hi + 1):
                if k < hi:
                    for d in "0123456789":
                        trans[states[k]][d] = states[k + 1]
                if k >= lo:
                    trans[states[k]].update(trans[nxt])
            nxt = states[0]
    return {"trans": trans, "start": nxt, "accept": acc}

VOCAB = ["<eos>", "Sure", ", here", " is", "\n", '{"', '{"sentiment":"', "sent", "iment", '":"', "pos", "itive", "positive", "neg", "ative", "negative",
         '","score":', "1", "2", "7", "0", "9", "42", "}", "}\n"]

def fake_logits(step):
    z = np.random.default_rng(step).normal(size=len(VOCAB))
    if step == 0:
        z[VOCAB.index("Sure")] += 5.0
    return list(z)

def dfa_step(dfa, state, token):
    for ch in token:
        state = dfa["trans"][state].get(ch)
        if state is None:
            return None
    return state

def token_mask(dfa, state, vocab):
    mask = []
    for t in vocab:
        if t == "<eos>":
            mask.append(state == dfa["accept"])
        else:
            mask.append(dfa_step(dfa, state, t) is not None)
    return mask

def constrained_greedy(dfa, vocab, logits_fn, max_steps=40):
    state, out = dfa["start"], ""
    for step in range(max_steps):
        z = np.array(logits_fn(step), dtype=float)
        z[~np.array(token_mask(dfa, state, vocab))] = -np.inf
        tok = vocab[int(np.argmax(z))]
        if tok == "<eos>":
            break
        out += tok
        state = dfa_step(dfa, state, tok)
    return out
