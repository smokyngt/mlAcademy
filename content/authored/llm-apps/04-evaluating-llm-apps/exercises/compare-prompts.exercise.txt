#@@ prompt
Two prompts were run on the same 40 test items and each output was graded pass (1) or fail (0): `A_SCORES` and `B_SCORES`. Prompt B passes more items. Is that evidence, or noise? Implement the tools to answer, offline:

- `paired_diffs(a, b)` -> NumPy array of `b_i - a_i`.
- `bootstrap_ci(d, B=2000, alpha=0.05, seed=0)` -> `(low, high)`, a percentile bootstrap confidence interval for the **mean** of `d`: draw `B` resamples of size `n` with replacement using `np.random.default_rng(seed).integers(0, n, size=(B, n))`.
- `sign_flip_pvalue(d)` -> exact two-sided p-value of the paired permutation (sign-flip) test on the mean difference: drop zero differences, enumerate every assignment of signs to the remaining `m` differences (`m <= 20`), and return the fraction of assignments whose `|sum|` is at least the observed `|sum|`.
- `required_n(delta, sd, z_alpha=1.96, z_power=0.8416)` -> the number of items needed to detect a mean paired difference `delta` when per-item differences have standard deviation `sd`: `ceil(((z_alpha + z_power) * sd / delta) ** 2)`.
- `cohen_kappa(human, judge)` -> chance-corrected agreement between an LLM judge's labels and human labels: `(p_o - p_e) / (1 - p_e)`, where `p_e = sum over classes of P_human(c) * P_judge(c)`.

import numpy as np
from math import ceil

A_SCORES = [0] * 9 + [1] * 3 + [1] * 16 + [0] * 12
B_SCORES = [1] * 9 + [0] * 3 + [1] * 16 + [0] * 12

def paired_diffs(a, b):
    return np.asarray(b, dtype=float) - np.asarray(a, dtype=float)

def bootstrap_ci(d, B=2000, alpha=0.05, seed=0):
    d = np.asarray(d, dtype=float)
    n = len(d)
    rng = np.random.default_rng(seed)
    idx = rng.integers(0, n, size=(B, n))
    means = d[idx].mean(axis=1)
    return float(np.percentile(means, 100 * alpha / 2)), float(np.percentile(means, 100 * (1 - alpha / 2)))

def sign_flip_pvalue(d):
    d = np.asarray(d, dtype=float)
    nz = d[d != 0]
    m = len(nz)
    if m == 0:
        return 1.0
    assert m <= 20
    bits = ((np.arange(2 ** m)[:, None] >> np.arange(m)) & 1) * 2 - 1
    sums = (bits * nz).sum(axis=1)
    return float(np.mean(np.abs(sums) >= abs(nz.sum()) - 1e-12))

def required_n(delta, sd, z_alpha=1.96, z_power=0.8416):
    return int(ceil(((z_alpha + z_power) * sd / delta) ** 2))

def cohen_kappa(human, judge):
    human, judge = list(human), list(judge)
    n = len(human)
    po = sum(h == j for h, j in zip(human, judge)) / n
    classes = set(human) | set(judge)
    pe = sum((human.count(c) / n) * (judge.count(c) / n) for c in classes)
    return (po - pe) / (1 - pe)
