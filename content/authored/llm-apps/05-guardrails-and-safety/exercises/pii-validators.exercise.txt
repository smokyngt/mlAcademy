A regex alone finds anything that *looks* like a card number, which produces many false alarms (order numbers, phone numbers). Real PII filters add a **validator** that checks a built-in checksum or structure rule, so random digit strings are rejected. Implement it, offline:

- `luhn_valid(s)` -> bool: the Luhn checksum over the digits of `s` (ignore spaces and dashes; need at least 2 digits).
- `iban_valid(s)` -> bool: structure `2 letters + 2 digits + 11..30 alphanumerics` (spaces ignored, case-insensitive) and the mod-97 rule.
- `ssn_valid(s)` -> bool: US SSN of the form `AAA-GG-SSSS` where `AAA` is not `000`, not `666` and does not start with `9`, `GG` is not `00`, and `SSSS` is not `0000`.
- `find_pii(text)` -> list of `(type, matched_text)` in order of position, using `PATTERNS` (given: a regex and a validator per type). A candidate counts only if its validator passes; a candidate overlapping an already-accepted span is ignored (this stops the digits inside an IBAN being re-read as a card).
- `redact(text)` -> the text with each accepted span replaced by `[TYPE]`.

import re

def luhn_valid(s):
    d = [int(c) for c in s if c.isdigit()]
    if len(d) < 2:
        return False
    total = 0
    for i, x in enumerate(reversed(d)):
        if i % 2 == 1:
            x *= 2
            if x > 9:
                x -= 9
        total += x
    return total % 10 == 0

def iban_valid(s):
    s = s.replace(" ", "").upper()
    if not re.fullmatch(r"[A-Z]{2}\d{2}[A-Z0-9]{11,30}", s):
        return False
    r = s[4:] + s[:4]
    return int("".join(str(int(c, 36)) for c in r)) % 97 == 1

def ssn_valid(s):
    m = re.fullmatch(r"(\d{3})-(\d{2})-(\d{4})", s)
    if not m:
        return False
    a, g, n = m.groups()
    return a not in ("000", "666") and not a.startswith("9") and g != "00" and n != "0000"

PATTERNS = [
    ("EMAIL", r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+", lambda s: True),
    ("IBAN", r"\b[A-Z]{2}\d{2}(?: ?[A-Z0-9]{4}){2,7}(?: ?[A-Z0-9]{1,4})?\b", lambda s: iban_valid(s)),
    ("CARD", r"\b\d(?:[ -]?\d){12,18}\b", lambda s: luhn_valid(s)),
    ("SSN", r"\b\d{3}-\d{2}-\d{4}\b", lambda s: ssn_valid(s)),
]

def _spans(text):
    claimed, out = [], []
    for typ, pat, ok in PATTERNS:
        for m in re.finditer(pat, text):
            if any(m.start() < e and s < m.end() for s, e in claimed):
                continue
            if ok(m.group()):
                claimed.append((m.start(), m.end()))
                out.append((m.start(), m.end(), typ, m.group()))
    return sorted(out)

def find_pii(text):
    return [(t, x) for _, _, t, x in _spans(text)]

def redact(text):
    for s, e, t, _ in reversed(_spans(text)):
        text = text[:s] + "[" + t + "]" + text[e:]
    return text
