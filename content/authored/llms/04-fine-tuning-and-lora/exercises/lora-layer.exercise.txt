import numpy as np

def lora_init(d_out, d_in, r, rng):
    A = rng.normal(0.0, 1.0 / np.sqrt(d_in), size=(r, d_in))
    B = np.zeros((d_out, r))
    return A, B

def lora_forward(x, W, A, B, alpha):
    s = alpha / A.shape[0]
    h = x @ A.T
    return x @ W.T + s * (h @ B.T), h

def lora_backward(G, x, h, W, A, B, alpha):
    s = alpha / A.shape[0]
    GB = G @ B
    dB = s * G.T @ h
    dA = s * GB.T @ x
    dx = G @ W + s * GB @ A
    return dx, dA, dB

def lora_param_count(d_out, d_in, r):
    return r * (d_out + d_in)

def merge(W, A, B, alpha):
    return W + (alpha / A.shape[0]) * B @ A

def fit_lora(x, y_target, W, r, alpha, lr, steps, seed):
    rng = np.random.default_rng(seed)
    A, B = lora_init(W.shape[0], W.shape[1], r, rng)
    n = len(x)
    losses = []
    for _ in range(steps):
        y, h = lora_forward(x, W, A, B, alpha)
        E = y - y_target
        losses.append(0.5 * float(np.sum(E ** 2)) / n)
        _, dA, dB = lora_backward(E / n, x, h, W, A, B, alpha)
        A = A - lr * dA
        B = B - lr * dB
    return A, B, losses

print(lora_param_count(4096, 4096, 16))
