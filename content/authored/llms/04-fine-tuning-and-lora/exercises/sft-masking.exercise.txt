import numpy as np

IGNORE = -100

def build_example(prompt_ids, response_ids, eos):
    ids = list(prompt_ids) + list(response_ids) + [eos]
    labels = [IGNORE] * len(prompt_ids) + list(response_ids) + [eos]
    return ids, labels

def to_xy(ids, labels):
    return list(ids[:-1]), list(labels[1:])

def collate(pairs, pad_id):
    T = max(len(x) for x, _ in pairs)
    X = np.full((len(pairs), T), pad_id, dtype=int)
    Y = np.full((len(pairs), T), IGNORE, dtype=int)
    for i, (x, y) in enumerate(pairs):
        X[i, :len(x)] = x
        Y[i, :len(y)] = y
    return X, Y

def masked_ce(logits, Y):
    z = logits - logits.max(axis=-1, keepdims=True)
    p = np.exp(z)
    p /= p.sum(axis=-1, keepdims=True)
    valid = Y != IGNORE
    n = int(valid.sum())
    if n == 0:
        return 0.0, np.zeros_like(logits)
    b, t = np.nonzero(valid)
    loss = float(-np.mean(np.log(p[b, t, Y[b, t]])))
    g = np.zeros_like(logits)
    g[b, t] = p[b, t]
    g[b, t, Y[b, t]] -= 1.0
    g /= n
    return loss, g

ids, labels = build_example([5, 6, 7], [8, 9], eos=1)
print(ids, labels)
print(to_xy(ids, labels))
