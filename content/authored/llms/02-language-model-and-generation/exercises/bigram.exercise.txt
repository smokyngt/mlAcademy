import numpy as np

def softmax(z):
    z = z - z.max(axis=-1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def bigram_counts(ids, V):
    C = np.zeros((V, V))
    for a, b in zip(ids[:-1], ids[1:]):
        C[a, b] += 1
    return C

def bigram_probs(C, k=0.0):
    Ck = C + k
    s = Ck.sum(axis=1, keepdims=True)
    P = np.where(s > 0, Ck / np.where(s > 0, s, 1), 1.0 / C.shape[1])
    return P

def mean_nll(P, ids):
    x, y = ids[:-1], ids[1:]
    return float(-np.mean(np.log(P[x, y])))

def grad_bigram(W, ids):
    x, y = ids[:-1], ids[1:]
    N = len(x)
    P = softmax(W[x])
    loss = float(-np.mean(np.log(P[np.arange(N), y])))
    G = P.copy()
    G[np.arange(N), y] -= 1.0
    G /= N
    dW = np.zeros_like(W)
    np.add.at(dW, x, G)
    return loss, dW

def train_bigram_gd(ids, V, lr=1.0, steps=2000):
    W = np.zeros((V, V))
    for _ in range(steps):
        _, dW = grad_bigram(W, ids)
        W -= lr * dW
    return W

ids = np.array([idx[w] for w in words])
print(vocab, ids)
