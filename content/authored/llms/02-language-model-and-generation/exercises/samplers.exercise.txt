import numpy as np

def softmax(z):
    z = np.asarray(z, dtype=float)
    z = z - z.max(axis=-1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def temperature_probs(logits, T):
    return softmax(np.asarray(logits, dtype=float) / T)

def top_k_probs(p, k):
    idx = np.argsort(-p, kind="stable")[:k]
    q = np.zeros_like(p)
    q[idx] = p[idx]
    return q / q.sum()

def top_p_probs(p, top_p):
    order = np.argsort(-p, kind="stable")
    cum = np.cumsum(p[order])
    n = int(np.searchsorted(cum, top_p - 1e-12) + 1)
    n = min(n, len(p))
    q = np.zeros_like(p)
    q[order[:n]] = p[order[:n]]
    return q / q.sum()

def sample(logits, rng, temperature=1.0, top_k=None, top_p=None):
    logits = np.asarray(logits, dtype=float)
    if temperature == 0:
        return int(np.argmax(logits))
    p = temperature_probs(logits, temperature)
    if top_k is not None:
        p = top_k_probs(p, top_k)
    if top_p is not None:
        p = top_p_probs(p, top_p)
    return int(rng.choice(len(p), p=p))

def perplexity(logits, targets):
    logits = np.asarray(logits, dtype=float)
    z = logits - logits.max(axis=-1, keepdims=True)
    logp = z - np.log(np.exp(z).sum(axis=-1, keepdims=True))
    nll = -logp[np.arange(len(targets)), targets]
    return float(np.exp(nll.mean()))

z = np.array([2.0, 1.0, 0.5, 0.0, -1.0])
print(np.round(temperature_probs(z, 1.0), 4))
