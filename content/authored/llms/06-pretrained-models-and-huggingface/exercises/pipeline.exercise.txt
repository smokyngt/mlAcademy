2. `WordTokenizer.batch_encode(texts)`: returns `(ids, mask)`, NumPy int arrays of shape `(B, T)` where `T` is the longest encoded text; padding uses `pad_id` and `mask` is 1 for real tokens, 0 for padding.

6. `check_model_card(card)`: required fields are `license`, `language`, `intended_use`, `limitations`, `training_data`, `metrics`. Report `missing:<field>` if absent, `empty:<field>` if the value is empty (empty string, list or dict), and `bad_metric:<name>` for each entry of `metrics` whose value is not a number in [0, 1] (booleans do not count as numbers). Return the sorted list of problems; an empty list means the card is acceptable.

import numpy as np

class WordTokenizer:
    def __init__(self, vocab, unk="[UNK]", pad="[PAD]"):
        self.vocab = dict(vocab)
        self.unk_id = self.vocab[unk]
        self.pad_id = self.vocab[pad]
        self.inv = {i: w for w, i in self.vocab.items()}

    def encode(self, text):
        return [self.vocab.get(w, self.unk_id) for w in text.lower().split()]

    def batch_encode(self, texts):
        enc = [self.encode(t) for t in texts]
        T = max(len(e) for e in enc)
        ids = np.full((len(enc), T), self.pad_id, dtype=int)
        mask = np.zeros((len(enc), T), dtype=int)
        for i, e in enumerate(enc):
            ids[i, :len(e)] = e
            mask[i, :len(e)] = 1
        return ids, mask

    def decode(self, ids):
        return " ".join(self.inv[i] for i in ids if i != self.pad_id)

class BagOfWordsSentiment:
    def __init__(self, table):
        self.table = np.asarray(table, dtype=float)
    def __call__(self, ids, mask):
        return (self.table[ids] * mask[..., None]).sum(axis=1)

def classification_decoder(logits, labels):
    z = np.asarray(logits, dtype=float)
    e = np.exp(z - z.max())
    p = e / e.sum()
    k = int(np.argmax(p))
    return {"label": labels[k], "score": float(p[k])}

class Pipeline:
    def __init__(self, tokenizer, model, decoder):
        self.tokenizer, self.model, self.decoder = tokenizer, model, decoder
    def __call__(self, inputs):
        single = isinstance(inputs, str)
        texts = [inputs] if single else list(inputs)
        ids, mask = self.tokenizer.batch_encode(texts)
        logits = self.model(ids, mask)
        out = [self.decoder(row) for row in logits]
        return out[0] if single else out

REGISTRY = {}

def register(model_type):
    def deco(cls):
        REGISTRY[model_type] = cls
        return cls
    return deco

def auto_from_config(config):
    t = config["model_type"]
    if t not in REGISTRY:
        raise KeyError("unknown model_type %r; known types: %s" % (t, sorted(REGISTRY)))
    return REGISTRY[t](config)

REQUIRED = ["license", "language", "intended_use", "limitations", "training_data", "metrics"]

def check_model_card(card):
    problems = []
    for f in REQUIRED:
        if f not in card:
            problems.append("missing:" + f)
        elif card[f] in ("", [], {}, None):
            problems.append("empty:" + f)
    m = card.get("metrics")
    if isinstance(m, dict):
        for name, v in m.items():
            if isinstance(v, bool) or not isinstance(v, (int, float)) or not (0 <= v <= 1):
                problems.append("bad_metric:" + name)
    return sorted(problems)
