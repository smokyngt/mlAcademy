- `get_stats(ids)` returns a `Counter` of adjacent pairs.
- `merge(ids, pair, new_id)` replaces every non-overlapping occurrence of `pair` (left to right) by `new_id`.

Round trips must be exact, for any text, even characters never seen in training.

import numpy as np
from collections import Counter
import re

def get_stats(ids):
    c = Counter()
    for a, b in zip(ids, ids[1:]):
        c[(a, b)] += 1
    return c

def merge(ids, pair, new_id):
    out = []
    i = 0
    while i < len(ids):
        if i < len(ids) - 1 and (ids[i], ids[i + 1]) == pair:
            out.append(new_id)
            i += 2
        else:
            out.append(ids[i])
            i += 1
    return out

def train_bpe(text, num_merges):
    ids = list(text.encode("utf-8"))
    merges = {}
    for k in range(num_merges):
        stats = get_stats(ids)
        if not stats:
            break
        pair = min(stats, key=lambda p: (-stats[p], p))
        if stats[pair] < 2:
            break
        merges[pair] = 256 + k
        ids = merge(ids, pair, 256 + k)
    return merges

def encode(text, merges):
    ids = list(text.encode("utf-8"))
    while len(ids) >= 2:
        stats = get_stats(ids)
        pair = min(stats, key=lambda p: merges.get(p, float("inf")))
        if pair not in merges:
            break
        ids = merge(ids, pair, merges[pair])
    return ids

def decode(ids, merges):
    vocab = {i: bytes([i]) for i in range(256)}
    for (a, b), new in merges.items():
        vocab[new] = vocab[a] + vocab[b]
    return b"".join(vocab[i] for i in ids).decode("utf-8", errors="replace")

def encode_with_specials(text, merges, specials):
    pattern = "(" + "|".join(re.escape(s) for s in specials) + ")"
    out = []
    for piece in re.split(pattern, text):
        if piece in specials:
            out.append(specials[piece])
        elif piece:
            out.extend(encode(piece, merges))
    return out

merges = train_bpe("banana bandana", 3)
print(merges)
print(encode("banana bandana", merges))
