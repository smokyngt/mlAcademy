import re
import numpy as np

def doc_stats(text):
    words = text.split()
    n = len(words)
    chars = len(text)
    alpha = sum(c.isalpha() for c in text)
    sym = sum(c in "#{}[]<>|^~@$%&*=_" for c in text)
    tri = [tuple(words[i:i + 3]) for i in range(n - 2)]
    rep = 1 - len(set(tri)) / len(tri) if tri else 0.0
    return {"n_words": n,
            "mean_word_len": sum(len(w) for w in words) / n if n else 0.0,
            "alpha_frac": alpha / chars if chars else 0.0,
            "symbol_frac": sym / chars if chars else 0.0,
            "repeat_frac": rep}

DEFAULT_CFG = {"min_words": 8, "max_words": 2000, "min_mean_len": 3.0, "max_mean_len": 10.0,
               "min_alpha": 0.6, "max_symbol": 0.1, "max_repeat": 0.3}

def failed_rules(text, cfg=DEFAULT_CFG):
    s = doc_stats(text)
    f = []
    if s["n_words"] < cfg["min_words"] or s["n_words"] > cfg["max_words"]: f.append("length")
    if not (cfg["min_mean_len"] <= s["mean_word_len"] <= cfg["max_mean_len"]): f.append("word_len")
    if s["alpha_frac"] < cfg["min_alpha"]: f.append("alpha")
    if s["symbol_frac"] > cfg["max_symbol"]: f.append("symbols")
    if s["repeat_frac"] > cfg["max_repeat"]: f.append("repetition")
    return f

def build_vocab(docs, min_count=2):
    cnt = {}
    for d in docs:
        for w in set(re.findall(r"[a-z0-9']+", d.lower())):
            cnt[w] = cnt.get(w, 0) + 1
    return sorted(w for w, c in cnt.items() if c >= min_count)

def bow(docs, vocab):
    idx = {w: i for i, w in enumerate(vocab)}
    X = np.zeros((len(docs), len(vocab)))
    for r, d in enumerate(docs):
        for w in set(re.findall(r"[a-z0-9']+", d.lower())):
            if w in idx: X[r, idx[w]] = 1.0
    return X

def sigmoid(z): return 1 / (1 + np.exp(-np.clip(z, -30, 30)))

def logreg_loss_grad(w, b, X, y, l2):
    p = sigmoid(X @ w + b)
    eps = 1e-12
    loss = -np.mean(y * np.log(p + eps) + (1 - y) * np.log(1 - p + eps)) + 0.5 * l2 * np.sum(w * w)
    r = (p - y) / len(y)
    return loss, X.T @ r + l2 * w, r.sum()

def train_quality_classifier(X, y, lr=0.5, epochs=300, l2=1e-3):
    w = np.zeros(X.shape[1]); b = 0.0; hist = []
    for _ in range(epochs):
        loss, gw, gb = logreg_loss_grad(w, b, X, y, l2)
        w -= lr * gw; b -= lr * gb; hist.append(loss)
    return w, b, hist

PII_PATTERNS = [
    ("EMAIL", re.compile(r"[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}")),
    ("PHONE", re.compile(r"(?<![\d.])(?:\+?\d{1,2}[ -])?(?:\(\d{3}\)\s?|\d{3}[ -])\d{3}[ -]\d{4}(?!\d)")),
    ("IP", re.compile(r"(?<![\d.])(?:(?:25[0-5]|2[0-4]\d|1?\d?\d)\.){3}(?:25[0-5]|2[0-4]\d|1?\d?\d)(?!\d)(?!\.\d)")),
]

def scrub_pii(text):
    counts = {}
    for name, pat in PII_PATTERNS:
        text, k = pat.subn(f"<{name}>", text)
        counts[name] = k
    return text, counts
