Implement the machinery of web-scale deduplication and benchmark decontamination, and measure how well it works.

- `normalize`, `shingles(text, k)` (word k-shingles), `jaccard`.
- `minhash_signature(shingle_set, hashes)`: each shingle is hashed with `zlib.crc32(...) % P`, then through `n` universal hashes `(A*x + B) mod P`; the signature keeps the minimum per hash. `estimate_jaccard` is the fraction of equal signature positions.
- `lsh_candidates(sigs, bands, rows)`: split each signature into `bands` bands of `rows` rows; two documents are candidates if some band is identical. `lsh_prob(s, bands, rows)` gives the theoretical collision probability.
- `exact_dedup(docs)`: indices of the first copy of each distinct normalised document.
- `ngram_set`, `contamination_score(item, train_ngrams, n)` and `decontaminate(train_docs, items, n, tau)`: flag a benchmark item when at least a fraction `tau` of its word n-grams occur in the training text.

The tests print recall and precision of near-duplicate detection for several band layouts, and the flag rates of verbatim leaks, paraphrases, clean items and items that merely share boilerplate wording.

import re, zlib
import numpy as np

def normalize(text):
    return re.sub(r"\s+", " ", text.lower()).strip()

def shingles(text, k=5):
    w = normalize(text).split(" ")
    if len(w) < k:
        return {" ".join(w)} if w and w != [""] else set()
    return {" ".join(w[i:i + k]) for i in range(len(w) - k + 1)}

def jaccard(a, b):
    if not a and not b:
        return 1.0
    return len(a & b) / len(a | b)

P = 2**31 - 1

def make_hashes(n, seed=0):
    rng = np.random.default_rng(seed)
    return (rng.integers(1, P, size=n, dtype=np.uint64), rng.integers(0, P, size=n, dtype=np.uint64))

def minhash_signature(sh, hashes):
    A, B = hashes
    x = np.array([zlib.crc32(s.encode()) % P for s in sh], dtype=np.uint64)
    return ((A[:, None] * x[None, :] + B[:, None]) % np.uint64(P)).min(axis=1)

def estimate_jaccard(s1, s2):
    return float(np.mean(s1 == s2))

def lsh_candidates(sigs, bands, rows):
    assert sigs.shape[1] == bands * rows
    pairs = set()
    for b in range(bands):
        buckets = {}
        for i, s in enumerate(sigs):
            buckets.setdefault(tuple(s[b * rows:(b + 1) * rows].tolist()), []).append(i)
        for members in buckets.values():
            for x in range(len(members)):
                for y in range(x + 1, len(members)):
                    pairs.add((members[x], members[y]))
    return pairs

def lsh_prob(s, bands, rows):
    return 1 - (1 - s ** rows) ** bands

def ngram_set(text, n):
    w = normalize(text).split(" ")
    return {" ".join(w[i:i + n]) for i in range(len(w) - n + 1)}

def contamination_score(item, train_ngrams, n):
    g = ngram_set(item, n)
    return len(g & train_ngrams) / len(g) if g else 0.0


def exact_dedup(docs):
    """Indices of the first occurrence of each distinct normalised document."""
    seen, keep = set(), []
    for i, d in enumerate(docs):
        key = normalize(d)
        if key not in seen:
            seen.add(key)
            keep.append(i)
    return keep

def decontaminate(train_docs, items, n, tau):
    """Flag benchmark items whose n-gram overlap with the training text is >= tau."""
    train_ngrams = set()
    for d in train_docs:
        train_ngrams |= ngram_set(d, n)
    return [contamination_score(it, train_ngrams, n) >= tau for it in items]
