import numpy as np

def gpt_params(V, d, L, ctx, tied=True, ffn_mult=4):
    m = ffn_mult
    block = (3 * d * d + 3 * d) + (d * d + d) + (d * m * d + m * d) + (m * d * d + d) + 4 * d
    total = V * d + ctx * d + L * block + 2 * d
    if not tied:
        total += V * d
    return total

def train_flops(N, D):
    return 6 * N * D

def compute_optimal(C, tokens_per_param=20):
    N = (C / (6 * tokens_per_param)) ** 0.5
    return N, tokens_per_param * N

def chinchilla_loss(N, D, A=406.4, B=410.7, alpha=0.34, beta=0.28, E=1.69):
    return A / N ** alpha + B / D ** beta + E

def best_split(C, points=4000):
    Ns = np.logspace(6, 13, points)
    Ds = C / (6 * Ns)
    Ls = chinchilla_loss(Ns, Ds)
    i = int(np.argmin(Ls))
    return float(Ns[i]), float(Ds[i]), float(Ls[i])

def training_memory_bytes(N, bytes_weights=4, bytes_grads=4, bytes_opt=8):
    out = {"weights": N * bytes_weights, "grads": N * bytes_grads, "opt": N * bytes_opt}
    out["total"] = out["weights"] + out["grads"] + out["opt"]
    return out

def kv_cache_bytes(L, d, T, B=1, bytes_per=2):
    return 2 * L * d * T * B * bytes_per

def days_to_train(C, flops_per_sec, utilisation):
    return C / (flops_per_sec * utilisation) / 86400

print(gpt_params(50257, 768, 12, 1024))
