Build the data side of a pretraining run, all in NumPy/pure Python.

- `dedup_exact(docs)`: remove documents whose whitespace/case-normalised text was already seen (keep the first).
- `jaccard(a, b, n=3)`: Jaccard similarity of the sets of word `n`-grams (shingles) of two strings. If a text has fewer than `n` words, its single shingle is the whole word tuple.
- `dedup_near(docs, threshold=0.8, n=3)`: greedily keep a document only if its Jaccard similarity with every previously kept one is `< threshold`.
- `split_docs(docs, val_frac, seed)`: seeded random split **by document** into `(train, val)`; `val` has `max(1, round(val_frac * len(docs)))` documents.
- `pack(docs_ids, eos, block)`: concatenate every token-id list followed by `eos`, and cut into `x, y` arrays of shape `(n, block)` with `y` the inputs shifted one position (so `y[:, :-1] == x[:, 1:]`). The incomplete tail is dropped.
- `shuffle_blocks(x, y, seed)`: apply the same seeded permutation to both arrays.
- `doc_mask(block_ids, eos)`: boolean `(T, T)` matrix, `True` where position `i` may attend to `j`: causal (`j <= i`) **and** in the same document (documents are separated by `eos`; the `eos` belongs to the document it ends).
- `padding_fraction(lengths, block)`: share of wasted slots if every document is padded up to a multiple of `block` instead of packed.

import numpy as np

def _norm(doc):
    return " ".join(doc.lower().split())

def dedup_exact(docs):
    seen, out = set(), []
    for d in docs:
        k = _norm(d)
        if k not in seen:
            seen.add(k)
            out.append(d)
    return out

def _shingles(text, n):
    w = text.split()
    if len(w) < n:
        return {tuple(w)}
    return {tuple(w[i:i + n]) for i in range(len(w) - n + 1)}

def jaccard(a, b, n=3):
    A, B = _shingles(a, n), _shingles(b, n)
    return len(A & B) / len(A | B)

def dedup_near(docs, threshold=0.8, n=3):
    kept = []
    for d in docs:
        if all(jaccard(d, k, n) < threshold for k in kept):
            kept.append(d)
    return kept

def split_docs(docs, val_frac, seed):
    rng = np.random.default_rng(seed)
    perm = rng.permutation(len(docs))
    n_val = max(1, int(round(val_frac * len(docs))))
    val = [docs[i] for i in perm[:n_val]]
    train = [docs[i] for i in perm[n_val:]]
    return train, val

def pack(docs_ids, eos, block):
    stream = []
    for d in docs_ids:
        stream.extend(list(d))
        stream.append(eos)
    stream = np.array(stream)
    n = (len(stream) - 1) // block
    x = np.stack([stream[i * block:(i + 1) * block] for i in range(n)])
    y = np.stack([stream[i * block + 1:(i + 1) * block + 1] for i in range(n)])
    return x, y

def shuffle_blocks(x, y, seed):
    perm = np.random.default_rng(seed).permutation(len(x))
    return x[perm], y[perm]

def doc_mask(block_ids, eos):
    ids = np.asarray(block_ids)
    T = len(ids)
    is_eos = (ids == eos).astype(int)
    doc_id = np.concatenate([[0], np.cumsum(is_eos)[:-1]])
    causal = np.tril(np.ones((T, T), dtype=bool))
    same = doc_id[:, None] == doc_id[None, :]
    return causal & same

def padding_fraction(lengths, block):
    slots = sum(-(-l // block) * block for l in lengths)
    return 1 - sum(lengths) / slots

x, y = pack([[1, 2, 3], [4, 5], [6, 7, 8, 9]], eos=0, block=3)
print(x, y, sep="\n")
