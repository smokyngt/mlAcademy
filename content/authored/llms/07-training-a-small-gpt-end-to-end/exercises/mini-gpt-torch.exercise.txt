#@@ prompt
Assemble and train a complete miniature GPT, using the same loop structure as a real pre-training run.

The corpus is generated by a small grammar (`make_corpus`): sentences like `the old fox finds the red bird.` It is a toy, but it has a measurable entropy and **new** sentences in the validation split, so memorising is not enough. Provided for you: the data helpers (`build_vocab`, `encode`, `decode`, `get_batch`), `lr_at`, `eval_loss`, `unigram_loss` (the loss of a model that only knows character frequencies) and `param_count`.

You implement:

- `CausalSelfAttention`, `Block` (pre-norm: LayerNorm -> attention -> residual, LayerNorm -> MLP -> residual) and `GPT` (token + learned position embeddings, `n_layer` blocks, final LayerNorm, LM head **tied** to the token embedding, weights initialised from $N(0, 0.02)$).
- `train(model, train_data, ...)`: AdamW with `betas=(0.9, 0.95)`, `weight_decay=0.1`, the warmup + cosine schedule, gradient clipping to global norm `clip`; returns `[(loss, grad_norm_before_clipping), ...]`.
- `generate(model, start_ids, n_new, temperature, top_k, seed)`: autoregressive sampling with temperature and top-k, cropping the context to `model.ctx`.

The tests train the model for 300 steps (about 20-30 seconds on a laptop CPU) and require the validation loss to beat the unigram baseline by a wide margin.

import math, time
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F

def make_corpus(n_sentences=1500, seed=0):
    rng = np.random.default_rng(seed)
    adj = ["small", "red", "old", "quiet", "bright"]
    noun = ["fox", "bird", "cat", "river", "tree", "child"]
    verb = ["sees", "hears", "follows", "finds"]
    out = []
    for _ in range(n_sentences):
        a1, n1, v, a2, n2 = (rng.choice(adj), rng.choice(noun), rng.choice(verb),
                             rng.choice(adj), rng.choice(noun))
        out.append(f"the {a1} {n1} {v} the {a2} {n2}.")
    return "\n".join(out) + "\n"

def build_vocab(text):
    chars = sorted(set(text))
    stoi = {c: i for i, c in enumerate(chars)}
    return chars, stoi

def encode(text, stoi):
    return torch.tensor([stoi[c] for c in text], dtype=torch.long)

def decode(ids, chars):
    return "".join(chars[int(i)] for i in ids)

class CausalSelfAttention(nn.Module):
    def __init__(self, d, n_head):
        super().__init__()
        self.n_head = n_head
        self.qkv = nn.Linear(d, 3 * d)
        self.proj = nn.Linear(d, d)
    def forward(self, x):
        B, T, D = x.shape
        q, k, v = self.qkv(x).split(D, dim=2)
        q, k, v = (t.view(B, T, self.n_head, D // self.n_head).transpose(1, 2) for t in (q, k, v))
        att = (q @ k.transpose(-2, -1)) / math.sqrt(D // self.n_head)
        mask = torch.tril(torch.ones(T, T, dtype=torch.bool))
        att = att.masked_fill(~mask, float("-inf")).softmax(dim=-1)
        y = (att @ v).transpose(1, 2).reshape(B, T, D)
        return self.proj(y)

class Block(nn.Module):
    def __init__(self, d, n_head):
        super().__init__()
        self.ln1 = nn.LayerNorm(d); self.attn = CausalSelfAttention(d, n_head)
        self.ln2 = nn.LayerNorm(d)
        self.mlp = nn.Sequential(nn.Linear(d, 4 * d), nn.GELU(), nn.Linear(4 * d, d))
    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        return x + self.mlp(self.ln2(x))

class GPT(nn.Module):
    def __init__(self, vocab, ctx, d, n_layer, n_head):
        super().__init__()
        self.ctx = ctx
        self.tok = nn.Embedding(vocab, d)
        self.pos = nn.Embedding(ctx, d)
        self.blocks = nn.ModuleList([Block(d, n_head) for _ in range(n_layer)])
        self.ln_f = nn.LayerNorm(d)
        self.head = nn.Linear(d, vocab, bias=False)
        self.head.weight = self.tok.weight          # weight tying
        self.apply(self._init)
    def _init(self, m):
        if isinstance(m, (nn.Linear, nn.Embedding)):
            nn.init.normal_(m.weight, mean=0.0, std=0.02)
            if isinstance(m, nn.Linear) and m.bias is not None:
                nn.init.zeros_(m.bias)
    def forward(self, idx):
        B, T = idx.shape
        x = self.tok(idx) + self.pos(torch.arange(T))
        for b in self.blocks:
            x = b(x)
        return self.head(self.ln_f(x))

def lr_at(step, max_lr, min_lr, warmup, total):
    if step < warmup:
        return max_lr * (step + 1) / warmup
    p = (step - warmup) / max(1, total - warmup)
    return min_lr + 0.5 * (max_lr - min_lr) * (1 + math.cos(math.pi * min(1.0, p)))

def get_batch(data, B, T, gen):
    ix = torch.randint(0, len(data) - T, (B,), generator=gen)
    x = torch.stack([data[i:i + T] for i in ix])
    y = torch.stack([data[i + 1:i + T + 1] for i in ix])
    return x, y

@torch.no_grad()
def eval_loss(model, data, B, T, n_batches, seed=123):
    model.eval()
    gen = torch.Generator().manual_seed(seed)
    tot = 0.0
    for _ in range(n_batches):
        x, y = get_batch(data, B, T, gen)
        tot += F.cross_entropy(model(x).reshape(-1, model.head.out_features), y.reshape(-1)).item()
    model.train()
    return tot / n_batches

def unigram_loss(train, val, V):
    counts = torch.bincount(train, minlength=V).float() + 1.0
    p = counts / counts.sum()
    return float(-torch.log(p[val]).mean())

def train(model, train_data, steps=300, B=32, T=32, max_lr=3e-3, min_lr=3e-4, warmup=30,
          clip=1.0, seed=0):
    gen = torch.Generator().manual_seed(seed)
    opt = torch.optim.AdamW(model.parameters(), lr=max_lr, betas=(0.9, 0.95), weight_decay=0.1)
    history = []
    for step in range(steps):
        for g in opt.param_groups:
            g["lr"] = lr_at(step, max_lr, min_lr, warmup, steps)
        x, y = get_batch(train_data, B, T, gen)
        loss = F.cross_entropy(model(x).reshape(-1, model.head.out_features), y.reshape(-1))
        opt.zero_grad(set_to_none=True)
        loss.backward()
        gnorm = torch.nn.utils.clip_grad_norm_(model.parameters(), clip)
        opt.step()
        history.append((loss.item(), float(gnorm)))
    return history

@torch.no_grad()
def generate(model, start_ids, n_new, temperature=0.8, top_k=5, seed=0):
    model.eval()
    gen = torch.Generator().manual_seed(seed)
    idx = start_ids.clone().unsqueeze(0)
    for _ in range(n_new):
        logits = model(idx[:, -model.ctx:])[:, -1, :] / temperature
        v, _ = torch.topk(logits, top_k)
        logits[logits < v[:, [-1]]] = float("-inf")
        nxt = torch.multinomial(logits.softmax(-1), 1, generator=gen)
        idx = torch.cat([idx, nxt], dim=1)
    model.train()
    return idx[0]

def param_count(V, ctx, d, L):
    per_block = 12 * d * d + 13 * d
    return V * d + ctx * d + L * per_block + 2 * d
