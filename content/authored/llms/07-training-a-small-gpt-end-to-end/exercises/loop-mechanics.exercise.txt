Implement the small pieces that every language-model training loop is built from.

- `make_windows(tokens, starts, T)` returns `(x, y)`, both of shape `(len(starts), T)`: `x[i] = tokens[s:s+T]` and `y[i] = tokens[s+1:s+T+1]` for `s = starts[i]`.
- `get_batch(tokens, B, T, rng)` draws `B` uniformly random legal start positions with `rng.integers` (every window must have its target inside the stream) and calls `make_windows`.
- `cross_entropy(logits, targets)` takes logits of shape `(N, V)` and integer targets `(N,)`, returns the mean negative log-likelihood in nats, numerically stable. `perplexity(loss)` is `exp(loss)`.
- `lr_at(step, max_lr, min_lr, warmup, total)`: linear warmup `max_lr * (step+1)/warmup` for `step < warmup`, then cosine decay from `max_lr` to `min_lr` at `step = total`.
- `accumulate_gradient(grad_fn, W, X, y, micro)` splits `(X, y)` into consecutive micro-batches of at most `micro` examples, calls `grad_fn(W, Xb, yb)` (which returns the gradient of the **mean** loss on that chunk) and returns the gradient of the mean loss over the whole batch.
- `clip_by_global_norm(grads, max_norm)` takes a list of arrays, returns `(clipped_list, global_norm)`; scale by `min(1, max_norm / (global_norm + 1e-6))`.
- `gpt_param_count(V, ctx, d, L, tied=True)`: GPT-2 style parameters with biases, two LayerNorms per block and a final LayerNorm.

import numpy as np

def make_windows(tokens, starts, T):
    tokens = np.asarray(tokens)
    x = np.stack([tokens[s:s + T] for s in starts])
    y = np.stack([tokens[s + 1:s + T + 1] for s in starts])
    return x, y

def get_batch(tokens, B, T, rng):
    starts = rng.integers(0, len(tokens) - T, size=B)
    return make_windows(tokens, starts, T)

def cross_entropy(logits, targets):
    m = logits.max(axis=1, keepdims=True)
    lse = m[:, 0] + np.log(np.exp(logits - m).sum(axis=1))
    return float(np.mean(lse - logits[np.arange(len(targets)), targets]))

def perplexity(loss):
    return float(np.exp(loss))

def lr_at(step, max_lr, min_lr, warmup, total):
    if step < warmup:
        return max_lr * (step + 1) / warmup
    p = min(1.0, (step - warmup) / max(1, total - warmup))
    return min_lr + 0.5 * (max_lr - min_lr) * (1 + np.cos(np.pi * p))

def softmax_grad(W, X, y):
    Z = X @ W
    Z = Z - Z.max(axis=1, keepdims=True)
    P = np.exp(Z); P /= P.sum(axis=1, keepdims=True)
    P[np.arange(len(y)), y] -= 1
    return X.T @ P / len(y)

def accumulate_gradient(grad_fn, W, X, y, micro):
    N = len(y)
    total = np.zeros_like(W)
    for s in range(0, N, micro):
        xb, yb = X[s:s + micro], y[s:s + micro]
        total += grad_fn(W, xb, yb) * len(yb) / N
    return total

def clip_by_global_norm(grads, max_norm):
    norm = float(np.sqrt(sum((g ** 2).sum() for g in grads)))
    coef = min(1.0, max_norm / (norm + 1e-6))
    return [g * coef for g in grads], norm

def gpt_param_count(V, ctx, d, L, tied=True):
    per_block = 12 * d * d + 13 * d
    total = V * d + ctx * d + L * per_block + 2 * d
    return total if tied else total + V * d
