import numpy as np

def sigmoid(x):
    x = np.asarray(x, dtype=float)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    e = np.exp(x[~pos])
    out[~pos] = e / (1.0 + e)
    return out if out.ndim else float(out)

def pref_prob(r_a, r_b):
    return float(sigmoid(r_a - r_b))

def bt_loss(r_w, r_l):
    d = np.asarray(r_w, dtype=float) - np.asarray(r_l, dtype=float)
    return float(np.mean(np.logaddexp(0.0, -d)))

def bt_grads(r_w, r_l):
    d = np.asarray(r_w, dtype=float) - np.asarray(r_l, dtype=float)
    g = -sigmoid(-d) / d.size
    return g, -g

def train_reward(phi_w, phi_l, lr=0.5, steps=500):
    theta = np.zeros(phi_w.shape[1])
    diff = phi_w - phi_l
    for _ in range(steps):
        d = diff @ theta
        dd = -sigmoid(-d) / len(d)
        theta = theta - lr * (diff.T @ dd)
    return theta

def pair_accuracy(theta, phi_w, phi_l):
    return float(np.mean(phi_w @ theta > phi_l @ theta))

print(pref_prob(1.5, 0.5))
