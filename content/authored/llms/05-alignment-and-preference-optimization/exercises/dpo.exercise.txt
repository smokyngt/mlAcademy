import numpy as np

def _sig(x):
    return 0.5 * (1.0 + np.tanh(0.5 * np.asarray(x, dtype=float)))

def rlhf_objective(pi, ref, r, beta):
    pi, ref, r = map(lambda v: np.asarray(v, dtype=float), (pi, ref, r))
    mask = pi > 0
    kl = np.sum(pi[mask] * np.log(pi[mask] / ref[mask]))
    return float(np.sum(pi * r) - beta * kl)

def optimal_policy(ref, r, beta):
    ref, r = np.asarray(ref, dtype=float), np.asarray(r, dtype=float)
    w = np.log(ref) + r / beta
    w = w - w.max()
    p = np.exp(w)
    return p / p.sum()

def dpo_loss(lp_w, lp_l, ref_w, ref_l, beta):
    z = beta * ((np.asarray(lp_w) - np.asarray(ref_w)) - (np.asarray(lp_l) - np.asarray(ref_l)))
    return float(np.mean(np.logaddexp(0.0, -z)))

def dpo_grads(lp_w, lp_l, ref_w, ref_l, beta):
    z = beta * ((np.asarray(lp_w) - np.asarray(ref_w)) - (np.asarray(lp_l) - np.asarray(ref_l)))
    g = -beta * _sig(-z) / z.size
    return g, -g

def implicit_reward(lp, ref, beta):
    return beta * (np.asarray(lp) - np.asarray(ref))

def train_dpo_tabular(ref_logp, pairs, beta=1.0, lr=0.5, steps=100):
    theta = np.array(ref_logp, dtype=float)
    w = np.array([p[0] for p in pairs])
    l = np.array([p[1] for p in pairs])
    ref_logp = np.asarray(ref_logp, dtype=float)
    for _ in range(steps):
        m = theta.max()
        logp = theta - (m + np.log(np.exp(theta - m).sum()))
        pi = np.exp(logp)
        gw, gl = dpo_grads(logp[w], logp[l], ref_logp[w], ref_logp[l], beta)
        grad = np.zeros_like(theta)
        for i in range(len(pairs)):
            grad[w[i]] += gw[i]
            grad[l[i]] += gl[i]
            grad -= (gw[i] + gl[i]) * pi
        theta = theta - lr * grad
    return theta

print(dpo_loss(np.array([-12.0]), np.array([-15.0]), np.array([-13.0]), np.array([-14.0]), 0.1))
