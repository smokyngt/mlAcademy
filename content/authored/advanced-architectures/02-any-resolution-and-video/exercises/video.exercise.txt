Implement a token and memory calculator for vision-language inputs (NumPy only).

1. `native_tokens(h, w, patch=14, merge=2, max_tokens=None)`: tokens for an image at native aspect ratio when `merge x merge` neighbouring patches are fused. Let `u = patch*merge`. Round each side to the nearest multiple of `u` (at least `u`), tokens = `(h2//u) * (w2//u)`. If `max_tokens` is given and exceeded, scale both sides down by `sqrt(max_tokens/tokens)`, floor each to a multiple of `u` (at least `u`) and recompute.
2. `packed_mask(lengths)`: boolean `(S, S)` block-diagonal mask, `S = sum(lengths)`, True where both indices belong to the same image.
3. `attention_entries(lengths)`: returns `(packed, padded)`: number of score entries when packing (sum of squares) vs padding every image to the longest (`n * max**2`).
4. `video_tokens(duration_s, fps, tokens_per_frame)`: total visual tokens.
5. `pooled_tokens(grid_h, grid_w, pool)`: tokens per frame after `pool x pool` average pooling of a `grid_h x grid_w` token grid (ceil division).
6. `kv_gib(layers, kv_heads, head_dim, n_tokens, bytes_per_value=2)`: KV cache size in GiB (2**30 bytes).
7. `max_frames(context, text_tokens, tokens_per_frame)`: how many frames fit in the context window next to the text.
8. `mrope_positions(t, h, w)`: array `(t*h*w, 3)` of `(time, row, col)` integer positions in row-major order within each frame, frames in time order.

import numpy as np

def _round_to(x, u):
    return max(u, int(round(x / u)) * u)

def native_tokens(h, w, patch=14, merge=2, max_tokens=None):
    u = patch * merge
    h2, w2 = _round_to(h, u), _round_to(w, u)
    n = (h2 // u) * (w2 // u)
    if max_tokens is not None and n > max_tokens:
        s = np.sqrt(max_tokens / n)
        h2 = max(u, int(np.floor(h2 * s / u)) * u)
        w2 = max(u, int(np.floor(w2 * s / u)) * u)
        n = (h2 // u) * (w2 // u)
    return n

def packed_mask(lengths):
    S = int(sum(lengths))
    M = np.zeros((S, S), dtype=bool)
    o = 0
    for l in lengths:
        M[o:o + l, o:o + l] = True
        o += l
    return M

def attention_entries(lengths):
    return int(sum(l * l for l in lengths)), len(lengths) * max(lengths) ** 2

def video_tokens(duration_s, fps, tokens_per_frame):
    return int(duration_s * fps) * tokens_per_frame

def pooled_tokens(grid_h, grid_w, pool):
    return -(-grid_h // pool) * -(-grid_w // pool)

def kv_gib(layers, kv_heads, head_dim, n_tokens, bytes_per_value=2):
    return 2 * layers * kv_heads * head_dim * n_tokens * bytes_per_value / 2 ** 30

def max_frames(context, text_tokens, tokens_per_frame):
    return (context - text_tokens) // tokens_per_frame

def mrope_positions(t, h, w):
    tt, rr, cc = np.meshgrid(np.arange(t), np.arange(h), np.arange(w), indexing="ij")
    return np.stack([tt.ravel(), rr.ravel(), cc.ravel()], axis=1)
