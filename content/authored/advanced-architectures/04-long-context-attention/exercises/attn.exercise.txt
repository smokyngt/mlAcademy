#@@ prompt
Implement attention masks and memory counters with NumPy. A boolean mask has `True` where a query (row) may attend to a key (column).

1. `causal_mask(n)`, `sliding_mask(n, window)` (query `i` sees keys `j` with `i-window < j <= i`), `strided_mask(n, window, stride)` (sliding window plus every key `j <= i` with `j % stride == 0`), `global_local_mask(n, window, global_idx)` (sliding window, plus every query may see the keys in `global_idx` that are `<= i`, plus the global positions themselves see all keys `<= their own index`; the result stays causal).
2. `n_entries(mask)` -> number of allowed pairs.
3. `gqa_attention(Q, K, V, mask)`: `Q` is `(H, N, d)`, `K` and `V` are `(G, N, d)` with `H % G == 0`; query head `h` uses KV head `h // (H // G)`. Return `(H, N, d)`. (G = H is multi-head, G = 1 is multi-query.)
4. `diff_attention(Q1, K1, Q2, K2, V, lam, mask)`: single head, `(N, d)` inputs. Return `(softmax(Q1 K1^T/sqrt(d)) - lam * softmax(Q2 K2^T/sqrt(d))) @ V` with the mask applied inside each softmax.
5. `kv_gib(layers, kv_heads, head_dim, n_tokens, window=None, bytes_per_value=2)`: KV cache in GiB; if `window` is given each layer keeps only `min(n_tokens, window)` tokens.
6. `mla_kv_gib(layers, latent_dim, n_tokens, rope_dim=0, bytes_per_value=2)`: MLA cache in GiB: per token per layer `latent_dim + rope_dim` values.
7. `hybrid_kv_gib(layers, global_every, n_tokens, window, kv_heads, head_dim, bytes_per_value=2)`: every `global_every`-th layer (count `layers // global_every`) is full attention, all others sliding-window.
8. `receptive_field(window, layers)` -> how many positions back information can travel through `layers` stacked sliding-window layers: `layers*(window-1)+1`.

import numpy as np

def causal_mask(n):
    return np.tril(np.ones((n, n), dtype=bool))

def sliding_mask(n, window):
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    return (j <= i) & (i - j < window)

def strided_mask(n, window, stride):
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    return sliding_mask(n, window) | ((j <= i) & (j % stride == 0))

def global_local_mask(n, window, global_idx):
    m = sliding_mask(n, window)
    c = causal_mask(n)
    for g in global_idx:
        m[:, g] = True
        m[g, :] = True
    return m & c

def n_entries(mask):
    return int(mask.sum())

def _softmax(s):
    s = s - s.max(axis=-1, keepdims=True)
    e = np.exp(s)
    return e / e.sum(axis=-1, keepdims=True)

def gqa_attention(Q, K, V, mask):
    H, N, d = Q.shape
    G = K.shape[0]
    out = np.zeros_like(Q)
    for h in range(H):
        g = h // (H // G)
        s = Q[h] @ K[g].T / np.sqrt(d)
        s = np.where(mask, s, -np.inf)
        out[h] = _softmax(s) @ V[g]
    return out

def diff_attention(Q1, K1, Q2, K2, V, lam, mask):
    d = Q1.shape[1]
    a1 = _softmax(np.where(mask, Q1 @ K1.T / np.sqrt(d), -np.inf))
    a2 = _softmax(np.where(mask, Q2 @ K2.T / np.sqrt(d), -np.inf))
    return (a1 - lam * a2) @ V

def kv_gib(layers, kv_heads, head_dim, n_tokens, window=None, bytes_per_value=2):
    kept = n_tokens if window is None else min(n_tokens, window)
    return 2 * layers * kv_heads * head_dim * kept * bytes_per_value / 2 ** 30

def mla_kv_gib(layers, latent_dim, n_tokens, rope_dim=0, bytes_per_value=2):
    return layers * (latent_dim + rope_dim) * n_tokens * bytes_per_value / 2 ** 30

def hybrid_kv_gib(layers, global_every, n_tokens, window, kv_heads, head_dim, bytes_per_value=2):
    n_global = layers // global_every
    n_local = layers - n_global
    return (kv_gib(n_global, kv_heads, head_dim, n_tokens, None, bytes_per_value)
            + kv_gib(n_local, kv_heads, head_dim, n_tokens, window, bytes_per_value))

def receptive_field(window, layers):
    return layers * (window - 1) + 1
