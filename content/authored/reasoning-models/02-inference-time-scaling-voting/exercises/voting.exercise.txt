import numpy as np
from collections import Counter
from math import comb, factorial


def plurality_vote(answers):
    counts = Counter(answers)
    top = max(counts.values())
    winners = [a for a, k in counts.items() if k == top]
    return (winners[0] if len(winners) == 1 else None), counts


def vote_with_tiebreak(answers, scores):
    _, counts = plurality_vote(answers)
    top = max(counts.values())
    tied = sorted(a for a, k in counts.items() if k == top)
    if len(tied) == 1:
        return tied[0]
    def mean_score(a):
        s = [sc for ans, sc in zip(answers, scores) if ans == a]
        return sum(s) / len(s)
    best = max(mean_score(a) for a in tied)
    return [a for a in tied if mean_score(a) == best][0]


def best_of_n(answers, scores):
    best = 0
    for i, s in enumerate(scores):
        if s > scores[best]:
            best = i
    return answers[best]


def majority_acc_binary(n, p, tie="random"):
    total = 0.0
    for c in range(n + 1):
        pr = comb(n, c) * p ** c * (1 - p) ** (n - c)
        if 2 * c > n:
            total += pr
        elif 2 * c == n:
            total += pr * (0.5 if tie == "random" else 0.0)
    return total


def _compositions(total, parts):
    if parts == 1:
        yield (total,)
        return
    for first in range(total + 1):
        for rest in _compositions(total - first, parts - 1):
            yield (first,) + rest


def plurality_acc(n, p, m, tie="random"):
    total = 0.0
    for c in range(n + 1):
        pc = comb(n, c) * p ** c * (1 - p) ** (n - c)
        wrong = n - c
        for comp in _compositions(wrong, m):
            coef = factorial(wrong)
            for x in comp:
                coef //= factorial(x)
            pr = pc * coef * (1.0 / m) ** wrong
            wmax = max(comp)
            if c > wmax:
                win = 1.0
            elif c == wmax:
                t = sum(1 for x in comp if x == c)
                win = 1.0 / (1 + t) if tie == "random" else 0.0
            else:
                win = 0.0
            total += pr * win
    return total


def mixture_curve(ps, ns, m, tie="random"):
    out = []
    for n in ns:
        if m == 1:
            out.append(float(np.mean([majority_acc_binary(n, p, tie) for p in ps])))
        else:
            out.append(float(np.mean([plurality_acc(n, p, m, tie) for p in ps])))
    return out


def accuracy_under_budget(lengths, correct, budget):
    return sum(1 for L, c in zip(lengths, correct) if c and L <= budget) / len(lengths)


def marginal_gain(accs, costs):
    return [(accs[i + 1] - accs[i]) / (costs[i + 1] - costs[i]) for i in range(len(accs) - 1)]
