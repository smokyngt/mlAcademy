#@@ prompt
Hard distillation is supervised fine-tuning on a teacher's traces. Implement the data and loss pieces on a toy vocabulary (no real model). Provided: `ToyTokenizer` (a whitespace tokenizer with growing vocabulary; id 0 is the end-of-sequence token) and `is_correct(content, gt)` (last boxed answer compared as a number or text).

- `format_trace(thinking, content)` -> `"<think>" + thinking.strip() + "</think>\n\n" + content.strip()`; raise `ValueError` if `content` is empty after stripping.
- `keep_trace(entry, tok, max_len)`: keep a teacher trace only if (1) its `message_content` is non-empty, (2) `is_correct(entry["message_content"], entry["gtruth_answer"])`, and (3) the total token count of prompt + formatted answer + 1 end token is at most `max_len`. `entry` has keys `problem`, `message_thinking`, `message_content`, `gtruth_answer`.
- `build_example(entry, tok)` -> `{"token_ids": prompt_ids + answer_ids + [tok.eos_id], "prompt_len": len(prompt_ids)}` where the prompt is `entry["problem"]` and the answer is `format_trace(...)`.
- `sft_loss(logits, token_ids, prompt_len)`: mean cross-entropy over the **answer** tokens only; `logits` has shape `(L-1, V)` for `L = len(token_ids)`.
- `kl_div(teacher_logits, student_logits)`: mean over rows of `KL(softmax(teacher) || softmax(student))` (soft distillation).
- `distill_loss(student_logits, teacher_logits, token_ids, prompt_len, lam)`: `sft_loss + lam * KL`, with the KL averaged over the **answer** rows only.
- `left_pad(seqs, pad_id)` -> `(ids, pad_mask)`: arrays of shape `(B, max_len)`; `pad_mask` is True on pad cells.
- `batched_padding_waste(lengths, batch_size, sort=False)`: fraction of cells that are padding when the sequences are cut into consecutive batches of `batch_size` (optionally after sorting by length), each batch padded to its longest member.
- `speedup(t_single, t_batched)`: ratio of run times.

import re
import numpy as np
import random
from fractions import Fraction


class ToyTokenizer:
    eos_id = 0

    def __init__(self):
        self.vocab = {}

    def encode(self, text):
        ids = []
        for w in text.split():
            if w not in self.vocab:
                self.vocab[w] = len(self.vocab) + 1
            ids.append(self.vocab[w])
        return ids


def is_correct(content, gt):
    m = re.findall(r"\\boxed\{([^{}]*)\}", content)
    if not m:
        return False
    def val(s):
        s = s.strip()
        try:
            return Fraction(s)
        except (ValueError, ZeroDivisionError):
            return s
    return val(m[-1]) == val(gt)


def format_trace(thinking, content):
    content = str(content).strip()
    if not content:
        raise ValueError("empty content")
    return "<think>" + str(thinking).strip() + "</think>\n\n" + content


def build_example(entry, tok):
    prompt_ids = tok.encode(entry["problem"])
    answer_ids = tok.encode(format_trace(entry["message_thinking"], entry["message_content"]))
    return {"token_ids": prompt_ids + answer_ids + [tok.eos_id], "prompt_len": len(prompt_ids)}


def keep_trace(entry, tok, max_len):
    content = str(entry.get("message_content", "")).strip()
    if not content:
        return False
    if not is_correct(content, entry["gtruth_answer"]):
        return False
    return len(build_example(entry, tok)["token_ids"]) <= max_len


def _log_softmax(z):
    z = np.asarray(z, float)
    z = z - z.max(axis=-1, keepdims=True)
    return z - np.log(np.exp(z).sum(axis=-1, keepdims=True))


def sft_loss(logits, token_ids, prompt_len):
    lp = _log_softmax(logits)
    ids = np.asarray(token_ids)
    rows = np.arange(max(prompt_len - 1, 0), len(ids) - 1)
    return float(-np.mean(lp[rows, ids[rows + 1]]))


def kl_div(teacher_logits, student_logits):
    lt, ls = _log_softmax(teacher_logits), _log_softmax(student_logits)
    return float(np.mean(np.sum(np.exp(lt) * (lt - ls), axis=-1)))


def distill_loss(student_logits, teacher_logits, token_ids, prompt_len, lam):
    start = max(prompt_len - 1, 0)
    kl = kl_div(np.asarray(teacher_logits)[start:], np.asarray(student_logits)[start:])
    return sft_loss(student_logits, token_ids, prompt_len) + lam * kl


def left_pad(seqs, pad_id):
    m = max(len(s) for s in seqs)
    ids = np.array([[pad_id] * (m - len(s)) + list(s) for s in seqs])
    mask = np.array([[True] * (m - len(s)) + [False] * len(s) for s in seqs])
    return ids, mask


def batched_padding_waste(lengths, batch_size, sort=False):
    lengths = sorted(lengths) if sort else list(lengths)
    cells = real = 0
    for i in range(0, len(lengths), batch_size):
        b = lengths[i:i + batch_size]
        cells += max(b) * len(b)
        real += sum(b)
    return (cells - real) / cells


def speedup(t_single, t_batched):
    return t_single / t_batched
